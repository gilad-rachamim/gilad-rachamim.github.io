השכמת הבוקר

מודה אני, נטילת ידיים, אשר יצר, אלקי נשמה

מוֹדֶה אֲנִי

מיד כשמתעוררים, עוד במיטה. אין בו שם ה', ולכן אומרים אותו גם לפני נטילת ידיים.

מוֹדֶה אֲנִי לְפָנֶיךָ, מֶלֶךְ חַי וְקַיָּם, שֶׁהֶחֱזַרְתָּ בִּי נִשְׁמָתִי בְּחֶמְלָה, רַבָּה אֱמוּנָתֶךָ.

אישה אומרת: מוֹדָה אֲנִי.

נְטִילַת יָדַיִם

נוטלים מכלי, שלוש פעמים על כל יד לסירוגין: ימין, שמאל, ימין, שמאל. מנגבים ומברכים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל נְטִילַת יָדָיִם.

אֲשֶׁר יָצַר

אחרי היציאה מהשירותים. בבוקר אומרים אותה מיד אחרי ברכת נטילת ידיים.

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר יָצַר אֶת־הָאָדָם בְּחָכְמָה וּבָרָא בוֹ נְקָבִים נְקָבִים חֲלוּלִים חֲלוּלִים גָּלוּי וְיָדוּעַ לִפְנֵי כִסֵּא כְבוֹדֶךָ שֶׁאִם יִפָּתֵחַ אֶחָד מֵהֶם אוֹ יִסָּתֵם אֶחָד מֵהֶם אִי אֶפְשַׁר לְהִתְקַיֵּם וְלַעֲמֹד לְפָנֶיךָ אֲפִילוּ שָׁעָה אֶחָת. בָּרוּךְ אַתָּה ה' רוֹפֵא כָל־בָּשָׂר וּמַפְלִיא לַעֲשׂוֹת:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר יָצַר אֶת הָאָדָם בְּחָכְמָה, וּבָרָא בוֹ נְקָבִים נְקָבִים, חֲלוּלִים חֲלוּלִים, גָּלוּי וְיָדוּעַ לִפְנֵי כִסֵּא כְבוֹדֶךָ, שֶׁאִם יִסָּתֵם אֶחָד מֵהֶם, אוֹ אִם יִפָּתֵח אֶחָד מֵהֶם, אֵי אֶפְשָׁר לְהִתְקַיֵּם אַפִלּוּ שָׁעָה אֶחָת. בָּרוּךְ אַתָּה ה', רוֹפֵא כָל־בָּשָׂר וּמַפְלִיא לַעֲשׂוֹת:

אֱלֹקַי נְשָׁמָה

אֱלֹקַי, נְשָׁמָה שֶׁנָּתַתָּ בִּי טְהוֹרָה, אַתָּה בְרָאתָהּ, אַתָּה יְצַרְתָּהּ, אַתָּה נְפַחְתָּהּ בִּי, וְאַתָּה מְשַׁמְּרָהּ בְּקִרְבִּי, וְאַתָּה עָתִיד לִטְּלָהּ מִמֶּנִּי, וּלְהַחֲזִירָהּ בִּי לֶעָתִיד לָבוֹא, כָּל־זְמַן שֶׁהַנְּשָׁמָה בְקִרְבִּי, מוֹדֶה אֲנִי לְפָנֶיךָ ה' אֱלֹקַי וֵאלֹקֵי אֲבוֹתַי, רִבּוֹן כָּל־הַמַּעֲשִׂים אֲדוֹן כָּל־הַנְּשָׁמוֹת. בָּרוּךְ אַתָּה ה', הַמַּחֲזִיר נְשָׁמוֹת לִפְגָרִים מֵתִים:

אישה אומרת: מוֹדָה אֲנִי לְפָנֶיךָ.

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם שֶׁלֹּא עָשַׂנִי גּוֹי:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם שֶׁלֹּא עָשַׂנִי עָבֶד:

האיש מברך:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם שֶׁלֹּא עָשַׂנִי אִשָּׁה:

האישה מברכת במקומה: בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁעָשַׂנִי כִּרְצוֹנוֹ.

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם פּוֹקֵחַ עִוְרִים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם מַלְבִּישׁ עֲרֻמִּים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם מַתִּיר אֲסוּרִים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם זוֹקֵף כְּפוּפִים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם רוֹקַע הָאָרֶץ עַל הַמָּיִם:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם שֶׁעָשָׂה לִי כָּל־צָרְכִּי:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם הַמֵּכִין מִצְעֲדֵי גָבֶר:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם אוֹזֵר יִשְׂרָאֵל בִּגְבוּרָה:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם עוֹטֵר יִשְׂרָאֵל בְּתִפְאָרָה:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם הַנּוֹתֵן לַיָּעֵף כֹּחַ:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם הַמַּעֲבִיר שֵׁנָה מֵעֵינָי וּתְנוּמָה מֵעַפְעַפָּי:

וִיהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ שֶׁתַּרְגִּילֵנוּ בְּתוֹרָתֶךָ וְדַבְּקֵנוּ בְּמִצְוֹתֶיךָ, וְאַל תְּבִיאֵנוּ לֹא לִידֵי חֵטְא וְלֹא לִידֵי עֲבֵרָה וְעָוֹן וְלֹא לִידֵי נִסָּיוֹן וְלֹא לִידֵי בִזָּיוֹן וְאַל יִשְׁלֹט בָּנוּ יֵצֶר הָרָע וְהַרְחִיקֵנוּ מֵאָדָם רָע וּמֵחָבֵר רָע וְדַבְּקֵנוּ בְּיֵצֶר הַטּוֹב וּבְמַעֲשִׂים טוֹבִים וְכוֹף אֶת־יִצְרֵנוּ לְהִשְׁתַּעְבֶּד־לָךְ וּתְנֵנוּ הַיּוֹם וּבְכָל־יוֹם לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל־רוֹאֵינוּ וְתִגְמְלֵנוּ חֲסָדִים טוֹבִים: בָּרוּךְ אַתָּה ה' גּוֹמֵל חֲסָדִים טוֹבִים לְעַמּוֹ יִשְׂרָאֵל:

יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹקַי וֵאלֹקֵי אֲבוֹתַי שֶׁתַּצִּילֵנִי הַיּוֹם וּבְכָל־יוֹם מֵעַזֵּי פָנִים וּמֵעַזּוּת פָּנִים מֵאָדָם רָע וּמֵחָבֵר רָע וּמִשָּׁכֵן רָע וּמִפֶּגַע רָע וּמִשָּׂטָן הַמַּשְׁחִית מִדִּין קָשֶׁה וּמִבַּעַל דִּין קָשֶׁה, בֵּין שֶׁהוּא בֶן בְּרִית וּבֵין שֶׁאֵינוֹ בֶן בְּרִית:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הַנּוֹתֵן לַשֶּׂכְוִי בִינָה, לְהַבְחִין בֵּין יוֹם וּבֵין לָיְלָה:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, פּוֹקֵחַ עִוְרִים:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, מַתִּיר אֲסוּרִים:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, זוֹקֵף כְּפוּפִים:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, מַלְבִּישׁ עֲרֻמִּים:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הַנּוֹתֵן לַיָּעֵף כֹּחַ:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, רוֹקַע הָאָרֶץ עַל הַמָּיִם:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הַמֵּכִין מִצְעֲדֵי גָבֶר:

בתשעה באב וביום הכיפורים לא אומרים את הברכה הבאה.

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁעָשָׂה לִי כָּל־צָרְכִּי:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אוֹזֵר יִשְׂרָאֵל בִּגְבוּרָה:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, עוֹטֵר יִשְׂרָאֵל בְּתִפְאָרָה:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁלֹּא עָשַׂנִי גוֹי:

אישה אומרת: בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁלֹּא עָשַׂנִי גּוֹיָה.

אישה אומרת: בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁלֹּא עָשַׂנִי שִׁפְחָה.

האיש מברך:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁלֹּא עָשַׂנִי אִשָּׁה:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הַמַּעֲבִיר חֶבְלֵי שֵׁנָה מֵעֵינָי וּתְנוּמָה מֵעַפְעַפָּי. וִיהִי רָצוֹן מִלְּפָנֶיךָ ה', אֱלֹקַי וֵאלֹקֵי אֲבוֹתַי, שֶׁתַּרְגִּילֵנִי בְּתוֹרָתֶךָ, וְתַדְבִּיקֵנִי בְּמִצְוֹתֶיךָ, וְאַל תְּבִיאֵנִי לִידֵי חֵטְא, וְלֹא לִידֵי עָוֹן, וְלֹא לִידֵי נִסָּיוֹן, וְלֹא לִידֵי בִזָּיוֹן, וְתַרְחִיקֵנִי מִיֵּצֶר הָרָע, וְתַדְבִּיקֵנִי בְּיֵצֶר הַטּוֹב, וְכוֹף אֶת־יִצְרִי לְהִשְׁתַּעְבֶּד לָּךְ, וּתְנֵנִי הַיּוֹם וּבְכָל־יוֹם לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל־רוֹאַי, וְגָמְלֵנִי חֲסָדִים טוֹבִים. בָּרוּךְ אַתָּה ה', גּוֹמֵל חֲסָדִים טוֹבִים לְעַמּוֹ יִשְׂרָאֵל:

1. תפילין של יד

מניחים על השריר של הזרוע ביד שמאל (מי שכותב ביד שמאל, מניח ביד ימין), כשהבית נוטה מעט כלפי הלב. לפני שמהדקים את הרצועה מברכים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לְהָנִיחַ תְּפִלִּין.

2. תפילין של ראש

מהדקים, וכורכים שבע כריכות סביב הזרוע. מניחים את של ראש מעל המצח, במקום שבו מתחילות השערות, באמצע הראש. לפני ההידוק מברכים:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל מִצְוַת תְּפִלִּין.

ומיד אחרי ההידוק:

בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד.

מהדקים, כורכים שבע כריכות סביב הזרוע, ומניחים מיד את של ראש מעל המצח, במקום שבו מתחילות השערות. הברכה הראשונה פוטרת את שתיהן, ולכן לא מדברים ולא מפסיקים ביניהן.

רק מי שדיבר או הפסיק בין של יד לשל ראש, מברך על של ראש:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל מִצְוַת תְּפִלִּין.

3. הכריכות על האצבע

אחרי של ראש גומרים את של יד: כורכים את הרצועה על האצבע האמצעית, כריכה אחת בפרק האמצעי ושתיים בפרק התחתון, ואומרים:

וְאֵרַשְׂתִּיךְ לִי לְעוֹלָם, וְאֵרַשְׂתִּיךְ לִי בְּצֶדֶק וּבְמִשְׁפָּט וּבְחֶסֶד וּבְרַחֲמִים: וְאֵרַשְׂתִּיךְ לִי בֶּאֱמוּנָה, וְיָדַעַתְּ אֶת־ה':

4. הפרשיות

אחרי ההנחה אומרים את פרשיות "קדש" ו"והיה כי יביאך", כך שבכל יום נאמרות כל ארבע הפרשיות שכתובות בתפילין (את "שמע" ו"והיה אם שמוע" אומרים בקריאת שמע). מי שאין לו זמן לפני התפילה, אומר אותן אחריה.

בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד.

וְאָהַבְתָּ אֵת ה' אֱלֹקֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשְׁךָ וּבְכָל־מְאֹדֶךָ: וְהָיוּ הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם עַל־לְבָבֶךָ: וְשִׁנַּנְתָּם לְבָנֶיךָ וְדִבַּרְתָּ בָּם בְּשִׁבְתְּךָ בְּבֵיתֶךָ וּבְלֶכְתְּךָ בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ: וּקְשַׁרְתָּם לְאוֹת עַל־יָדֶךָ וְהָיוּ לְטֹטָפֹת בֵּין עֵינֶיךָ: וּכְתַבְתָּם עַל־מְזֻזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ:

וְהָיָה אִם שָׁמֹעַ דברים יא, יג–כא

וְהָיָה אִם־שָׁמֹעַ תִּשְׁמְעוּ אֶל־מִצְוֹתַי אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם לְאַהֲבָה אֶת־ה' אֱלֹקֵיכֶם וּלְעָבְדוֹ בְּכָל־לְבַבְכֶם וּבְכָל־נַפְשְׁכֶם: וְנָתַתִּי מְטַר־אַרְצְכֶם בְּעִתּוֹ יוֹרֶה וּמַלְקוֹשׁ וְאָסַפְתָּ דְגָנֶךָ וְתִירֹשְׁךָ וְיִצְהָרֶךָ: וְנָתַתִּי עֵשֶׂב בְּשָׂדְךָ לִבְהֶמְתֶּךָ וְאָכַלְתָּ וְשָׂבָעְתָּ: הִשָּׁמְרוּ לָכֶם פֶּן יִפְתֶּה לְבַבְכֶם וְסַרְתֶּם וַעֲבַדְתֶּם אֱלֹקִים אֲחֵרִים וְהִשְׁתַּחֲוִיתֶם לָהֶם: וְחָרָה אַף־ה' בָּכֶם וְעָצַר אֶת־הַשָּׁמַיִם וְלֹא־יִהְיֶה מָטָר וְהָאֲדָמָה לֹא תִתֵּן אֶת־יְבוּלָהּ וַאֲבַדְתֶּם מְהֵרָה מֵעַל הָאָרֶץ הַטֹּבָה אֲשֶׁר ה' נֹתֵן לָכֶם: וְשַׂמְתֶּם אֶת־דְּבָרַי אֵלֶּה עַל־לְבַבְכֶם וְעַל־נַפְשְׁכֶם וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל־יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם: וְלִמַּדְתֶּם אֹתָם אֶת־בְּנֵיכֶם לְדַבֵּר בָּם בְּשִׁבְתְּךָ בְּבֵיתֶךָ וּבְלֶכְתְּךָ בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ: וּכְתַבְתָּם עַל־מְזוּזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ: לְמַעַן יִרְבּוּ יְמֵיכֶם וִימֵי בְנֵיכֶם עַל הָאֲדָמָה אֲשֶׁר נִשְׁבַּע ה' לַאֲבֹתֵיכֶם לָתֵת לָהֶם כִּימֵי הַשָּׁמַיִם עַל־הָאָרֶץ:

השמות הקדושים כתובים בכינוי (ה', אלקינו), ואומרים אותם כרגיל. הנוסח לפי סידור אשכנז וסידור עדות המזרח (מתוך ספריא). נוסח ספרד קרוב מאוד לאשכנז בחלקים האלה. מי שנוהג אחרת, הולך לפי המנהג שלו; ובכל שאלה, שואלים את הרב.

שימו בטלפון, ליד התפילין. בתפריט הדפדפן: "הוספה למסך הבית". הדף נפתח כמו אפליקציה ועובד גם בלי אינטרנט.

לשלוח לחבר בווטסאפ

ברכות השחר בטלפון, בנוסח שלכם

כל סדר הבוקר בדף אחד: מודה אני, נטילת ידיים, אשר יצר, אלקי נשמה, ברכות השחר, ברכות התורה, עטיפת טלית והנחת תפילין שלב אחרי שלב. בלי סידור ביד, בלי לחפש עמודים, ובלי פרסומות באמצע הברכה.

נוסח אשכנז וספרד או עדות המזרח

בוחרים נוסח פעם אחת למעלה, והדף זוכר אותו. ההבדלים בין הנוסחים קטנים, בעיקר בניסוח ובסדר של כמה ברכות, וכל אחד ממשיך במנהג אבותיו.

אות גדולה ומסך שלא נכבה

כפתורי א+ ו-א- מגדילים את האותיות, ויש מצב לילה. המסך לא נכבה באמצע הברכות, כך שאפשר להניח את הטלפון ליד התפילין ולקרוא בשקט. אחרי הכניסה הראשונה הדף עובד גם בלי אינטרנט, טוב לבסיס, לטיסה ולשטח.

הנחת תפילין שלב אחרי שלב

מה מברכים, מתי מניחים את של יד ומתי את של ראש, והליפופים על האצבעות, בסדר ברור. בסוף הדף נמצאות ארבע הפרשיות שכתובות בתפילין. מי שרוצה דף בגודל כיס עם תפילת הדרך וקריאת שמע שעל המיטה ימצא אותו בדף לחייל.

שאלות ותשובות

מה אומרים בבוקר כשמתעוררים?

מודה אני, עוד במיטה. אין בו שם ה', ולכן אומרים אותו גם לפני נטילת ידיים. אחר כך נוטלים ידיים, ומברכים על נטילת ידיים ואשר יצר.

אישה מברכת ברכות השחר?

כן. במקום "שלא עשני אישה" אישה מברכת "שעשני כרצונו", ובמודה אני היא אומרת "מודה אני". שני הנוסחים מופיעים בדף.

מה קודם, טלית או תפילין?

קודם מתעטפים בטלית ואחר כך מניחים תפילין, כמו הסדר בדף. בכל שאלה למעשה הולכים לפי המנהג שלכם והרב.

הדף עובד בלי אינטרנט?

כן. אחרי שנכנסים פעם אחת, הדף נשמר בטלפון ונפתח גם בלי קליטה. כדאי להוסיף אותו למסך הבית, ואז הוא נפתח כמו אפליקציה.

זה בחינם?

כן. בלי תשלום, בלי הרשמה ובלי פרסומות.
