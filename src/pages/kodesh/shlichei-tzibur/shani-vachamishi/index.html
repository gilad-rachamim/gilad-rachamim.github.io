קריאת התורה שני וחמישי נוסח מרוקו

הסדר של שני וחמישי, מהוצאת הספר ועד החזרתו להיכל, בנוסח מרוקו כפי שבסידור.

[סוף חצי קדיש, ממשיך מעמ' 130] וְיִשְׁתַּבַּח, וְיִתְפָּאַר, וְיִתְרוֹמַם, וְיִתְנַשֵּׂא, וְיִתְהַדָּר, וְיִתְעַלֶּה וְיִתְהַלָּל, שְׁמֵיהּ דְּקֻדְשָׁא בְּרִיךְ־הוּא (עונים: אמן). לְעֵלָּא מִן כָּל־בִּרְכָתָא, שִׁירָתָא, תִּשְׁבְּחָתָא וְנֶחֱמָתָא, דַּאֲמִירָן בְּעָלְמָא, וְאִמְרוּ אָמֵן (עונים: אמן).

הוֹצָאַת סֵפֶר תּוֹרָה

[הוראה בסידור] בימים שאומרים תחנון, יאמרו מעומד לפני הוצאת ספר תורה:

אֵל אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וֶאֱמֶת, אַל־בְּאַפְּךָ תוֹכִיחֵנוּ. חוּסָה יְיָ עַל יִשְׂרָאֵל עַמֶּךָ, וְהוֹשִׁיעֵנוּ מִכָּל־רָע. חָטָאנוּ לְךָ אָדוֹן, סְלַח נָא כְּרֹב רַחֲמֶיךָ אֵל:

אֵל אֶרֶךְ אַפַּיִם וּמָלֵא רַחֲמִים, אַל תַּסְתֵּר פָּנֶיךָ מִמֶּנּוּ. חוּסָה יְיָ עַל שְׁאֵרִית יִשְׂרָאֵל עַמֶּךָ, וְהַצִּילֵנוּ מִכָּל־רָע. חָטָאנוּ לְךָ אָדוֹן, סְלַח נָא כְּרֹב רַחֲמֶיךָ אֵל:

[הוראה בסידור] בימים שאין אומרים בהם תחנון, יאמרו:

יְהִי יְיָ אֱלֹהֵינוּ עִמָּנוּ כַּאֲשֶׁר הָיָה עִם־אֲבֹתֵינוּ, אַל־יַעַזְבֵנוּ וְאַל־יִטְּשֵׁנוּ: הוֹשִׁיעָה אֶת־עַמֶּךָ וּבָרֵךְ אֶת־נַחֲלָתֶךָ וּרְעֵם וְנַשְּׂאֵם עַד־הָעוֹלָם: בַּעֲבוּר דָּוִד עַבְדֶּךָ, אַל־תָּשֵׁב פְּנֵי מְשִׁיחֶךָ:

[הוראה בסידור] וממשיכים:

כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה, וּדְבַר יְיָ מִירוּשָׁלִָם: בָּרוּךְ הַבָּא, בָּרוּךְ הַבָּא, בָּרוּךְ הַמָּקוֹם בָּרוּךְ הוּא. בָּרוּךְ שֶׁנָּתַן תּוֹרָה לְעַמּוֹ יִשְׂרָאֵל בִּקְדֻשָּׁה וּבְטָהֳרָה [לאימות] בָּרוּךְ הוּא.

הריני מקבל

[הוראה בסידור] בקהילות מראכש והמחוז, יש שנהגו להוסיף ולומר (ראה הלכה 197):

הֲרֵינִי מְקַבֵּל קַבָּלַת הַתּוֹרָה, שֶׁנִּתְּנָה עַל יְדֵי מֹשֶׁה רַבֵּנוּ עָלָיו הַשָּׁלוֹם. וַהֲרֵינִי מְקַבְּלָהּ מֵחָדָשׁ, קַבָּלָה גְּמוּרָה וּשְׁלֵמָה. וְקַבָּלָה זוֹ הִיא בְּכָל־לִבִּי, וּבְכָל־נַפְשִׁי וּבְכָל־מְאוֹדִי, וְרַמַ"ח אֵבָרַי וְשַׁס"ה גִּידַי. וְקַבָּלָה זוֹ הִיא בְּלִי שׁוּם זֵכֶר אֹנֶס כְּלָל וְעִקָּר, אֶלָּא בְּשִׂמְחָה וּבְטוּב לֵבָב. וְקַבָּלָה זוֹ הִיא לְדַעַת רַבִּי שִׁמְעוֹן בַּר יוֹחַאי הַקָּדוֹשׁ, וּלְדַעַת רַבִּי אֶלְעָזָר בְּנוֹ הַקָּדוֹשׁ, וּלְדַעַת יוֹתָם בֶּן עֻזִּיָּהוּ הַקָּדוֹשׁ, וּפִי כְפִיהֶם, וְהוֹדָאָתִי כְּהוֹדָאָתָם, וְקַבָּלָתִי כְּקַבָּלָתָם, וְדַעְתִּי כְּדַעְתָּם. וַהֲרֵינִי סוֹמֵךְ עֲלֵיהֶם דֶּרֶךְ כְּלָל וּפְרָט וּבִכְלָל כָּל־יִשְׂרָאֵל. וְקַבָּלָה זוֹ תְהֵא עָלַי וְעַל כָּל־זַרְעִי וְזֶרַע זַרְעִי, עַד סוֹף כָּל־הָעוֹלָם, עַד וָעַד בִּכְלָל. אָמֵן:

רִבּוֹנוֹ שֶׁל עוֹלָם

רִבּוֹנוֹ שֶׁל עוֹלָם, מַלֵּא מִשְׁאֲלוֹתַי לְטוֹבָה, וְהָפֵק רְצוֹנִי, וְתֵן שְׁאֵלָתִי, וְזַכֵּנִי עִם־כָּל־בְּנֵי בֵיתִי לַעֲשׂוֹת רְצוֹנְךָ בְּלֵבָב שָׁלֵם, וּמַלְּטֵנִי מִיֵּצֶר הָרָע וּפְגָעִים רָעִים, וְתֵן חֶלְקִי בְּתוֹרָתֶךָ. וְזַכֵּנִי לִלְמֹד תּוֹרָה וְחָכְמָה, וְלַעֲשׂוֹת מַעֲשִׂים טוֹבִים לְפָנֶיךָ, לָלֶכֶת בְּדַרְכֵי יְשָׁרִים וּלְחַיֵּי הָעוֹלָם הַבָּא. וְתִשְׁמְרֵנִי מִפֶּגַע רָע, וּמִיֵּצֶר הָרָע, וּמִכָּל־שָׁעוֹת רָעוֹת הַמִּתְרַגְּשׁוֹת וּבָאוֹת לָעוֹלָם. וְהַבּוֹטֵחַ בַּיְיָ חֶסֶד יְסוֹבְבֶנּוּ: אָמֵן סֶלָה:

[הוראה בסידור] אין אומרים 'בְּרִיךְ שְׁמֵהּ' בימות החול (ראה הלכה 199). אלא נוהגים לומר מזמור זה:

מִזְמוֹר לְדָוִד, יְיָ רֹעִי לֹא אֶחְסָר: בִּנְאוֹת דֶּשֶׁא יַרְבִּיצֵנִי, עַל־מֵי מְנֻחוֹת יְנַהֲלֵנִי: נַפְשִׁי יְשׁוֹבֵב, יַנְחֵנִי בְמַעְגְּלֵי־צֶדֶק לְמַעַן שְׁמוֹ: גַּם כִּי־אֵלֵךְ בְּגֵיא צַלְמָוֶת לֹא־אִירָא רָע, כִּי־אַתָּה עִמָּדִי, שִׁבְטְךָ וּמִשְׁעַנְתֶּךָ הֵמָּה יְנַחֲמֻנִי: תַּעֲרֹךְ לְפָנַי | שֻׁלְחָן נֶגֶד צֹרְרָי, דִּשַּׁנְתָּ בַשֶּׁמֶן רֹאשִׁי, כּוֹסִי רְוָיָה: אַךְ | [המשך בעמ' 133]

[המשך מזמור כג, מעמ' 132] טוֹב וָחֶסֶד יִרְדְּפוּנִי כָּל־יְמֵי חַיַּי, וְשַׁבְתִּי בְּבֵית־יְיָ לְאֹרֶךְ יָמִים:

[הוראה בסידור] בימי החנוכה אומרים פסוקים מיוחדים שנזכר בהם 'נר' או 'אור', כאשר כל פסוק אומרו החזן תחילה והציבור שב ואומר אחריו (ראה הלכה 768):

נֵר־לְרַגְלִי דְבָרֶךָ, וְאוֹר לִנְתִיבָתִי:

נֵר יְיָ נִשְׁמַת אָדָם, חֹפֵשׂ כָּל־חַדְרֵי־בָטֶן:

כִּי נֵר מִצְוָה וְתוֹרָה אוֹר, וְדֶרֶךְ חַיִּים תּוֹכְחוֹת מוּסָר:

כִּי־אַתָּה תָּאִיר נֵרִי, יְיָ אֱלֹהַי יַגִּיהַּ חָשְׁכִּי:

אוֹר זָרֻעַ לַצַּדִּיק, וּלְיִשְׁרֵי־לֵב שִׂמְחָה:

וְאֹרַח צַדִּיקִים כְּאוֹר נֹגַהּ, הוֹלֵךְ וָאוֹר עַד־נְכוֹן הַיּוֹם:

לְדָוִד, יְיָ | אוֹרִי וְיִשְׁעִי מִמִּי אִירָא, יְיָ מָעוֹז־חַיַּי מִמִּי אֶפְחָד:

קוּמִי אוֹרִי כִּי בָא אוֹרֵךְ, וּכְבוֹד יְיָ עָלַיִךְ זָרָח:

מֵאֵין כָּמוֹךְ יְיָ, גָּדוֹל אַתָּה וְגָדוֹל שִׁמְךָ בִּגְבוּרָה: [לאימות]

[הוראה בסידור] יש אומרים פסוק זה:

צְאֶנָה | וּרְאֶינָה בְּנוֹת צִיּוֹן בַּמֶּלֶךְ שְׁלֹמֹה, בָּעֲטָרָה שֶׁעִטְּרָה־לּוֹ אִמּוֹ בְּיוֹם חֲתֻנָּתוֹ, וּבְיוֹם שִׂמְחַת לִבּוֹ:

[הוראה בסידור] בראש חודש ובחול המועד מוסיף החזן פסוק זה בנעימה, והציבור אומרים אחריו:

אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ, אַשְׁרֵי הָעָם שֶׁיְיָ אֱלֹהָיו:

[הוראה בסידור] מוציאים את ספר התורה, ואומר החזן לבדו:

גַּדְּלוּ לַיְיָ אִתִּי, וּנְרוֹמְמָה שְׁמוֹ יַחְדָּו:

[הוראה בסידור] וממשיכים החזן עם הציבור:

רוֹמְמוּ יְיָ אֱלֹהֵינוּ, וְהִשְׁתַּחֲווּ לַהֲדֹם רַגְלָיו, קָדוֹשׁ הוּא: רוֹמְמוּ יְיָ אֱלֹהֵינוּ, וְהִשְׁתַּחֲווּ לְהַר קָדְשׁוֹ, כִּי־קָדוֹשׁ יְיָ אֱלֹהֵינוּ:

[המשך עמ' 133 – "רוממו יי"] תּוֹרָה צִוָּה־לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: עֵץ־חַיִּים הִיא לַמַּחֲזִיקִים בָּהּ, וְתֹמְכֶיהָ מְאֻשָּׁר: דְּרָכֶיהָ דַרְכֵי־נֹעַם, וְכָל־נְתִיבוֹתֶיהָ שָׁלוֹם: שָׁלוֹם רָב לְאֹהֲבֵי תוֹרָתֶךָ, וְאֵין־לָמוֹ מִכְשׁוֹל: יְיָ עֹז לְעַמּוֹ יִתֵּן, יְיָ | יְבָרֵךְ אֶת־עַמּוֹ בַשָּׁלוֹם:

[הוראה בסידור] כשמגיע ספר התורה אל התיבה, נוהגים לומר תפילה זו (מסכת סופרים יד, יב). ובימות החול נאמרת כאשר הספר עדיין לא הגיע אל התיבה:

עַל הַכֹּל יִתְגַּדַּל וְיִתְקַדַּשׁ, יִתְבָּרַךְ וְיִתְרוֹמַם וְיִתְנַשֵּׂא שְׁמוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ־הוּא: בָּעוֹלָמוֹת שֶׁבָּרָא, הָעוֹלָם הַזֶּה וְהָעוֹלָם הַבָּא כִּרְצוֹנוֹ וְכִרְצוֹן יְרֵאָיו וְכִרְצוֹן כָּל־עַמּוֹ בֵּית יִשְׂרָאֵל, תִּגָּלֶה וְתֵרָאֶה מַלְכוּתוֹ עָלֵינוּ בִּמְהֵרָה, וְיָחֹן וְיָחוּס וִירַחֵם עַל פְּלֵטָתֵנוּ: וְעַל פְּלֵיטַת עַמּוֹ בֵּית יִשְׂרָאֵל, בְּחֵן בְּחֶסֶד בְּרַחֲמִים וּבְרָצוֹן, וְאִמְרוּ כָּל־הָעָם אָמֵן:

הָרַחֲמָן הוּא יַחֲזִיר שְׁבוּיֵנוּ וּשְׁבִית עַמּוֹ בֵּית יִשְׂרָאֵל בִּמְהֵרָה בְיָמֵינוּ, וְיוֹצִיאֵנוּ מִכָּל־צָרוֹתֵינוּ לִרְוָחָה, וּמֵאֲפֵלָה לְאוֹרָה בְּקָרוֹב, וְאִמְרוּ כָּל־הָעָם אָמֵן:

כִּי שֵׁם יְיָ אֶקְרָא, הָבוּ גֹדֶל לֵאלֹהֵינוּ: יְהִי שֵׁם יְיָ מְבֹרָךְ, מֵעַתָּה וְעַד־עוֹלָם: הַכֹּל תְּנוּ עֹז לֵאלֹהִים, וּתְנוּ כָבוֹד לַתּוֹרָה:

[הוראה בסידור] מגביה את ספר התורה ומראה את הכתב לציבור, ואומרים:

וְזֹאת הַתּוֹרָה אֲשֶׁר־שָׂם מֹשֶׁה לִפְנֵי בְּנֵי יִשְׂרָאֵל: תּוֹרָה צִוָּה־לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: אֵל שַׁדַּי אֱמֶת, וּמֹשֶׁה אֱמֶת, וְתוֹרָתוֹ אֱמֶת.

[הוראה בסידור] והחזן מסיים ואומר:

הָאֵל תָּמִים דַּרְכּוֹ, אִמְרַת־יְיָ צְרוּפָה, מָגֵן הוּא לְכֹל | הַחֹסִים בּוֹ:

[הוראה בסידור] בימי שני וחמישי קוראים בספר התורה שלשה עולים בפרשת השבוע – כהן, לוי וישראל. ואין מוסיפים עליהם (קישו"ע [טולידאנו] קכב, א; וראה הלכה 206).

בִּרְכוֹת הַתּוֹרָה

[הוראה בסידור] העולה אומר: הַשֵּׁם עִמָּכֶם. ועונים: יְבָרֶכְךָ הַשֵּׁם:

[הוראה בסידור] העולה: בָּרְכוּ אֶת יְיָ הַמְבֹרָךְ:

[הוראה בסידור] הקהל: בָּרוּךְ יְיָ הַמְבֹרָךְ לְעוֹלָם וָעֶד:

[הוראה בסידור] וחוזר העולה: בָּרוּךְ יְיָ הַמְבֹרָךְ לְעוֹלָם וָעֶד:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר בָּחַר־בָּנוּ מִכָּל־הָעַמִּים, וְנָתַן לָנוּ אֶת־תּוֹרָתוֹ. בָּרוּךְ אַתָּה יְיָ, נוֹתֵן הַתּוֹרָה:

[הוראה בסידור] אחר קריאת התורה אומר העולה: אֱמֶת תּוֹרָתֵנוּ הַקְּדוֹשָׁה:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר נָתַן־לָנוּ [אֶת־תּוֹרָתוֹ] תּוֹרַת אֱמֶת, וְחַיֵּי עוֹלָם נָטַע בְּתוֹכֵנוּ. בָּרוּךְ אַתָּה יְיָ, נוֹתֵן הַתּוֹרָה:

בִּרְכַּת הַגּוֹמֵל

[הוראה בסידור] כאשר היה בסכנה וניצל (ראה הלכה 223), יברך 'הגומל':

הַלְלוּיָהּ | אוֹדֶה יְיָ בְּכָל־לֵבָב, בְּסוֹד יְשָׁרִים וְעֵדָה: הוֹדוּ לַיְיָ כִּי־טוֹב, כִּי לְעוֹלָם חַסְדּוֹ: יוֹדוּ לַיְיָ חַסְדּוֹ, וְנִפְלְאוֹתָיו לִבְנֵי אָדָם:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, הַגּוֹמֵל לַחַיָּבִים טוֹבוֹת, שֶׁגְּמָלַנִי כָּל־טוֹב:

[הוראה בסידור] והקהל עונים 'אמן', ואומרים:

הָאֵל אֲשֶׁר גְּמָלְךָ כָּל־טוֹב וְחֵן וָחֶסֶד וְחֶמְלָה, יִתְבָּרַךְ וְיִתְרוֹמַם עַל כָּל־בְּרָכָה וּתְהִלָּה, הוּא בְּרַחֲמָיו יִשְׁמְרְךָ וְיִגְמָלְךָ לָעַד כָּל־טוֹב, סֶלָה:

חצי קדיש בסיום קריאת התורה

[הוראה בסידור] בסיום קריאת התורה אומרים חצי קדיש. והמנהג הנפוץ שהעולה השלישי אומר קדיש זה בסיום ברכתו האחרונה:

יִתְגַּדַּל וְיִתְקַדַּשׁ שְׁמֵיהּ רַבָּא (עונים: אמן). בְּעָלְמָא דִּי בְרָא כִּרְעוּתֵהּ, וְיַמְלִיךְ מַלְכוּתֵהּ, וְיַצְמַח פֻּרְקָנֵהּ וִיקָרֵב מְשִׁיחֵהּ (עונים: אמן). בְּחַיֵּיכוֹן וּבְיוֹמֵיכוֹן וּבְחַיֵּי דְכָל־בֵּית יִשְׂרָאֵל, בַּעֲגָלָא וּבִזְמַן קָרִיב, וְאִמְרוּ אָמֵן (עונים: אמן 'יְהֵא שְׁמֵיהּ רַבָּא' עד 'דַּאֲמִירָן בְּעָלְמָא'). יְהֵא שְׁמֵיהּ רַבָּא מְבָרַךְ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא יִתְבָּרַךְ, וְיִשְׁתַּבַּח, וְיִתְפָּאַר, וְיִתְרוֹמַם, וְיִתְנַשֵּׂא, וְיִתְהַדָּר, וְיִתְעַלֶּה וְיִתְהַלָּל

[המשך בעמ' 137] שְׁמֵיהּ דְּקֻדְשָׁא בְּרִיךְ־הוּא (עונים: אמן) לְעֵלָּא מִן כָּל־בִּרְכָתָא, שִׁירָתָא, תִּשְׁבְּחָתָא וְנֶחֱמָתָא, דַּאֲמִירָן בְּעָלְמָא, וְאִמְרוּ אָמֵן (עונים: אמן).

סיום התפילה – אשרי

[כותרת בסידור] סִיּוּם הַתְּפִלָּה

[יְהִי־חַסְדְּךָ יְיָ עָלֵינוּ, כַּאֲשֶׁר יִחַלְנוּ לָךְ:]

אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ, עוֹד יְהַלְלוּךָ סֶּלָה: אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ, אַשְׁרֵי הָעָם שֶׁיְיָ אֱלֹהָיו:

תְּהִלָּה לְדָוִד, אֲרוֹמִמְךָ אֱלוֹהַי הַמֶּלֶךְ, וַאֲבָרְכָה שִׁמְךָ לְעוֹלָם וָעֶד: בְּכָל־יוֹם אֲבָרְכֶךָּ, וַאֲהַלְלָה שִׁמְךָ לְעוֹלָם וָעֶד: גָּדוֹל יְיָ וּמְהֻלָּל מְאֹד, וְלִגְדֻלָּתוֹ אֵין חֵקֶר: דּוֹר לְדוֹר יְשַׁבַּח מַעֲשֶׂיךָ, וּגְבוּרֹתֶיךָ יַגִּידוּ: הֲדַר כְּבוֹד הוֹדֶךָ, וְדִבְרֵי נִפְלְאֹתֶיךָ אָשִׂיחָה: וֶעֱזוּז נוֹרְאֹתֶיךָ יֹאמֵרוּ, וּגְדֻלָּתְךָ אֲסַפְּרֶנָּה: זֵכֶר רַב־טוּבְךָ יַבִּיעוּ, וְצִדְקָתְךָ יְרַנֵּנוּ: חַנּוּן וְרַחוּם יְיָ, אֶרֶךְ אַפַּיִם וּגְדָל־ [המשך בעמ' 138 – עדיין לא הוקלד]

[המשך אשרי מעמ' 137, "…אֶרֶךְ אַפַּיִם וּגְדָל־"] חָסֶד: טוֹב־יְיָ לַכֹּל, וְרַחֲמָיו עַל־כָּל־מַעֲשָׂיו: יוֹדוּךָ יְיָ כָּל־מַעֲשֶׂיךָ, וַחֲסִידֶיךָ יְבָרְכוּכָה: כְּבוֹד מַלְכוּתְךָ יֹאמֵרוּ, וּגְבוּרָתְךָ יְדַבֵּרוּ: לְהוֹדִיעַ | לִבְנֵי הָאָדָם גְּבוּרֹתָיו, וּכְבוֹד הֲדַר מַלְכוּתוֹ: מַלְכוּתְךָ מַלְכוּת כָּל־עֹלָמִים, וּמֶמְשַׁלְתְּךָ בְּכָל־דּוֹר וָדֹר: סוֹמֵךְ יְיָ לְכָל־הַנֹּפְלִים, וְזוֹקֵף לְכָל־הַכְּפוּפִים: עֵינֵי־כֹל אֵלֶיךָ יְשַׂבֵּרוּ, וְאַתָּה נוֹתֵן־לָהֶם אֶת־אָכְלָם בְּעִתּוֹ:

[הוראה בסידור] פורשים ידיים כלפי מעלה, ואומרים בכוונה:

פּוֹתֵחַ אֶת־יָדֶךָ, וּמַשְׂבִּיעַ לְכָל־חַי רָצוֹן: צַדִּיק יְיָ בְּכָל־דְּרָכָיו, וְחָסִיד בְּכָל־מַעֲשָׂיו: קָרוֹב יְיָ לְכָל־קֹרְאָיו, לְכֹל אֲשֶׁר יִקְרָאֻהוּ בֶאֱמֶת: רְצוֹן־יְרֵאָיו יַעֲשֶׂה, וְאֶת־שַׁוְעָתָם יִשְׁמַע וְיוֹשִׁיעֵם: שׁוֹמֵר יְיָ אֶת־כָּל־אֹהֲבָיו, וְאֵת כָּל־הָרְשָׁעִים יַשְׁמִיד: תְּהִלַּת יְיָ יְדַבֶּר־פִּי, וִיבָרֵךְ כָּל־בָּשָׂר שֵׁם [המשך בעמ' 139: קָדְשׁוֹ לְעוֹלָם וָעֶד…]

[סיום אשרי, המשך מעמ' 138 – "…וִיבָרֵךְ כָּל־בָּשָׂר שֵׁם"] קָדְשׁוֹ לְעוֹלָם וָעֶד: וַאֲנַחְנוּ | נְבָרֵךְ יָהּ מֵעַתָּה וְעַד־עוֹלָם, הַלְלוּיָהּ:

[הוראה בסידור] בימים שאין אומרים בהם תחנון, אף אין אומרים מזמור זה (שכנה"ג קלא, הגה"ט, ג):

לַמְנַצֵּחַ מִזְמוֹר לְדָוִד: יַעַנְךָ יְיָ בְּיוֹם צָרָה, יְשַׂגֶּבְךָ שֵׁם | אֱלֹהֵי יַעֲקֹב: יִשְׁלַח־עֶזְרְךָ מִקֹּדֶשׁ, וּמִצִּיּוֹן יִסְעָדֶךָּ: יִזְכֹּר כָּל־מִנְחֹתֶיךָ, וְעוֹלָתְךָ יְדַשְּׁנֶה סֶלָה: יִתֶּן־לְךָ כִלְבָבֶךָ, וְכָל־עֲצָתְךָ יְמַלֵּא: נְרַנְּנָה | בִּישׁוּעָתֶךָ וּבְשֵׁם־אֱלֹהֵינוּ נִדְגֹּל, יְמַלֵּא יְיָ כָּל־מִשְׁאֲלוֹתֶיךָ: עַתָּה יָדַעְתִּי כִּי הוֹשִׁיעַ | יְיָ מְשִׁיחוֹ, יַעֲנֵהוּ מִשְּׁמֵי קָדְשׁוֹ בִּגְבֻרוֹת יֵשַׁע יְמִינוֹ: אֵלֶּה בָרֶכֶב וְאֵלֶּה בַסּוּסִים, וַאֲנַחְנוּ | בְּשֵׁם־יְיָ אֱלֹהֵינוּ נַזְכִּיר: הֵמָּה כָּרְעוּ וְנָפָלוּ, וַאֲנַחְנוּ קַּמְנוּ וַנִּתְעוֹדָד: יְיָ הוֹשִׁיעָה, הַמֶּלֶךְ יַעֲנֵנוּ בְיוֹם־קָרְאֵנוּ:

וּבָא לְצִיּוֹן

[הוראה בסידור] בתשעה באב ובבית האבל מתחילים מ'וְאַתָּה קָדוֹשׁ'.

[הוראה בסידור] אין מדקדקים לומר את פסוקי התרגום בלחש (שלמי צבור; וראה הלכה 230).

וּבָא לְצִיּוֹן גּוֹאֵל, וּלְשָׁבֵי פֶשַׁע בְּיַעֲקֹב, נְאֻם יְיָ: וַאֲנִי זֹאת בְּרִיתִי אוֹתָם, אָמַר יְיָ, רוּחִי אֲשֶׁר עָלֶיךָ, וּדְבָרַי אֲשֶׁר־שַׂמְתִּי בְּפִיךָ, לֹא־יָמוּשׁוּ מִפִּיךָ, וּמִפִּי זַרְעֲךָ, וּמִפִּי זֶרַע זַרְעֲךָ, אָמַר יְיָ, מֵעַתָּה וְעַד־עוֹלָם:

וְאַתָּה קָדוֹשׁ, יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל: וְקָרָא זֶה אֶל־זֶה וְאָמַר: קָדוֹשׁ | קָדוֹשׁ קָדוֹשׁ יְיָ צְבָאוֹת, מְלֹא כָל־הָאָרֶץ כְּבוֹדוֹ:

וּמְקַבְּלִין דֵּין מִן־דֵּין וְאָמְרִין: קַדִּישׁ בִּשְׁמֵי מְרוֹמָא עִלָּאָה בֵּית שְׁכִינְתֵּהּ, קַדִּישׁ עַל־אַרְעָא עוֹבַד גְּבוּרְתֵּהּ, קַדִּישׁ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא יְיָ צְבָאוֹת, מַלְיָא כָל־אַרְעָא זִיו יְקָרֵהּ:

וַתִּשָּׂאֵנִי רוּחַ, וָאֶשְׁמַע אַחֲרַי קוֹל רַעַשׁ גָּדוֹל: בָּרוּךְ כְּבוֹד־יְיָ מִמְּקוֹמוֹ:

וּנְטָלַתְנִי רוּחָא, וְשָׁמְעִית בַּתְרַי קַל זִיעַ [המשך בעמ' 141] רַעַשׁ [לאימות]…

המשך

[…] רַעַשׁ גָּדוֹל – ראה למעלה. בראש עמ' 141 נראה: שַׁגִּיא דִּמְשַׁבְּחִין וְאָמְרִין: בְּרִיךְ יְקָרָא דַייָ מֵאֲתַר בֵּית שְׁכִינְתֵּהּ: יְיָ | יִמְלֹךְ לְעֹלָם וָעֶד: יְיָ מַלְכוּתֵהּ קָאִים לְעָלַם וּלְעָלְמֵי עָלְמַיָּא: יְיָ אֱלֹהֵי אַבְרָהָם יִצְחָק וְיִשְׂרָאֵל אֲבוֹתֵינוּ, שָׁמְרָה־זֹּאת לְעוֹלָם, לְיֵצֶר מַחְשְׁבוֹת לְבַב עַמֶּךָ, וְהָכֵן לְבָבָם אֵלֶיךָ: וְהוּא רַחוּם | יְכַפֵּר עָוֹן, וְלֹא־יַשְׁחִית, וְהִרְבָּה לְהָשִׁיב אַפּוֹ, וְלֹא־יָעִיר כָּל־חֲמָתוֹ: כִּי־אַתָּה אֲדֹנָי טוֹב וְסַלָּח, וְרַב־חֶסֶד לְכָל־קֹרְאֶיךָ: צִדְקָתְךָ צֶדֶק לְעוֹלָם, וְתוֹרָתְךָ אֱמֶת: תִּתֵּן אֱמֶת לְיַעֲקֹב, חֶסֶד לְאַבְרָהָם, אֲשֶׁר־נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ מִימֵי קֶדֶם: בָּרוּךְ אֲדֹנָי יוֹם | יוֹם יַעֲמָס־לָנוּ, הָאֵל יְשׁוּעָתֵנוּ סֶלָה: יְיָ צְבָאוֹת עִמָּנוּ, מִשְׂגָּב־לָנוּ אֱלֹהֵי יַעֲקֹב סֶלָה: יְיָ צְבָאוֹת, אַשְׁרֵי אָדָם בֹּטֵחַ בָּךְ: יְיָ הוֹשִׁיעָה, הַמֶּלֶךְ יַעֲנֵנוּ בְיוֹם־קָרְאֵנוּ:

[המשך בעמ' 142 – עדיין לא הוקלד]

ובא לציון (המשך)

[המשך מעמ' 141 – אחרי "…בְיוֹם־קָרְאֵנוּ:"] בָּרוּךְ אֱלֹהֵינוּ שֶׁבְּרָאָנוּ לִכְבוֹדוֹ, וְהִבְדִּילָנוּ מִן הַתּוֹעִים, וְנָתַן לָנוּ תּוֹרַת אֱמֶת, וְחַיֵּי עוֹלָם נָטַע בְּתוֹכֵנוּ. הוּא יִפְתַּח לִבֵּנוּ בְּתוֹרָתוֹ, וְיָשִׂים בְּלִבֵּנוּ אַהֲבָתוֹ וְיִרְאָתוֹ לַעֲשׂוֹת רְצוֹנוֹ, וּלְעָבְדוֹ בְּלֵבָב שָׁלֵם. לְמַעַן לֹא נִיגַע לָרִיק, וְלֹא נֵלֵד לַבֶּהָלָה.

יְהִי רָצוֹן מִלְּפָנֶיךָ יְיָ אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, שֶׁנִּשְׁמֹר חֻקֶּיךָ וּמִצְוֹתֶיךָ בָּעוֹלָם הַזֶּה, וְנִזְכֶּה וְנִחְיֶה וְנִרְאֶה וְנִירַשׁ טוֹבָה וּבְרָכָה לְחַיֵּי הָעוֹלָם הַבָּא:

לְמַעַן | יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם, יְיָ אֱלֹהַי לְעוֹלָם אוֹדֶךָּ: יְיָ חָפֵץ לְמַעַן צִדְקוֹ, יַגְדִּיל תּוֹרָה וְיַאְדִּיר: וְיִבְטְחוּ בְךָ יוֹדְעֵי שְׁמֶךָ, כִּי לֹא־עָזַבְתָּ דֹרְשֶׁיךָ יְיָ: יְיָ אֲדֹנֵינוּ מָה־אַדִּיר שִׁמְךָ בְּכָל־הָאָרֶץ: חִזְקוּ וְיַאֲמֵץ לְבַבְכֶם, כָּל־הַמְיַחֲלִים לַיְיָ:

סיום התפילה

בִּטְחוּ בַיְיָ עֲדֵי־עַד, כִּי בְּיָהּ יְיָ צוּר עוֹלָמִים: וְאֲנִי | עָלֶיךָ בָטַחְתִּי יְיָ, אָמַרְתִּי אֱלֹהַי אָתָּה: [לאימות]

[הוראה בסידור] ואומר החזן קדיש תתקבל:

יִתְגַּדַּל וְיִתְקַדַּשׁ שְׁמֵיהּ רַבָּא (עונים: אמן). בְּעָלְמָא דִּי בְרָא כִּרְעוּתֵהּ, וְיַמְלִיךְ מַלְכוּתֵהּ, וְיַצְמַח פֻּרְקָנֵהּ וִיקָרֵב מְשִׁיחֵהּ (עונים: אמן). בְּחַיֵּיכוֹן וּבְיוֹמֵיכוֹן וּבְחַיֵּי דְכָל־בֵּית יִשְׂרָאֵל, בַּעֲגָלָא וּבִזְמַן קָרִיב, וְאִמְרוּ אָמֵן (עונים: אמן 'יְהֵא שְׁמֵיהּ רַבָּא' עד 'דַּאֲמִירָן בְּעָלְמָא'). יְהֵא שְׁמֵיהּ רַבָּא מְבָרַךְ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא יִתְבָּרַךְ, וְיִשְׁתַּבַּח, וְיִתְפָּאַר, וְיִתְרוֹמַם, וְיִתְנַשֵּׂא, וְיִתְהַדָּר, וְיִתְעַלֶּה וְיִתְהַלָּל שְׁמֵיהּ דְּקֻדְשָׁא בְּרִיךְ־הוּא (עונים: אמן) לְעֵלָּא מִן כָּל־בִּרְכָתָא, שִׁירָתָא, תֻּשְׁבְּחָתָא וְנֶחֱמָתָא, דַּאֲמִירָן בְּעָלְמָא, וְאִמְרוּ אָמֵן (עונים: אמן).

תִּתְקַבַּל [צְלוֹתְנָא וּבָעוּתְנָא עִם] צְלוֹתְהוֹן וּבָעוּתְהוֹן דְּכָל־בֵּית יִשְׂרָאֵל, קֳדָם אֲבוּנָא מָרָנָא דִבְשְׁמַיָּא וְאַרְעָא [לאימות], וְאִמְרוּ אָמֵן (עונים: אמן).

[המשך קדיש תתקבל מעמ' 143] יְהֵא שְׁלָמָא רַבָּא מִן שְׁמַיָּא, חַיִּים וְשָׂבָע, וִישׁוּעָה וְנֶחָמָה, וְשֵׁיזָבָא, וּרְפוּאָה, וּגְאֻלָּה, וּסְלִיחָה וְכַפָּרָה, וְרֶוַח וְהַצָּלָה, לָנוּ וּלְכָל־עַמּוֹ יִשְׂרָאֵל, וְאִמְרוּ אָמֵן (עונים: אמן).

עֹשֶׂה שָׁלוֹם (בעשי"ת: הַשָּׁלוֹם) בִּמְרוֹמָיו, הוּא בְּרַחֲמָיו יַעֲשֶׂה שָׁלוֹם עָלֵינוּ וְעַל כָּל־עַמּוֹ יִשְׂרָאֵל, וְאִמְרוּ אָמֵן (עונים: אמן).

[הוראה בסידור] בשני וחמישי מחזירים את ספר התורה להיכל כאן (ראה הלכה 227), ואומרים:

יְהַלְלוּ | אֶת־שֵׁם יְיָ, כִּי־נִשְׂגָּב שְׁמוֹ לְבַדּוֹ, הוֹדוֹ עַל־אֶרֶץ וְשָׁמָיִם: וַיָּרֶם קֶרֶן | לְעַמּוֹ תְּהִלָּה לְכָל־חֲסִידָיו, לִבְנֵי יִשְׂרָאֵל עַם קְרֹבוֹ, הַלְלוּיָהּ: יְיָ הוּא הָאֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל־הָאָרֶץ מִתָּחַת אֵין עוֹד: אֵין־כָּמוֹךָ | בָאֱלֹהִים אֲדֹנָי, וְאֵין כְּמַעֲשֶׂיךָ: וּבְנֻחֹה יֹאמַר, שׁוּבָה יְיָ רִבְבוֹת אַלְפֵי יִשְׂרָאֵל: הֲשִׁיבֵנוּ יְיָ | אֵלֶיךָ וְנָשׁוּבָה, חַדֵּשׁ יָמֵינוּ כְּקֶדֶם:

[הוראה בסידור – ראה הערה] בראש חודש, בחול המועד, באסרו חג, בחנוכה ובפורים, בשושן פורים, בערב ראש השנה ובערב יום הכיפורים, בימי התעניות, בי"א בתשרי, כשיש חתן או מילה, וכן בבית האבל, מדלגים על 'בֵּית יַעֲקֹב' ושיר של יום, ואומרים רק מזמור המיוחד לאותו יום (ראה הלכה 234).

שאלות נפוצות

מי קורא בשני וחמישי?

לפי ההוראה בסידור, שלושה קוראים בשני ובחמישי.

מתי אומרים את מזמור 'יענך'?

בימים שאין בהם תחנון לא אומרים אותו – כך כתוב בסידור.

מה עם ראש חודש?

ראו את הדף של שבת מברכים וראש חודש.

מה עושים בסוף הקריאה?

אחרי קדיש תתקבל מחזירים את ספר התורה להיכל, כמופיע בסוף הדף.

הנוסח מרוקאי?

כן, מילה במילה מסידור כמנהג מרוקו, מנוקד.

האם הדף בחינם?

כן, בחינם, בלי הרשמה ובלי פרסומות, מיזם חסד.
