מי שברך נוסח מרוקאי

מי שברך לקהל כפי שבסידור, ולצידו הנוסחים לעולים ולמקרים השונים. במקום "(שם ושם משפחה)" אומרים את השם. ההוראות שבסוגריים הן הוראות הסידור עצמו. לכלי עם מילוי שמות אוטומטי:

לכלי מי שברך והשכבות עם מילוי שמות

מי שברך לעולה לתורה

יִשְׁתַּבַּח וְיִתְעַלֶּה שְׁמוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ־הוּא, שֶׁהוּא רִאשׁוֹן וְהוּא אַחֲרוֹן, וּמִבַּלְעָדָיו אֵין אֱלֹהִים:

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ הַקְּדוֹשִׁים וְהַטְּהוֹרִים, אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה וְאַהֲרֹן, דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ אֶת הַשֵּׁם הַטּוֹב כְּבוֹד רִבִּי (שם ושם משפחה). מַלְכָּא דְעָלְמָא הוּא יְבָרֵךְ יָתֵיהּ, וִיזַכֵּה יָתֵיהּ, וְיִשְׁמַע בְּקָל צְלוֹתֵיהּ, וְיִשְׁמֹר לוֹ כָּל־אֲשֶׁר לוֹ, וְיִשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יָדָיו, וּמִכָּל־צָרָה וָנֶזֶק יַצִּילֵהוּ, וְיַעַזְרֵהוּ הָאֵל עַל דְּבַר כְּבוֹד שְׁמוֹ, וִימַלֵּא כָּל־מִשְׁאֲלוֹת לִבּוֹ לְטוֹבָה, וְיַאֲרִיךְ יָמָיו בַּטּוֹב וּשְׁנוֹתָיו בַּנְּעִימִים, וּבְכָל־אֲשֶׁר יִפְנֶה יַשְׂכִּיל וְיַצְלִיחַ, וּזְכוּת הַתּוֹרָה תָּגֵן בַּעֲדוֹ וּבְעַד בֵּיתוֹ וּבְעַד כָּל־אֲשֶׁר לוֹ. וְיִכְלְלוּ בוֹ כָּל־הַבְּרָכוֹת כֻּלָּן. (בשבת שובה: וְיִכְתְּבֵהוּ הָאֵל בְּסֵפֶר חַיִּים טוֹבִים), וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לעולה – נוסח קצר

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ הַקְּדוֹשִׁים וְהַטְּהוֹרִים, אַבְרָהָם יִצְחָק וְיַעֲקֹב, הוּא יְבָרֵךְ אֶת הַשֵּׁם הַטּוֹב (שם ושם משפחה), בַּעֲבוּר שֶׁעָלָה לִכְבוֹד הַמָּקוֹם וְלִכְבוֹד הַתּוֹרָה וְלִכְבוֹד הַשַּׁבָּת (ביו"ט: וְלִכְבוֹד הָרֶגֶל). בִּשְׂכַר זֶה הַקָּדוֹשׁ בָּרוּךְ־הוּא יִשְׁמְרֵהוּ וְיַצִּילֵהוּ מִכָּל־צָרָה וְצוּקָה, וּמִכָּל־נֶגַע וּמַחֲלָה, וְיִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂה יָדָיו (בשבת שובה: וְיִכְתְּבֵהוּ הָאֵל בְּסֵפֶר חַיִּים טוֹבִים) עִם כָּל־יִשְׂרָאֵל אֶחָיו. וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך ליולדת או לאישה

מִי שֶׁבֵּרַךְ אִמּוֹתֵינוּ הַקְּדוֹשׁוֹת וְהַטְּהוֹרוֹת, שָׂרָה רִבְקָה רָחֵל וְלֵאָה, מִרְיָם הַנְּבִיאָה וַאֲבִיגַיִל, וְאֶסְתֵּר בַּת אֲבִיחַיִל, הוּא יְבָרֵךְ אֶת הָאִשָּׁה הַצְּנוּעָה וְהַכְּבוּדָה (הַיּוֹלֶדֶת) מָרַת (שם ושם משפחה). מַלְכָּא דְעָלְמָא הוּא יְבָרֵךְ יָתַהּ, וִיזַכֵּה יָתַהּ, וְיִשְׁמַע בְּקָל צְלוֹתַהּ, וְיִשְׁמֹר לָהּ כָּל־אֲשֶׁר לָהּ, וְיַצִּילֶהָ מִכָּל־צָרָה וָנֶזֶק, וְיִשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יָדֶיהָ, וִימַלֵּא כָּל־מִשְׁאֲלוֹת לִבָּהּ לְטוֹבָה, וְיַאֲרִיךְ יָמֶיהָ בַּטּוֹב וּשְׁנוֹתֶיהָ בַּנְּעִימִים. וּבְכָל־אֲשֶׁר תִּפְנֶה תַּשְׂכִּיל וְתַצְלִיחַ, וּזְכוּת הַתּוֹרָה תָּגֵן בַּעֲדָהּ וּבְעַד בֵּיתָהּ וּבְעַד בַּעְלָהּ וּבְעַד כָּל־אֲשֶׁר לָהּ. (ליולדת: וְיִשְׁלַח לָהּ הָאֵל רְפוּאָה שְׁלֵמָה, רְפוּאַת הַנֶּפֶשׁ וּרְפוּאַת הַגּוּף, וְתִזְכֶּה לִרְאוֹת נַחַת מִכָּל־יוֹצְאֵי חֲלָצֶיהָ, בִּבְרִיאוּת וּבְכָבוֹד, דְּשֵׁנִים וְרַעֲנַנִּים, בְּשֵׂיבָה יְנוּבוּן), וְיִכְלְלוּ בָהּ כָּל־הַבְּרָכוֹת כֻּלָּן. וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לחולה

יִשְׁתַּבַּח וְיִתְעַלֶּה שְׁמוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ־הוּא, שֶׁהוּא רִאשׁוֹן וְהוּא אַחֲרוֹן, וּמִבַּלְעָדָיו אֵין אֱלֹהִים:

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ הַקְּדוֹשִׁים וְהַטְּהוֹרִים, אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה וְאַהֲרֹן, דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ אֶת הַחוֹלֶה (שם החולה ושם משפחתו) וְיִשְׁלַח לוֹ רְפוּאָה שְׁלֵמָה בִּרְמַ"ח אֵבָרָיו וּבְשַׁסַ"ה גִּידָיו. אֵל נָא רְפָא נָא לוֹ, אֵל נָא רְפָא נָא לוֹ. וְהַחֲלִימֵהוּ וְהַחֲזִיקֵהוּ וְהַבְרִיאֵהוּ וְהַחֲזִירֵהוּ לְאֵיתָנוֹ הָרִאשׁוֹן, (בשבת: שַׁבָּת הִיא מִלִּזְעֹק וּרְפוּאָה קְרוֹבָה לָבֹא), הַשְׁתָּא בַּעֲגָלָא וּבִזְמַן קָרִיב, וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לחתן

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ הַקְּדוֹשִׁים וְהַטְּהוֹרִים, אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה וְאַהֲרֹן, דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ אֶת הֶחָתָן הַמְּפֹאָר הַשֵּׁם הַטּוֹב כְּבוֹד רִבִּי (שם ושם משפחה). מַלְכָּא דְעָלְמָא הוּא יְבָרֵךְ יָתֵיהּ, וִיזַכֵּה יָתֵיהּ, וְיִשְׁמַע בְּקָל צְלוֹתֵיהּ, וְיִשְׁמֹר לוֹ כָּל־אֲשֶׁר לוֹ, וְיִשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יָדָיו, וּמִכָּל־צָרָה וָנֶזֶק יַצִּילֵהוּ, וְיַעַזְרֵהוּ הָאֵל עַל דְּבַר כְּבוֹד שְׁמוֹ, וִימַלֵּא כָּל־מִשְׁאֲלוֹת לִבּוֹ לְטוֹבָה, וְיַאֲרִיךְ יָמָיו בַּטּוֹב וּשְׁנוֹתָיו בַּנְּעִימִים, וּבְכָל־אֲשֶׁר יִפְנֶה יַשְׂכִּיל וְיַצְלִיחַ, וּזְכוּת הַתּוֹרָה תָּגֵן בַּעֲדוֹ וּבְעַד בֵּיתוֹ וּבְעַד כָּל אֲשֶׁר לוֹ. וְיִכְלְלוּ בוֹ כָּל־הַבְּרָכוֹת כֻּלָּן. (בשבת שובה: וְיִכְתְּבֵהוּ הָאֵל בְּסֵפֶר חַיִּים טוֹבִים), וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לכלה

מִי שֶׁבֵּרַךְ אִמּוֹתֵינוּ הַקְּדוֹשׁוֹת וְהַטְּהוֹרוֹת, שָׂרָה רִבְקָה רָחֵל וְלֵאָה, מִרְיָם הַנְּבִיאָה וַאֲבִיגַיִל, וְאֶסְתֵּר בַּת אֲבִיחַיִל, הוּא יְבָרֵךְ אֶת הָאִשָּׁה הַצְּנוּעָה וְהַכְּבוּדָה הַכַּלָּה הַנְּעִימָה, מָרַת (שם ושם משפחה). מַלְכָּא דְעָלְמָא הוּא יְבָרֵךְ יָתַהּ, וִיזַכֵּה יָתַהּ, וְיִשְׁמַע בְּקָל צְלוֹתַהּ, וְיִשְׁמֹר לָהּ כָּל־אֲשֶׁר לָהּ, וְיַצִּילָהּ מִכָּל־צָרָה וָנֶזֶק, וְיִשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יָדֶיהָ, וִימַלֵּא כָּל־מִשְׁאֲלוֹת לִבָּהּ לְטוֹבָה, וְיַאֲרִיךְ יָמֶיהָ בַּטּוֹב וּשְׁנוֹתֶיהָ בַּנְּעִימִים. וּבְכָל־אֲשֶׁר תִּפְנֶה תַּשְׂכִּיל וְתַצְלִיחַ, וּזְכוּת הַתּוֹרָה תָּגֵן בַּעֲדָהּ וּבְעַד בֵּיתָהּ וּבְעַד בַּעֲלָהּ וּבְעַד כָּל־אֲשֶׁר לָהּ, וְיִכְלְלוּ בָהּ כָּל־הַבְּרָכוֹת כֻּלָּן. וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לבר מצווה

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ הַקְּדוֹשִׁים וְהַטְּהוֹרִים, אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה וְאַהֲרֹן, דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ אֶת הֶחָתָן הַמְּפֹאָר הַשֵּׁם הַטּוֹב, הַתַּלְמִיד הַנֶּחְמָד וְהַנָּעִים (שם ושם משפחה). מַלְכָּא דְעָלְמָא הוּא יְבָרֵךְ יָתֵיהּ, וִיזַכֵּה יָתֵיהּ, וְיִשְׁמַע בְּקָל צְלוֹתֵיהּ, וְיִשְׁמֹר לוֹ כָּל־אֲשֶׁר לוֹ, וְיִשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יָדָיו, וּמִכָּל־צָרָה וָנֶזֶק יַצִּילֵהוּ, וְיַעַזְרֵהוּ הָאֵל עַל דְּבַר כְּבוֹד שְׁמוֹ, וִימַלֵּא כָּל־מִשְׁאֲלוֹת לִבּוֹ לְטוֹבָה, וְיַאֲרִיךְ יָמָיו בַּטּוֹב וּשְׁנוֹתָיו בַּנְּעִימִים, וּבְכָל־אֲשֶׁר יִפְנֶה יַשְׂכִּיל וְיַצְלִיחַ. וּכְשֵׁם שֶׁזָּכָה לְהִכָּנֵס לַמִּצְוֹת כָּךְ יִזְכֶּה לַעֲשׂוֹת חַיִל בַּתּוֹרָה וּבְמַעֲשִׂים טוֹבִים, וִיזַכֵּהוּ הָאֵל לְחֻפָּתוֹ וּלְשִׂמְחָתוֹ בְּחַיֵּי כָּל־קְרוֹבָיו וּבְנֵי מִשְׁפַּחְתּוֹ. הַמַּלְאָךְ הַגֹּאֵל אֹתִי מִכָּל־רָע יְבָרֵךְ אֶת הַנְּעָרִים וְיִקָּרֵא בָהֶם שְׁמִי וְשֵׁם אֲבֹתַי אַבְרָהָם וְיִצְחָק, וְיִדְגּוּ לָרֹב בְּקֶרֶב הָאָרֶץ: יְבָרֶכְךָ ה' וְיִשְׁמְרֶךָ: יָאֵר ה' פָּנָיו אֵלֶיךָ וִיחֻנֶּךָּ: יִשָּׂא ה' פָּנָיו אֵלֶיךָ וְיָשֵׂם לְךָ שָׁלוֹם: וְיִכְלְלוּ בוֹ כָּל־הַבְּרָכוֹת כֻּלָּן. (בשבת שובה: וְיִכְתְּבֵהוּ הָאֵל בְּסֵפֶר חַיִּים טוֹבִים), וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לבת מצווה

מִי שֶׁבֵּרַךְ אִמּוֹתֵינוּ הַקְּדוֹשׁוֹת וְהַטְּהוֹרוֹת, שָׂרָה רִבְקָה רָחֵל וְלֵאָה, מִרְיָם הַנְּבִיאָה וַאֲבִיגַיִל, וְאֶסְתֵּר בַּת אֲבִיחַיִל, הוּא יְבָרֵךְ אֶת (שם ושם משפחה), שֶׁמָּלְאוּ לָהּ שְׁתֵּים עֶשְׂרֵה שָׁנָה וְהִגִּיעָה לְמִצְוֹת, וְנוֹתֶנֶת שֶׁבַח וְהוֹדָיָה לְהַשֵּׁם יִתְבָּרַךְ עַל כָּל־הַטּוֹבָה שֶׁגָּמַל אִתָּהּ. יִשְׁמְרֶהָ הַקָּדוֹשׁ בָּרוּךְ הוּא וִיחַיֶּהָ, וִיכוֹנֵן אֶת לִבָּהּ לִהְיוֹת שָׁלֵם עִם ה' וְלָלֶכֶת בִּדְרָכָיו וְלִשְׁמֹר כָּל־מִצְוֹתָיו כָּל־הַיָּמִים, וְכֵן יְהִי רָצוֹן, וְנֹאמַר אָמֵן.

מי שברך לחיילי צה"ל

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב, הוּא יְבָרֵךְ אֶת־חַיָּלֵי צְבָא הַהֲגַנָּה לְיִשְׂרָאֵל וְאַנְשֵׁי כּוֹחוֹת הַבִּטָּחוֹן, הָעוֹמְדִים עַל־מִשְׁמַר אַרְצֵנוּ וְעָרֵי אֱלֹהֵינוּ, מִגְּבוּל הַלְּבָנוֹן וְעַד מִדְבַּר מִצְרַיִם וּמִן הַיָּם הַגָּדוֹל עַד לְבוֹא הָעֲרָבָה, בַּיַּבָּשָׁה בָּאֲוִיר וּבַיָּם. יִתֵּן ה' אֶת אוֹיְבֵינוּ הַקָּמִים עָלֵינוּ נִגָּפִים לִפְנֵיהֶם. הַקָּדוֹשׁ בָּרוּךְ־הוּא יִשְׁמֹר וְיַצִּיל אֶת־חַיָּלֵינוּ מִכָּל־צָרָה וְצוּקָה, וּמִכָּל־נֶגַע וּמַחֲלָה, וְיִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂה יְדֵיהֶם. יַדְבֵּר שׂוֹנְאֵינוּ תַּחְתֵּיהֶם וִיעַטְּרֵם בְּכֶתֶר יְשׁוּעָה וּבַעֲטֶרֶת נִצָּחוֹן. וִיקֻיַּם בָּהֶם הַכָּתוּב: כִּי ה' אֱלֹהֵיכֶם הַהֹלֵךְ עִמָּכֶם לְהִלָּחֵם לָכֶם עִם־אֹיְבֵיכֶם לְהוֹשִׁיעַ אֶתְכֶם. וְנֹאמַר אָמֵן:

תפילה לדרך

לֵךְ לְשָׁלוֹם בִּמְקוֹמְךָ, יִרְצֶה הָאֱלֹהִים אֶת מַעֲשֶׂיךָ. בָּרוּךְ אַתָּה בְּבֹאֶךָ וּבָרוּךְ אַתָּה בְּצֵאתֶךָ:

שאלות נפוצות

אפשר למלא שמות?

כן, בכלי מי שברך והשכבות שבקישור.

הנוסח מרוקאי?

כן, מילה במילה מסידור כמנהג מרוקו.

איזה נוסחים יש בדף?

לקהל, לעולה, לחולה, לחתן ולכלה, לבר ולבת מצווה, לחיילי צה"ל ותפילה לדרך.

מה בשבת שובה?

ההוראה 'בשבת שובה' מופיעה בסוגריים בנוסח, כפי שבסידור.

אפשר להדפיס?

כן, יש עיצוב הדפסה נקי.

האם הדף בחינם?

כן, בחינם, בלי הרשמה ובלי פרסומות, מיזם חסד.
