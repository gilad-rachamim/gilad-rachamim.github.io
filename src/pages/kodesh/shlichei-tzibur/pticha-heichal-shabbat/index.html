פתיחת ההיכל בשבת בריך שמיה

הסדר בפתיחת ההיכל בשבת בבוקר, מ'בריך שמיה'. 'הריני מקבל' ו'ריבונו של עולם' שלפניו אופציונליים – ומופיעים בסוף הדף.

וְלֶהֱוֵי אֲנָא עַבְדָּךְ פְּקִידָא בְּגוֹ צַדִּיקַיָּא, לְמִרְחַם עֲלַי וּלְמִנְטַר יָתִי וְיָת כָּל־דִּי לִי וְדִי לְעַמָּךְ יִשְׂרָאֵל. אַנְתְּ הוּא זָן לְכֹלָּא וּמְפַרְנֵס לְכֹלָּא, אַנְתְּ הוּא שַׁלִּיט עַל־כֹּלָּא, אַנְתְּ הוּא דְּשַׁלִּיט עַל־מַלְכַיָּא, וּמַלְכוּתָא דִילָךְ הִיא. אֲנָא עַבְדָּא דְקֻדְשָׁא בְּרִיךְ־הוּא, דְּסָגִידְנָא קַמֵּהּ וּמִן קַמֵּי דִּיקַר אוֹרַיְתֵהּ בְּכָל־עִדָּן וְעִדָּן. לָא עַל־אֱנָשׁ רְחִיצְנָא, וְלָא עַל־בַּר אֱלָהִין סְמִיכְנָא, אֶלָּא בֶּאֱלָהָא דִשְׁמַיָּא, דְּהוּא אֱלָהָא דִקְשׁוֹט, וְאוֹרַיְתֵהּ קְשׁוֹט, וּנְבִיאוֹהִי קְשׁוֹט, וּמַסְגֵּי לְמֶעְבַּד טַבְוָן וּקְשׁוֹט. בֵּיהּ אֲנָא רָחִיץ, וְלִשְׁמֵהּ יַקִּירָא קַדִּישָׁא אֲנָא אֵמַר תֻּשְׁבְּחָן. יְהֵא רַעֲוָא קֳדָמָךְ דְּתִפְתַּח לִבָּאי בְּאוֹרַיְתָךְ, [וְתִיהַב לִי בְּנִין דִּכְרִין דְּעָבְדִין רְעוּתָךְ], וְתַשְׁלִים מִשְׁאֲלִין

[המשך בצילום 15]

[הערה בסידור, "שערי תפילה", על "ותיהב לי בנין דכרין"] התיבות הובאו בסוגריים. הסידור מציין שאין כל טעם להשמיטן: מי שטרם נשא אשה ראוי שיבקש על העתיד, ומי שכבר יש לו בנים – "דעבדין רעותך", שבניו יהיו עושי רצונו של מקום.

[המשך עמ' 456] דְלִיבָּאִי וְלִבָּא דְכָל־עַמָּךְ יִשְׂרָאֵל לְטַב וּלְחַיִּין וְלִשְׁלָם, אָמֵן:

[הוראה בסידור] בימי החנוכה אומרים פסוקים מיוחדים שנזכר בהם "נר" או "אור", כאשר כל פסוק אומרו החזן תחילה והציבור שב ואומר אחריו (ראה הלכה 768):

נֵר־לְרַגְלִי דְבָרֶךָ, וְאוֹר לִנְתִיבָתִי:

נֵר יְיָ נִשְׁמַת אָדָם, חֹפֵשׂ כָּל־חַדְרֵי־בָטֶן:

כִּי נֵר מִצְוָה וְתוֹרָה אוֹר, וְדֶרֶךְ חַיִּים תּוֹכְחוֹת מוּסָר:

כִּי־אַתָּה תָּאִיר נֵרִי, יְיָ אֱלֹהַי יַגִּיהַּ חָשְׁכִּי:

אוֹר זָרֻעַ לַצַּדִּיק, וּלְיִשְׁרֵי־לֵב שִׂמְחָה:

וְאֹרַח צַדִּיקִים כְּאוֹר נֹגַהּ, הוֹלֵךְ וָאוֹר עַד־נְכוֹן הַיּוֹם:

לְדָוִד, יְיָ אוֹרִי וְיִשְׁעִי מִמִּי אִירָא, יְיָ מָעוֹז־חַיַּי מִמִּי אֶפְחָד:

קוּמִי אוֹרִי כִּי בָא אוֹרֵךְ, וּכְבוֹד יְיָ עָלַיִךְ זָרָח:

מֵאֵין כָּמוֹךָ יְיָ גָּדוֹל אַתָּה וְגָדוֹל שִׁמְךָ בִּגְבוּרָה:

[הוראה בסידור] נוהגים להוסיף ולומר "מי שברך" לחולים (עמ' 1251), וכן בקשות שונות כמו תפילה לחיילי צה"ל ולשלום המדינה, מאחר וזמן פתיחת ההיכל הוא עת רחמים.

מי שברך לחיילי צה"ל

מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב, הוּא יְבָרֵךְ אֶת־חַיְלֵי צְבָא הַהֲגָנָה לְיִשְׂרָאֵל וְאַנְשֵׁי כֹּחוֹת הַבִּטָּחוֹן, הָעוֹמְדִים עַל־מִשְׁמַר אַרְצֵנוּ וְעָרֵי אֱלֹהֵינוּ, מִגְּבוּל הַלְּבָנוֹן וְעַד מִדְבַּר מִצְרַיִם וּמִן הַיָּם הַגָּדוֹל עַד לִבוֹא הָעֲרָבָה, בַּיַּבָּשָׁה בָּאֲוִיר וּבַיָּם. יִתֵּן יְיָ אֶת אוֹיְבֵינוּ הַקָּמִים עָלֵינוּ נִגָּפִים לִפְנֵיהֶם. הַקָּדוֹשׁ בָּרוּךְ־הוּא יִשְׁמֹר וְיַצִּיל אֶת־חַיָּלֵינוּ מִכָּל־צָרָה וְצוּקָה, וּמִכָּל־נֶגַע וּמַחֲלָה, וְיִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל־מַעֲשֵׂי יְדֵיהֶם.

[המשך – בעמוד הבא, שלא צולם בצילום זה; ראה את "tzahal" בכלי הקיים, אומת מול הצילומים הקודמים]

[הוראה בסידור] בעת פתיחת ההיכל בשחרית של שבת, אומרים שמונה פסוקים כנגד שמונת העולים לתורה – שבעת העולים והמפטיר. וביום טוב שחל בחול, עולים חמישה לתורה, לפיכך מתחילים מהפסוק "יְהִי יְיָ אֱלֹהֵינוּ עִמָּנוּ".

אַתָּה הָרְאֵתָ לָדַעַת, כִּי יְיָ הוּא הָאֱלֹהִים, אֵין עוֹד מִלְּבַדּוֹ: אֵין־כָּמוֹךָ בָאֱלֹהִים | אֲדֹנָי, וְאֵין כְּמַעֲשֶׂיךָ: יְהִי יְיָ אֱלֹהֵינוּ עִמָּנוּ כַּאֲשֶׁר הָיָה עִם־אֲבֹתֵינוּ, אַל־יַעַזְבֵנוּ וְאַל־יִטְּשֵׁנוּ: הוֹשִׁיעָה | אֶת־עַמֶּךָ וּבָרֵךְ אֶת־נַחֲלָתֶךָ, וּרְעֵם וְנַשְּׂאֵם עַד־הָעוֹלָם: וַיְהִי בִּנְסֹעַ הָאָרֹן וַיֹּאמֶר מֹשֶׁה, קוּמָה | יְיָ וְיָפֻצוּ אֹיְבֶיךָ, וְיָנֻסוּ מְשַׂנְאֶיךָ מִפָּנֶיךָ: קוּמָה יְיָ לִמְנוּחָתֶךָ, אַתָּה וַאֲרוֹן עֻזֶּךָ: כֹּהֲנֶיךָ יִלְבְּשׁוּ־צֶדֶק, וַחֲסִידֶיךָ יְרַנֵּנוּ: בַּעֲבוּר דָּוִד עַבְדֶּךָ, אַל־תָּשֵׁב פְּנֵי מְשִׁיחֶךָ: פִּתְחוּ־לִי שַׁעֲרֵי־צֶדֶק, אָבֹא־בָם אוֹדֶה יָהּ: זֶה הַשַּׁעַר לַיְיָ, צַדִּיקִים יָבֹאוּ בוֹ:

כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה, וּדְבַר־יְיָ מִירוּשָׁלִָם:

בָּרוּךְ הַבָּא בָּרוּךְ הַבָּא, בָּרוּךְ הַמָּקוֹם בָּרוּךְ הוּא.

בָּרוּךְ שֶׁנָּתַן תּוֹרָה לְעַמּוֹ יִשְׂרָאֵל בִּקְדֻשָּׁה וּבְטָהֳרָה בָּרוּךְ הוּא.

[הוראה בסידור] בקהילות מראכש והמחוז, יש שנהגו להוסיף ולומר (ראה הלכה 197):

הֲרֵינִי מְקַבֵּל קַבָּלַת הַתּוֹרָה, שֶׁנִּתְּנָה עַל יְדֵי מֹשֶׁה רַבֵּנוּ עָלָיו הַשָּׁלוֹם.

וַהֲרֵינִי מְקַבְּלָהּ מֵחָדָשׁ, קַבָּלָה גְּמוּרָה וּשְׁלֵמָה. וְקַבָּלָה זוֹ הִיא בְּכָל־לִבִּי, וּבְכָל־נַפְשִׁי וּבְכָל־מְאוֹדִי, וְרַמַ"ח אֵבָרַי וְשַׁס"ה גִּידַי. וְקַבָּלָה זוֹ הִיא בְּלִי שׁוּם זֵכֶר אֹנֶס כְּלָל וְעִקָּר, אֶלָּא בְּשִׂמְחָה וּבְטוּב לֵבָב. וְקַבָּלָה זוֹ הִיא לְדַעַת רַבִּי שִׁמְעוֹן בַּר יוֹחַאי הַקָּדוֹשׁ, וּלְדַעַת רַבִּי אֶלְעָזָר בְּנוֹ הַקָּדוֹשׁ, וּלְדַעַת יוֹתָם בֶּן עֻזִּיָּהוּ הַקָּדוֹשׁ, וּפִי כְפִיהֶם, וְהוֹדָאָתִי כְּהוֹדָאָתָם, וְקַבָּלָתִי כְּקַבָּלָתָם, וְדַעְתִּי כְּדַעְתָּם. וַהֲרֵינִי סוֹמֵךְ עֲלֵיהֶם דֶּרֶךְ כְּלָל וּפְרָט וּבִכְלָל כָּל־יִשְׂרָאֵל. וְקַבָּלָה זוֹ תְהֵא עָלַי וְעַל כָּל־זַרְעִי וְזֶרַע זַרְעִי, עַד סוֹף כָּל־הָעוֹלָם, עַד וָעַד בִּכְלָל. אָמֵן:

רִבּוֹנוֹ שֶׁל עוֹלָם, מַלֵּא מִשְׁאֲלוֹתַי לְטוֹבָה, וְהָפֵק רְצוֹנִי, וְתֵן שְׁאֵלָתִי, וְזַכֵּנִי עִם־כָּל־בְּנֵי בֵיתִי לַעֲשׂוֹת רְצוֹנְךָ בְּלֵבָב שָׁלֵם, וּמַלְּטֵנִי מִיֵּצֶר הָרָע וּפְגָעִים רָעִים, וְתֵן חֶלְקִי בְּתוֹרָתֶךָ. וְזַכֵּנִי לִלְמוֹד תּוֹרָה וְחָכְמָה, וְלַעֲשׂוֹת מַעֲשִׂים טוֹבִים לְפָנֶיךָ, לָלֶכֶת בְּדַרְכֵי יְשָׁרִים וּלְחַיֵּי הָעוֹלָם הַבָּא. וְתִשְׁמְרֵנִי מִפֶּגַע רָע, וּמִיֵּצֶר הָרָע, וּמִכָּל־שָׁעוֹת רָעוֹת הַמִּתְרַגְּשׁוֹת וּבָאוֹת לָעוֹלָם. וְהַבּוֹטֵחַ בַּיְיָ חֶסֶד יְסוֹבְבֶנּוּ: אָמֵן סֶלָה:

[הוראה בסידור] ב"בְּרִיךְ שְׁמֵהּ", באמירת התיבות "דְּסָגִידְנָא קַמֵּהּ" מנהגנו לכרוע מעט (ראה הלכה 199).

רִבִּי שִׁמְעוֹן פָּתַח: כַּד מַפְקִין סֵפֶר תּוֹרָה בִּצְבוּרָא לְמִקְרֵי בֵיהּ, מִתְפַּתְּחִין תַּרְעֵי דְּרַחֲמֵי, וּמְעוֹרְרִין אֶת־הָאַהֲבָה לְעֵלָּא, וּמִבָּעֵי לֵיהּ לְבַר־נָשׁ לְמֵימַר הָכִי:

בְּרִיךְ שְׁמֵהּ דְּמָארֵי עָלְמָא, בְּרִיךְ כִּתְרָךְ וְאַתְרָךְ. יְהֵא רְעוּתָךְ עִם עַמָּךְ יִשְׂרָאֵל לְעָלַם, וּפֻרְקַן יְמִינָךְ אַחֲזֵי לְעַמָּךְ בְּבֵית מַקְדְּשָׁךְ, וּלְאַמְטוֹיֵי לָנָא מִטּוּב נְהוֹרָךְ, וּלְקַבֵּל צְלוֹתָנָא בְּרַחֲמִין. יְהֵא רַעֲוָא קֳדָמָךְ דְּתוֹרִיךְ לָן חַיִּין בְּטִיבוּ,

[המשך בצילום 13]

מתי משתמשים בדף

ברגע שפותחים את ארון הקודש בשבת בבוקר, לפני הוצאת ספר התורה.

מאיפה מתחילים

מ'בריך שמיה'. מה שלפניו אופציונלי ומובא בסוף הדף, כדי שלא יעכב את הקורא.

שמירה על המנהג

הדף מיועד לשמור על מנהג מרוקו גם הרחק מהבית, כך שהמתפללים ישמעו את הנוסח המוכר להם.

הערה לשליח הציבור

הנוסח מוקלד מילה במילה מהסידור, מנוקד, ובסדר הקריאה בפועל. אפשר לעבור על הדף לפני התפילה, כדי שהקריאה בציבור תהיה שוטפת וברורה.

איך להשתמש בדף

אפשר לקרוא ישר מהטלפון, או להדפיס את הדף בעיצוב הנקי ולהניח על התיבה.

שאלות נפוצות

מאיפה מתחילים?

מ'בריך שמיה'. 'הריני מקבל' ו'ריבונו של עולם' אופציונליים.

איפה התפילה למדינה?

בדף שחרית של שבת, ובכלי מי שברך.

הנוסח מרוקאי?

כן, מסידור כמנהג מרוקו, מילה במילה, מנוקד.

מה אחרי פתיחת ההיכל?

הוצאת ספר התורה וקריאה, בדף שחרית של שבת.

אפשר להדפיס?

כן, יש עיצוב הדפסה נקי.

האם הדף בחינם?

כן, בחינם, בלי הרשמה ובלי פרסומות, מיזם חסד.

עוד דפים בערכה

תפילת שחרית של שבת נוסח מרוקוקריאת התורה שני וחמישי נוסח מרוקושבת מברכים תפילהמי שברך נוסח מרוקאישבת חתן תפילה
הנוסח הוקלד מילה במילה מסידור כמנהג מרוקו. מקומות שסומנו "לאימות" טרם אומתו מול צילום חד.
