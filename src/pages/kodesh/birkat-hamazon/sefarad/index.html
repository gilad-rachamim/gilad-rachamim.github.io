ארבע הברכות, והתוספות לשבת, לחג, לראש חודש, לחנוכה ולפורים – כל אחת במקומה

לפני ברכת המזון: על נהרות בבל (בחול) ושיר המעלות (בשבת ובחג)

בימי חול:

עַל־נַהֲרוֹת בָּבֶל שָׁם יָשַׁבְנוּ גַּם־בָּכִינוּ בְּזָכְרֵנוּ אֶת־צִיּוֹן: עַל־עֲרָבִים בְּתוֹכָהּ תָּלִינוּ כִּנֹּרוֹתֵינוּ: כִּי שָׁם שְׁאֵלוּנוּ שׁוֹבֵינוּ דִּבְרֵי־שִׁיר וְתוֹלָלֵינוּ שִׂמְחָה שִׁירוּ לָנוּ מִשִּׁיר צִיּוֹן: אֵיךְ נָשִׁיר אֶת־שִׁיר־ה' עַל אַדְמַת נֵכָר: אִם־אֶשְׁכָּחֵךְ יְרוּשָׁלָםִ תִּשְׁכַּח יְמִינִי: תִּדְבַּק לְשׁוֹנִי לְחִכִּי אִם־לֹא אֶזְכְּרֵכִי אִם־לֹא אַעֲלֶה אֶת־יְרוּשָׁלַםִ עַל רֹאשׁ שִׂמְחָתִי: זְכֹר ה' לִבְנֵי אֱדוֹם אֵת יוֹם יְרוּשָׁלָםִ הָאֹמְרִים עָרוּ עָרוּ עַד הַיְסוֹד בָּהּ: בַּת־בָּבֶל הַשְּׁדוּדָה אַשְׁרֵי שֶׁיְּשַׁלֶּם־לָךְ אֶת־גְּמוּלֵךְ שֶׁגָּמַלְתְּ לָנוּ: אַשְׁרֵי שֶׁיֹּאחֵז וְנִפֵּץ אֶת־עֹלָלַיִךְ אֶל־הַסָּלַע:

בשבת ובחג:

שִׁיר הַמַּעֲלוֹת בְּשׁוּב ה' אֶת־שִׁיבַת צִיּוֹן הָיִינוּ כְּחֹלְמִים: אָז יִמָּלֵא שְׂחוֹק פִּינוּ וּלְשׁוֹנֵנוּ רִנָּה, אָז יֹאמְרוּ בַגּוֹיִם הִגְדִּיל ה' לַעֲשׂוֹת עִם־אֵלֶּה: הִגְדִּיל ה' לַעֲשׂוֹת עִמָּנוּ, הָיִינוּ שְׂמֵחִים: שׁוּבָה ה' אֶת־שְׁבִיתֵנוּ כַּאֲפִיקִים בַּנֶּגֶב: הַזֹּרְעִים בְּדִמְעָה, בְּרִנָּה יִקְצֹרוּ: הָלוֹךְ יֵלֵךְ וּבָכֹה, נֹשֵׂא מֶשֶׁךְ־הַזָּרַע, בֹּא־יָבֹא בְרִנָּה, נֹשֵׂא אֲלֻמֹּתָיו:

ברכה זו משה רבינו ע"ה תקנה כשירד המן לישראל

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם הַזָּן אֶת־הָעוֹלָם כֻּלּוֹ בְּטוּבוֹ בְּחֵן בְּחֶסֶד וּבְרַחֲמִים הוּא נוֹתֵן לֶחֶם לְכָל־בָּשָׂר כִּי לְעוֹלָם חַסְדּוֹ וּבְטוּבוֹ הַגָּדוֹל תָּמִיד לֹא־חָסַר לָנוּ וְאַל־יֶחְסַר לָנוּ מָזוֹן לְעוֹלָם וָעֶד בַּעֲבוּר שְׁמוֹ הַגָּדוֹל כִּי הוּא אֵל זָן וּמְפַרְנֵס לַכֹּל וּמֵטִיב לַכֹּל וּמֵכִין מָזוֹן לְכָל־בְּרִיּוֹתָיו אֲשֶׁר בָּרָא. כָּאָמוּר, פּוֹתֵחַ אֶת־יָדֶךָ וּמַשְׂבִּיעַ לְכָל־חַי רָצוֹן: בָּרוּךְ אַתָּה ה' הַזָּן אֶת־הַכֹּל:

ברכה זו יהושע תקנה כשנכנסו ישראל לארץ

נוֹדֶה לְּךָ ה' אֱלֹקֵינוּ עַל שֶׁהִנְחַלְתָּ לַאֲבוֹתֵינוּ אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה וְעַל שֶׁהוֹצֵאתָנוּ ה' אֱלֹקֵינוּ מֵאֶרֶץ מִצְרַיִם וּפְדִיתָנוּ מִבֵּית עֲבָדִים וְעַל בְּרִיתְךָ שֶׁחָתַמְתָּ בִּבְשָׂרֵנוּ וְעַל תּוֹרָתְךָ שֶׁלִּמַּדְתָּנוּ וְעַל חֻקֶּיךָ שֶׁהוֹדַעְתָּנוּ וְעַל חַיִּים חֵן וָחֶסֶד שֶׁחוֹנַנְתָּנוּ וְעַל אֲכִילַת מָזוֹן שָׁאַתָּה זָן וּמְפַרְנֵס אוֹתָנוּ תָּמִיד בְּכָל־יוֹם וּבְכָל־עֵת וּבְכָל שָׁעָה:

בחנוכה ובפורים: על הניסים

וְעַל הַנִּסִּים וְעַל הַפֻּרְקָן וְעַל הַגְּבוּרוֹת וְעַל הַתְּשׁוּעוֹת וְעַל הַנִּפְלָאוֹת וְעַל הַנֶּחָמוֹת וְעַל הַמִּלְחָמוֹת שֶׁעָשִׂיתָ לַאֲבוֹתֵינוּ בַּיָּמִים הָהֵם בִּזְּמַן הַזֶּה:

בחנוכה:

בִּימֵי מַתִּתְיָהוּ בֶּן־יוֹחָנָן כֹּהֵן גָּדוֹל חַשְׁמוֹנָאִי וּבָנָיו כְּשֶׁעָמְדָה מַלְכוּת יָוָן הָרְשָׁעָה עַל־עַמְּךָ יִשְׂרָאֵל לְהַשְׁכִּיחָם תּוֹרָתֶךָ וּלְהַעֲבִירָם מֵחֻקֵּי רְצוֹנֶךָ: וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים עָמַדְתָּ לָהֶם בְּעֵת צָרָתָם רַבְתָּ אֶת־רִיבָם דַּנְתָּ אֶת־דִּינָם נָקַמְתָּ אֶת־נִקְמָתָם מָסַרְתָּ גִבּוֹרִים בְּיַד חַלָּשִׁים וְרַבִּים בְּיַד מְעַטִּים וּטְמֵאִים בְּיַד טְהוֹרִים וּרְשָׁעִים בְּיַד צַדִּיקִים וְזֵדִים בְּיַד עוֹסְקֵי תוֹרָתֶךָ וּלְךָ עָשִׂיתָ שֵׁם גָּדוֹל וְקָדוֹשׁ בְּעוֹלָמֶךָ וּלְעַמְּךָ יִשְׂרָאֵל עָשִׂיתָ תְּשׁוּעָה גְדוֹלָה וּפֻרְקָן כְּהַיּוֹם הַזֶּה וְאַחַר כַּךְ בָּאוּ בָנֶיךָ לִדְבִיר בֵּיתֶךָ וּפִנּוּ אֶת הֵיכָלֶךָ וְטִהֲרוּ אֶת מִקְדָּשֶׁךָ וְהִדְלִיקוּ נֵרוֹת בְּחַצְרוֹת קָדְשֶׁךָ וְקָבְעוּ שְׁמוֹנַת יְמֵי חֲנֻכָּה אֵלּוּ לְהוֹדוֹת וּלְהַלֵּל לְשִׁמְךָ הַגָּדוֹל:

בפורים:

בִּימֵי מָרְדְּכַי וְאֶסְתֵּר בְּשׁוּשַׁן הַבִּירָה כְּשֶׁעָמַד עֲלֵיהֶם הָמָן הָרָשָׁע בִּקֵּשׁ לְהַשְׁמִיד לַהֲרוֹג וּלְאַבֵּד אֶת־כָּל־הַיְּהוּדִים מִנַּעַר וְעַד־זָקֵן טַף וְנָשִׁים בְּיוֹם אֶחָד בִּשְׁלוֹשָׁה עָשָׂר לְחֹדֶשׁ שְׁנֵים־עָשָׂר הוּא חֹדֶשׁ אֲדָר וּשְׁלָלָם לָבוֹז: וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים הֵפַרְתָּ אֶת־עֲצָתוֹ וְקִלְקַלְתָּ אֶת מַחֲשַׁבְתּוֹ וַהֲשֵׁבוֹתָ לּוֹ גְּמוּלוֹ בְּרֹאשׁוֹ וְתָלוּ אוֹתוֹ וְאֶת־בָּנָיו עַל־הָעֵץ:

וְעַל הַכֹּל ה' אֱלֹקֵינוּ אֲנַחְנוּ מוֹדִים לָךְ וּמְבָרְכִים אוֹתָךְ יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל־חַי תָּמִיד לְעוֹלָם וָעֶד כַּכָּתוּב וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת־ה' אֱלֹקֶיךָ עַל־הָאָרֶץ הַטּוֹבָה אֲשֶׁר נָתַן־לָךְ בָּרוּךְ אַתָּה ה' עַל־הָאָרֶץ וְעַל־הַמָּזוֹן:

בונה ירושלים דוד ושלמה תקנוה. דוד תקן על ישראל עמך ועל ירושלים עירך. ושלמה תיקן ועל הבית הגדול וכו'.

רַחֵם ה' אֱלֹקֵינוּ עַל־יִשְׂרָאֵל עַמֶּךָ וְעַל יְרוּשָׁלַיִם עִירֶךָ וְעַל צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ וְעַל מַלְכוּת בֵּית דָּוִד מְשִׁיחֶךָ וְעַל־הַבַּיִת הַגָּדוֹל וְהַקָּדוֹשׁ שֶׁנִּקְרָא שִׁמְךָ עָלָיו אֱלֹקֵינוּ אָבִינוּ רְעֵנוּ זוּנֵנוּ פַּרְנְסֵנוּ וְכַלְכְּלֵנוּ וְהַרְוִיחֵנוּ וְהַרְוַח־לָנוּ ה' אֱלֹקֵינוּ מְהֵרָה מִכָּל־צָרוֹתֵינוּ וְנָא אַל־תַּצְרִיכֵנוּ ה' אֱלֹקֵינוּ לֹא לִידֵי מַתְּנַת בָּשָׂר וָדָם וְלֹא לִידֵי הַלְוָאָתָם כִּי אִם לְיָדְךָ הַמְּלֵאָה הַפְּתוּחָה הַקְּדוֹשָׁה וְהָרְחָבָה שֶׁלֹּא נֵבוֹשׁ וְלֹא נִכָּלֵם לְעוֹלָם וָעֶד:

בשבת: רצה

זָכְרֵנוּ ה' אֱלֹקֵינוּ בּוֹ לְטוֹבָה, וּפָקְדֵנוּ בוֹ לִבְרָכָה, וְהוֹשִׁיעֵנוּ בוֹ לְחַיִּים טוֹבִים, וּבִדְבַר יְשׁוּעָה וְרַחֲמִים חוּס וְחָנֵּנוּ, וְרַחֵם עָלֵינוּ וְהוֹשִׁיעֵנוּ, כִּי אֵלֶיךָ עֵינֵינוּ, כִּי אֵל מֶלֶךְ חַנּוּן וְרַחוּם אָתָּה:

הטוב והמטיב ביבנה תקנוה ביום שנתנו הרוגי ביתר לקבורה. הטוב שלא הסריחו. והמטיב שנתנו לקבורה.

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הָאֵל אָבִינוּ, מַלְכֵּנוּ, אַדִּירֵנוּ בּוֹרְאֵנוּ, גּוֹאֲלֵנוּ, יוֹצְרֵנוּ, קְדוֹשֵׁנוּ קְדוֹשׁ יַעֲקֹב, רוֹעֵנוּ רוֹעֵה יִשְׂרָאֵל, הַמֶּלֶךְ הַטּוֹב, וְהַמֵּטִיב לַכֹּל, שֶׁבְּכָל יוֹם וָיוֹם הוּא הֵטִיב, הוּא מֵטִיב, הוּא יֵיטִיב לָנוּ, הוּא גְמָלָנוּ, הוּא גוֹמְלֵנוּ, הוּא יִגְמְלֵנוּ לָעַד לְחֵן וּלְחֶסֶד וּלְרַחֲמִים וּלְרֶוַח הַצָּלָה וְהַצְלָחָה בְּרָכָה וִישׁוּעָה, נֶחָמָה, פַּרְנָסָה וְכַלְכָּלָה, וְרַחֲמִים, וְחַיִּים וְשָׁלוֹם, וְכָל־טוֹב, וּמִכָּל־טוּב לְעוֹלָם אַל יְחַסְּרֵנוּ:

הָרַחֲמָן הוּא יִמְלוֹךְ עָלֵינוּ לְעוֹלָם וָעֶד:

הָרַחֲמָן הוּא יִתְבָּרֵךְ בַּשָּׁמַיִם וּבָאָרֶץ:

הָרַחֲמָן הוּא יְפַרְנְסֵנוּ בְּכָבוֹד:

הָרַחֲמָן הוּא יְבָרֵךְ אוֹתִי (וְאֶת־אִשְׁתִּי/בַּעֲלִי וְאֶת־זַרְעִי) וְאֶת־כָּל־אֲשֶׁר לִי.

אוֹתָנוּ וְאֶת־כָּל־אֲשֶׁר לָנוּ, כְּמוֹ שֶׁנִּתְבָּרְכוּ אֲבוֹתֵינוּ, אַבְרָהָם יִצְחָק וְיַעֲקֹב: בַּכֹּל, מִכֹּל, כֹּל, כֵּן יְבָרֵךְ אוֹתָנוּ כֻּלָּנוּ יַחַד, בִּבְרָכָה שְׁלֵמָה, וְנֹאמַר אָמֵן:

בורא נפשות

אחרי פירות העץ (שאינם משבעת המינים), ירקות, משקים, בשר, ביצים וכל השאר

על פירות האילן חוץ מז' מינים ופירות האדמה וירקות ועל כל המשקין ועל דבר שאין גידולו מן הארץ מברכין אחר אכילתן זה:

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם בּוֹרֵא נְפָשׁוֹת רַבּוֹת וְחֶסְרוֹנָן עַל כָּל מַה שֶׁבָּרָאתָ לְהַחֲיוֹת בָּהֶם נֶפֶשׁ כָּל חָי. בָּרוּךְ חֵי הָעוֹלָמִים:

המילים והניקוד מתוך ספריא, מילה במילה (סידור "מצודה", 1981 (רישיון CC-BY)). טעמי המקרא הורדו, והשמות הקדושים כתובים בכינוי (ה', אלוקינו) ואומרים אותם כרגיל. בדיקה בקוד מוודאת שהטקסט זהה למקור. מי שנוהג אחרת, הולך לפי המנהג שלו; ובכל שאלה, שואלים רב.

לשלוח בווטסאפ

ברכת המזון נוסח ספרד, מוכנה בטלפון

כאן כתובה ברכת המזון בנוסח ספרד (חסידי) בלבד, מההתחלה ועד הסוף, כדי שמי שנוהג בנוסח הזה יפתח את הדף ויברך, בלי לבחור ובלי לדלג בין נוסחים. היא כתובה בניקוד מלא ובאות גדולה, בלי פרסומות באמצע, ואפשר להגדיל את האותיות או לעבור לרקע כהה בכפתורים שבראש הדף.

הכול במקום: זימון, רצה, יעלה ויבוא ועל הניסים

הזימון בראש הדף, ואחריו ארבע הברכות. כל תוספת מופיעה בדיוק איפה שאומרים אותה, בתוך מסגרת עם כותרת: רצה בשבת, יעלה ויבוא בראש חודש, בחגים ובחול המועד, על הניסים בחנוכה ובפורים, והרחמן המיוחד לכל יום. ככה לא שוכחים ולא מוסיפים בטעות. מי ששכח רצה או יעלה ויבוא ימצא את מה שאומרים בחלק "מי ששכח".

וגם מעין שלוש ובורא נפשות

בסוף הדף נמצאות שתי הברכות האחרונות האחרות, באותו נוסח: ברכה מעין שלוש (על המחיה, על הגפן ועל העץ) ובורא נפשות. יש להן גם דפים משלהן: מעין שלוש ובורא נפשות.

בשאר הנוסחים

ברכת המזון עדות המזרח · ברכת המזון אשכנז. וברכון מודפס ומעוצב לשמחה, עם עמוד הקדשה, נמצא בברכונים להדפסה עם הקדשה.

איך משתמשים בדף

ברכת המזון ספרד נפתח מהר בכל טלפון, גם ישן. הכפתורים א+ וא- מגדילים ומקטינים את האותיות, הירח מחליף לרקע כהה, והמסך לא נכבה באמצע הקריאה. בתפריט הדפדפן אפשר לבחור "הוספה למסך הבית", והדף ייפתח כמו אפליקציה. ומי שרוצה לשתף, שולח את הקישור בווטסאפ. אין כאן פרסומות, אין הרשמה, ולא נאספים פרטים אישיים.

שאלות ותשובות

מאיזה סידור לקוח נוסח ספרד?

ברכת המזון בנוסח ספרד לקוחה מסידור "מצודה", 1981 (רישיון CC-BY), כמו שהיא בספריא, מילה במילה. טעמי המקרא הורדו, והשמות הקדושים כתובים בכינוי.

מה יש בדף של ברכת המזון ספרד?

הזימון, ארבע הברכות, והתוספות במקומן: רצה בשבת, יעלה ויבוא בראש חודש ובחגים, על הניסים בחנוכה ובפורים והרחמן לימים מיוחדים. בסוף הדף יש גם ברכה מעין שלוש ובורא נפשות באותו נוסח.

כמה אנשים צריך בשביל זימון?

שְׁלשָׁה שֶׁאָכְלוּ בְּיַחַד צְרִיכִין לְבָרֵךְ בְּזִמּוּן (קיצור שולחן ערוך מה, א). נוסח הזימון נמצא בראש הדף.

שכחתי לומר רצה בשבת. מה עושים?

בקיצור שולחן ערוך: אִם נִזְכַּר קֹדֶם שֶׁהִזְכִּיר אֶת הַשֵּׁם מִן הַבְּרָכָה שֶׁל בָּרוּךְ אַתָּה ה' בּוֹנֶה בְּרַחֲמָיו יְרוּשָׁלָיִם, אוֹמֵר שָׁם רְצֵה אוֹ יַעֲלֶה וְיָבֹא (מד, יב). בכל ספק למעשה, שואלים רב.

נוהגים בנוסח אחר?

ברכת המזון בשאר הנוסחים: ברכת המזון עדות המזרח · ברכת המזון אשכנז. ובדף הראשי אפשר לעבור בין הנוסחים בלחיצה: ברכת המזון המלאה.
