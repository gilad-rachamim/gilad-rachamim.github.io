אשִׁיר מִזְמוֹר לְדָוִד: בנָכוֹן לִבִּי אֱלֹקִים אָשִׁירָה וַאֲזַמְּרָה אַף־כְּבוֹדִי: געוּרָה הַנֵּבֶל וְכִנּוֹר אָעִירָה שָּׁחַר: דאוֹדְךָ בָעַמִּים ה' וַאֲזַמֶּרְךָ בַּלְאֻמִּים: הכִּי־גָדוֹל מֵעַל־שָׁמַיִם חַסְדֶּךָ וְעַד־שְׁחָקִים אֲמִתֶּךָ: ורוּמָה עַל־שָׁמַיִם אֱלֹקִים וְעַל כָּל־הָאָרֶץ כְּבוֹדֶךָ: זלְמַעַן יֵחָלְצוּן יְדִידֶיךָ הוֹשִׁיעָה יְמִינְךָ וַעֲנֵנִי: חאֱלֹקִים דִּבֶּר בְּקָדְשׁוֹ אֶעְלֹזָה אֲחַלְּקָה שְׁכֶם וְעֵמֶק סֻכּוֹת אֲמַדֵּד: טלִי גִלְעָד לִי מְנַשֶּׁה וְאֶפְרַיִם מָעוֹז רֹאשִׁי יְהוּדָה מְחֹקְקִי: ימוֹאָב סִיר רַחְצִי עַל־אֱדוֹם אַשְׁלִיךְ נַעֲלִי עֲלֵי־פְלֶשֶׁת אֶתְרוֹעָע: יאמִי יֹבִלֵנִי עִיר מִבְצָר מִי נָחַנִי עַד־אֱדוֹם: יבהֲלֹא־אֱלֹקִים זְנַחְתָּנוּ וְלֹא־תֵצֵא אֱלֹקִים בְּצִבְאֹתֵינוּ: יגהָבָה־לָּנוּ עֶזְרָת מִצָּר וְשָׁוְא תְּשׁוּעַת אָדָם: ידבֵּאלֹקִים נַעֲשֶׂה־חָיִל וְהוּא יָבוּס צָרֵינוּ:

על תהילים פרק קח

"נכון ליבי אלוקים": פרק שמורכב מסוף פרק נז ומסוף פרק ס, שירה ובקשת עזרה במלחמה.

במחזור החודשי הוא נקרא ביום כג בחודש, ובמחזור השבועי ביום שישי (פרקים קז–קיט). הפרק בספר חמישי של תהילים (פרקים קז–קנ), נפתח במילים "שיר מזמור לדוד נכון לבי אלקים אשירה", ויש בו 14 פסוקים.

לכל הפרקים, ולחלוקה לפי ימי החודש וימי השבוע: ספר תהילים המלא. רוצים לגמור את הספר יחד עם עוד אנשים, לעילוי נשמה או לרפואה? סיום תהילים בקישור. ולפי אותיות השם: תהילים לפי שם.
