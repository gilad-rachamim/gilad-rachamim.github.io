אלַמְנַצֵּחַ לִבְנֵי־קֹרַח מִזְמוֹר: ברָצִיתָ ה' אַרְצֶךָ שַׁבְתָּ שְׁבִית יַעֲקֹב: גנָשָׂאתָ עֲוֹן עַמֶּךָ כִּסִּיתָ כָל־חַטָּאתָם סֶלָה: דאָסַפְתָּ כָל־עֶבְרָתֶךָ הֱשִׁיבוֹתָ מֵחֲרוֹן אַפֶּךָ: השׁוּבֵנוּ אֱלֹקֵי יִשְׁעֵנוּ וְהָפֵר כַּעַסְךָ עִמָּנוּ: והַלְעוֹלָם תֶּאֱנַף־בָּנוּ תִּמְשֹׁךְ אַפְּךָ לְדֹר וָדֹר: זהֲלֹא־אַתָּה תָּשׁוּב תְּחַיֵּנוּ וְעַמְּךָ יִשְׂמְחוּ־בָךְ: חהַרְאֵנוּ ה' חַסְדֶּךָ וְיֶשְׁעֲךָ תִּתֶּן־לָנוּ: טאֶשְׁמְעָה מַה־יְדַבֵּר הָאֵל ה' כִּי יְדַבֵּר שָׁלוֹם אֶל־עַמּוֹ וְאֶל־חֲסִידָיו וְאַל־יָשׁוּבוּ לְכִסְלָה: יאַךְ קָרוֹב לִירֵאָיו יִשְׁעוֹ לִשְׁכֹּן כָּבוֹד בְּאַרְצֵנוּ: יאחֶסֶד־וֶאֱמֶת נִפְגָּשׁוּ צֶדֶק וְשָׁלוֹם נָשָׁקוּ: יבאֱמֶת מֵאֶרֶץ תִּצְמָח וְצֶדֶק מִשָּׁמַיִם נִשְׁקָף: יגגַּם־ה' יִתֵּן הַטּוֹב וְאַרְצֵנוּ תִּתֵּן יְבוּלָהּ: ידצֶדֶק לְפָנָיו יְהַלֵּךְ וְיָשֵׂם לְדֶרֶךְ פְּעָמָיו:
