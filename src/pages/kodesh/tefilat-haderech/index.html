תפילת הדרך

אומרים אחרי שיוצאים מהעיר, פעם אחת ביום

היוצא לדרך, ושוהה בנסיעתו מחוץ לעיר לפחות מהלך שיעור פרסה (72 דקות) אומר תפלת הדרך. ברכה זו מברך לאחר שיצא מהעיר, וטוב לסמוך אותה לברכה אחרונה של אכילה או שתיה או אשר יצר.

יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, שֶׁתּוֹלִיכֵנוּ לְשָׁלוֹם, וְתַצְעִידֵנוּ לְשָׁלוֹם, וְתִסְמְכֵנוּ לְשָׁלוֹם, (אם חוזר בו ביום מוסיף וְתַחֲזִירֵנוּ לְשָׁלוֹם), וְתַצִּילֵנוּ מִכַּף כָּל־אוֹיֵב וְאוֹרֵב בַּדֶּרֶךְ, וְתִשְׁלַח בְּרָכָה בְּמַעֲשֵׂה יָדֵינוּ, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל־רוֹאֵינוּ. בָּרוּךְ (אַתָּה ה') שׁוֹמֵעַ תְּפִלָּה:

ויש מוסיפים פסוקים אלה לשמירה

וְיַעֲקֹב הָלַךְ לְדַרְכּוֹ וַיִּפְגְּעוּ־בוֹ מַלְאֲכֵי אֱלֹקִים: וַיֹּאמֶר יַעֲקֹב כַּאֲשֶׁר רָאָם מַחֲנֵה אֱלֹקִים זֶה וַיִּקְרָא שֵׁם־הַמָּקוֹם הַהוּא מַחֲנָיִם:

וַיִּסָּעוּ וַיְהִי חִתַּת אֱלֹקִים עַל־הֶעָרִים אֲשֶׁר סְבִיבוֹתֵיהֶם וְלֹא רָדְפוּ אַחֲרֵי בְּנֵי יַעֲקֹב: לִישׁוּעָתְךָ קִוִּיתִי ה': ג' פעמים

יְבָרֶכְךָ ה' וְיִשְׁמְרֶךָ: יָאֵר ה' פָּנָיו אֵלֶיךָ וִיחֻנֶּךָּ: יִשָּׂא ה' פָּנָיו אֵלֶיךָ וְיָשֵׂם לְךָ שָׁלוֹם: ג' פעמים

הִנֵּה אָנֹכִי שֹׁלֵחַ מַלְאָךְ לְפָנֶיךָ לִשְׁמָרְךָ בַּדָּרֶךְ וְלַהֲבִיאֲךָ אֶל־הַמָּקוֹם אֲשֶׁר הֲכִנֹתִי: ג' פעמים

הַמַּלְאָךְ הַגֹּאֵל אֹתִי מִכָּל־רָע יְבָרֵךְ אֶת־הַנְּעָרִים וְיִקָּרֵא בָהֶם שְׁמִי וְשֵׁם אֲבֹתַי אַבְרָהָם וְיִצְחָק וְיִדְגּוּ לָרֹב בְּקֶרֶב הָאָרֶץ: ג' פעמים

היוצא לדרך אפילו רק עד פרסה צריך להתפלל תפלת הדרך, אבל אם הוא צריך ללכת פחות מפרסה מתפלל בלא חתימה, אבל אם הוא מקום סכנה אפילו בפחות מפרסה צריך להתפלל בחתימה: תיכף כשיוצא מעיבורה של עיר צריך להתפלל תפילה זו ואם שכח מתפלל כל זמן שהוא בדרך ובלבד שלא הגיע עדיין תוך פרסה הסמוכה לעירו או לעיר שרוצה ללון בה דמשם ואילך יאמר אותה בלא חתימה: יש להסמיך תפלת הדרך לברכה אחרת כדי שתהא ברכה הסמוכה לחברתה לכן כשיוצא בדרך בשחרית קודם התפילה כגון שהוא נחוץ י"ל ברכות השחר על הדרך ולהסמיך תפלת הדרך לברכת השחר וביום יאכל איזה פרי ויסמכנה לברכת הפרי או כשמטיל מים יסמיך אותה לברכת אשר יצר:

יְהִי רָצוֹן מִלְפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, שֶׁתּוֹלִיכֵנוּ לְשָׁלוֹם וְתַצְעִידֵנוּ לְשָׁלוֹם וְתַדְרִיכֵנוּ לְשָׁלוֹם, וְתִסְמְכֵנִי לְשָׁלוֹם, וְתַגִּיעֵנוּ לִמְחוֹז חֶפְצֵנוּ לְחַיִּים לְשִׂמְחָה וּלְשָׁלוֹם, [אם דעתו לחזור מיד אומר: וְתַחְזִירֵנִי לְשָׁלוֹם] וְתַצִּילֵנוּ מִכַּף כָּל אוֹיֵב וְאוֹרֵב וְלִסְטִים וְחַיּוֹת רָעוֹת בַּדֶּרֶךְ, וּמִכָּל מִינֵי פֻרְעָנִיּוֹת הַמִּתְרַגְּשׁוֹת וּבָאוֹת לָעוֹלָם וְתִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל מַעֲשֵׂה יָדֵינוּ, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל רֹאֵינוּ, וְתִגְמְלֵינוּ חֲסָדִים טוֹבִים וְתִשְׁמַע קוֹל תַּחֲנוּנֵינוּ, כִּי אֵל שֹׁמֵעַ תְּפִלָּה וְתַחֲנוּן אָתָּה: בָּרוּךְ אַתָּה ה', שׁוֹמֵעַ תְּפִלָה:

ואומר ג"פ:

וְיַעֲקֹב הָלַךְ לְדַרְכּוֹ וַיִּפְגְּעוּ בוֹ מַלְאֲכֵי אֱלֹקִים: וַיֹּאמֶר יַעֲקֹב כַּאֲשֶׁר רָאָם, מַחֲנֵה אֱלֹקִים זֶה, וַיִּקְרָא שֵׁם הַמָּקוֹם הַהוּא מַחֲנָיִם:

(ויאמר ל' פעמים:) לִישׁוּעָתְךָ קִוִּיתִי ה', קִוִּיתִי ה' לִישׁוּעָתְךָ, ה' לִישׁוּעָתְךָ קִוִּיתִי:

(ויאמר ו' פעמים:) וִיהִי נוֹעַם אֲ-דֹנָי אֱלֹקֵינוּ עָלֵינוּ, וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנָה עָלֵינוּ, וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנֵהוּ:

(ויאמר י' פעמים:) יְבָרֶכְךָ ה' וְיִשְׁמְרֶךָ: יָאֵר ה' פָּנָיו אֵלֶיךָ, וִיחֻנֶּךָּ: יִשָּׂא ה' פָּנָיו אֵלֶיךָ, וְיָשֵׂם לְךָ שָׁלוֹם:

(ויאמר ה' פעמים:) הַמַּלְאָךְ הַגּוֹאֵל אוֹתִי מִכָּל רָע, יְבָרֵךְ אֶת הַנְּעָרִים, וְיִקָּרֵא בָהֶם שְׁמִי וְשֵׁם אֲבוֹתַי, אַבְרָהָם וְיִצְחָק, וְיִדְגּוּ לָרוֹב בְּקֶרֶב הָאָרֶץ:

(ויאמר ג' פעמים:) הַמַּלְאָךְ הַגּוֹאֵל אוֹתִי מִכָּל רָע, יְבָרֵךְ אֶת הַנְּעָרִים, וְיִקָּרֵא בָהֶם שְׁמִי וְשֵׁם אֲבוֹתַי, אַבְרָהָם וְיִצְחָק, וְיִדְגּוּ לָרוֹב בְּקֶרֶב הָאָרֶץ:

(ויאמר ג' פעמים:) לִישׁוּעָתְךָ קִוִּיתִי ה', קִוִּיתִי ה' לִישׁוּעָתְךָ, ה' לִישׁוּעָתְךָ קִוִּיתִי:

(ויאמר ג"פ:) הִנֵּה אָנֹכִי שֹׁלֵחַ מַלְאָךְ לְפָנֶיךָ לִשְׁמָרְךָ בַּדָּרֶךְ וְלַהֲבִיאֲךָ אֶל הַמָּקוֹם אֲשֶׁר הֲכִינֹתִי:

(ויאמר ז' פעמים:) וִיהִי נוֹעַם אֲ-דֹנָי אֱלֹקֵינוּ עָלֵינוּ, וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנָה עָלֵינוּ, וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנֵהוּ:

(ואומר:) אַתָּה סֵתֶר לִי מִצַּר תִּצְּרֵנִי רָנֵּי פַלֵּט תְּסוֹבְבֵנִי סֶלָה: בִּטְחוּ בַה' עֲדֵי עַד כִּי בְּיָהּ ה' צוּר עוֹלָמִים: ה' עֹז לְעַמּוֹ יִתֵּן. ה' יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם: ה' צְבָאוֹת עִמָּנוּ. מִשְׂגָּב לָנוּ אֱלֹקֵי יַעֲקֹב סֶלָה: ה' צְבָאוֹת אַשְׁרֵי אָדָם בֹּטֵחַ בָּךְ.

(ויאמר ג"פ:) ה' הוֹשִׁיעָה הַמֶּלֶךְ יַעֲנֵנוּ בְיוֹם קָרְאֵנוּ: (ויכוין בר"ת "יב"ק"):

שִׁיר לַמַּעֲלוֹת, אֶשָּׂא עֵינַי אֶל הֶהָרִים, מֵאַיִן יָבֹא עֶזְרִי: עֶזְרִי מֵעִם ה' עֹשֵׂה שָׁמַיִם וָאָרֶץ: אַל יִתֵּן לַמּוֹט רַגְלֶךָ. אַל יָנוּם שֹׁמְרֶךָ: הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן, שׁוֹמֵר יִשְׂרָאֵל: ה' שֹׁמְרֶךָ. ה' צִלְּךָ עַל יַד יְמִינֶךָ: יוֹמָם הַשֶּׁמֶשׁ לֹא יַכֶּכָּה. וְיָרֵחַ בַּלָּיְלָה: ה' יִשְׁמָרְךָ מִכָּל רָע, יִשְׁמֹר אֶת נַפְשֶׁךָ: ה' יִשְׁמָר צֵאתְךָ וּבוֹאֶךָ, מֵעַתָּה וְעַד עוֹלָם:

(ואומר:) מִגְדַּל עֹז שֵׁם ה' בּוֹ יָרוּץ צַדִּיק וְנִשְׂגָּב: בְּשֵׁם ה' אֱלֹקֵי יִשְׂרָאֵל. מִימִינִי מִיכָאֵל. וּמִשְּׂמֹאלִי גַבְרִיאֵל, וּמִלְּפָנַי אוֹרִיאֵל, וּמֵאֲחוֹרַי רְפָאֵל, וְעַל רֹאשִׁי שְׁכִינַת אֵל:

בטיסה: מוסיפים באווירון

באוירון מוסיפים תפילה זו:

וְיְהִי רָצוֹן מִלְפָנֶיךָ שֶׁתִּשְׁמְרֵנוּ מֵרוּחַ סוֹעָה וְסַעַר וּמִצַּעַר. וְתוֹצִיא רוּחַ טוֹב מֵאוֹצְרוֹתֶיךָ לְהַנְהִיג אֶת סְפִינַת הָאֲוִיר, וְחַזֵּק כָּל מַנְהִיגֶיהָ שֶׁיִּנְהֲגוּ אוֹתָהּ כָּרָאוּי וְהַגִּיעֵנוּ אֶל מְחוֹז חֶפְצֵנוּ לְחַיִּים וּלְשָׁלוֹם בְּלִי שׁוּם פַּחַד. שָׁמְרָה נַפְשִׁי וְהַצִּילֵנִי, אַל אֵבוֹשׁ כִּי חָסִיתִי בָּךְ, וַאֲנַחְנוּ נְבָרֵךְ יָהּ מֵעַתָּה וְעַד עוֹלָם הַלְלוּיָהּ:

מוֹדֶה אֲנִי לְפָנֶיךָ ה' אֱלֹקַי וֵאלֹקֵי אֲבוֹתַי, שֶׁהוֹצֵאתָנִי מִכְּרַךְ זֶה לְשָׁלוֹם, וּכְשֵׁם שֶׁהוֹצֵאתָנִי לְשָׁלוֹם, כֵּן תּוֹלִיכֵנִי לְשָׁלוֹם, וְתִסְמְכֵנִי לְשָׁלוֹם, וְתַצְעִידֵנוּ לְשָׁלוֹם, וְתַצִּילֵנוּ מִכַּף כָּל אוֹיֵב וְאוֹרֵב וְלִסְטִים וְחַיּוֹת רָעוֹת בַּדֶּרֶךְ, וּמִכָּל מִינֵי פֻרְעָנִיּוֹת הַמִּתְרַגְּשׁוֹת וּבָאוֹת לָעוֹלָם וְתִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל מַעֲשֹה יָדֵינוּ, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל רֹאֵינוּ, וְתִגְמְלֵינוּ חֲסָדִים טוֹבִים וְתִשְׁמַע קוֹל תְּפִלָּתֵנוּ. כִּי אַתָּה שׁוֹמֵעַ תְּפִלַּת כָּל פֶּה: בָּרוּךְ אַתָּה ה', שׁוֹמֵעַ תְּפִלָה:

וְשֶׁל אֱלִישָׁע הַנָּבִיא נְבִיאֶיךָ נְבִיאֵי אֶמֶת, וּלְּמַעַן זְכוּת כָּל הַצַּדִּיקִים וְהַחֲסִידִים שֶׁהָיוּ בְּכָל דּוֹר מְגִנִּים עַל יִשְׂרָאֵל, תַּעֲמִיד הַיָּם (הַנָּהָר) מִזַּעְפּוֹ, וְיֶחֱשׁוּ גַלָּיו, וְתַעֲשֶׂה רְצוֹן לִבֵּנוּ וְתוֹלִיכֵנוּ מְהֵרָה אֶל מְחוֹז חֶפְצֵנוּ לְטוֹבָה, כַּאֲשֶׁר בְּיָדְךָ לַעֲשׂוֹת, וְתִשְׁמַע תְּפִלָּתֵנוּ וְתַאֲזִין תְּחִנָּתֵנוּ וַעֲנֵנוּ בְּשָׁעָה הַזֹּאת שֶׁאָנוּ מִתְפַּלְּלִים וּמִתְחַנְּנִים לְפָנֶיךָ, וְתוֹלִיכֵנוּ בְּנַחַת רוּחַ טוֹב כְּפִי הַצֹּרֶךְ, וְתִשְׁמְרֵנוּ מֵהֲמוֹן גַּלִּים וּמִשְּׁאוֹן מִשְׁבְּרֵי יָם, מֵרוּחַ סֹעָה מִסַּעַר וּמִצַּעַר וּמִבִּלְבּוּל עִתִּים וּמֵחִלּוּף זְמַנִּים וּמֵהַבְּרִיּוֹת הָרָעוֹת הַגְּדֵלוֹת בַּיָּם וּבַיַּבָּשָׁה, וּמִכָּל צָרָה וּמִכָּל הֶפְסֵד. וְאַל יְעַכְּבֵנוּ לֹא מַשְֹׁטִין וְלֹא מֵצַּר וְלֹא מַזִּיק וְלֹא מְבַלְבֵּל וְלֹא מַפְחִיד וְלֹא מְאַיֵּם. וְתוֹצִיא רוּחַ טוֹב מֵאוֹצְרוֹתֶיךָ לְהַנְהִיג הַסְּפִינָה, וְחַזֵּק כָּל תּוֹפְשֵׂי מָשׁוֹט וּמַנְהִיגֶיהָ וּמַלָּחֶיהָ וְחוֹבְלֶיהָ וְאָמָנֶיהָ שֶׁיִּנְהֲגוּ אוֹתָהּ כָּרָאוּי, בְּלֹא עַצְלוּת וּבְלֹא חִשּׁוּל וּבְלֹא תִשּׁוּת כֹּחַ, וְהַגִּיעֵנוּ אֶל מְחוֹז חֶפְצֵנוּ בְּלִי שׁוּם הֶזֵּק וּבְלִי שׁוּם צַעַר וּבְלִי שׁוּם פַּחַד, כִּי אַתָּה מוֹשֵׁל בְּגֵאוּת הַיָּם בְּשׂוֹא גַלָּיו אַתָּה תְשַׁבְּחֵם. הָקֵם סְעָרָה לִדְמָמָה וְיֶחֱשׁוּ גַלָּיו. שָׁמְרָה נַפְשִׁי וְהַצִּילֵנִי, אַל אֵבוֹשׁ כִּי חָסִיתִי בָךְ. וַאֲנַחְנוּ נְבָרֵךְ יָהּ מֵעַתָּה וְעַד עוֹלָם, הַלְלוּיָהּ:

וממשיך יהי רצון כ"סדר תפילת הדרך":

יְהִי רָצוֹן מִלְפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, שֶׁתּוֹלִיכֵנוּ לְשָׁלוֹם וְתַצְעִידֵנוּ לְשָׁלוֹם וְתַדְרִיכֵנוּ לְשָׁלוֹם, וְתִסְמְכֵנוּ לְשָׁלוֹם, וְתַגִּיעֵנוּ לִמְחוֹז חֶפְצֵנוּ לְחַיִּים וּלְשִׂמְחָה וּלְשָׁלוֹם. (אם דעתו לחזור מיד אומר וְתַחְזִירֵנוּ לְשָׁלוֹם) וְתַצִּילֵנוּ מִכַּף כָּל אוֹיֵב וְאוֹרֵב וְלִסְטִים וְחַיּוֹת רָעוֹת בַּדֶּרֶךְ, וּמִכָּל מִינֵי פֻּרְעָנֻיּוֹת הַמִּתְרַגְּשׁוֹת לָבוֹא לָעוֹלָם, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל רֹאֵינוּ, כִּי אל שׁוֹמֵעַ תְּפִלָּה וְתַחֲנוּן אַתָּה. בָּרוּךְ אַתָּה ה' שׁוֹמֵעַ תְּפִלָּה:

כִּי מַלְאָכָיו יְצַוֶּה לָךְ לִשְׁמָרְךָ בְּכָל דְרָכֶיךָ: ה' יִשְׁמָר צֵאתְךָ וּבוֹאֶךָ מֵעַתָּה וְעַד עוֹלָם:

וְיַעֲקֹב הָלַךְ לְדַרְכּוֹ וַיִּפְגְּעוּ בוֹ מַלְאֲכֵי אֱלֹקִים. וַיֹּאמֶר יַעֲקֹב כַּאֲשֶׁר רָאָם ר"ת רפאל אוריאל מיכאל: מַחֲנֵה אֱלֹקִים זֶה, וַיִּקְרָא שֵׁם הַמָּקוֹם הַהוּא מַחֲנָיִם.

ברכת הגומל

מי שחזר בשלום מדרך מסוכנת, החלים ממחלה או ניצל מסכנה – מברך בפני עשרה

ארבעה צריכין להודות שהם: חבוש, יסורין, ים, מדבר וסימנם: "וכל ה'חיים' יודוך סלה"

צריך לברך בפני עשרה ותרי מנייהו דתנו הלכתא, וטוב להזהר שיהיו י"ב עם המברך וגם בהדי תרי רבנן הנזכרים, ואם לא נזדמן לו תרי דתנו הלכתא אפילו הכי יברך ואין לבטל הברכה בעבור זה, ונהגו לברך אחר קריאת התורה בציבור שעל הרוב נמצא שם תרי דתנו הלכתא (בא"ח עקב ה"ב)

לפני הברכה אומר

אוֹדֶה ה' בְּכָל־לֵבָב בְּסוֹד יְשָׁרִים וְעֵדָה:

ומברך

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, הַגּוֹמֵל לְחַיָּבִים טוֹבוֹת, שֶׁגְּמָלַנִי כָּל־טוּב:

והקהל עונים

אָמֵן. הָאֵל שֶׁגְּמָלְךָ כָּל־טוּב, הוּא יִגְמָלְךָ כָּל־טוּב סֶלָה:

ועונה המברך

אָמֵן כֵּן יְהִי רָצוֹן:

ארבעה צריכים להודות יורדי הים כשעלו ממנה והולכי מדברות כשיגיעו ליישוב ומי שהיה חולה ונתרפא ומי שהיה חבוש בבית האסורים ויצא. ואומרה על הס"ת לאחר ברכה אחרונה. אם איחר יש לו תשלומין לברך כל זמן שירצה ונכון שלא לאחר שלשה ימים (שו"ע או"ח ריט).

בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם. הַגּוֹמֵל לְחַיָּבִים טוֹבוֹת. שֶׁגְּמָלַנִי כָּל טוֹב:

הקהל עונה אמן. ואומרים: מִי שֶׁגְּמָלְךָ טוֹב. הוּא יִגְמָלְךָ כָּל טוֹב סֶלָה:

בסידור אשכנז שבספריא אין את ברכת הגומל כקטע נפרד, ולכן בנוסח אשכנז מוצג כאן הנוסח מסידור ספרד.

המילים והניקוד מתוך ספריא, מילה במילה: עדות המזרח – סידור עדות המזרח, מהדורת שלייסבו (CC0); ספרד – סידור ספרד, "תורת אמת" (נחלת הכלל); אשכנז – סידור אשכנז, מהדורת "דעת" (נחלת הכלל). טעמי המקרא הורדו, והשמות הקדושים כתובים בכינוי (ה', אלוקינו) ואומרים אותם כרגיל. בדיקה בקוד מוודאת שהטקסט זהה למקור. מי שנוהג אחרת, הולך לפי המנהג שלו; ובכל שאלה, שואלים רב.

לחיילים ולמי שנוסע הרבה: דף לחייל בגודל כיס, עם תפילת הדרך, ברכת הגומל וקריאת שמע על המיטה, להדפסה או כתמונה לווטסאפ.

בתפריט הדפדפן: "הוספה למסך הבית", והדף נפתח כמו אפליקציה, גם בלי קליטה בדרך.

לשלוח למי שיוצא לדרך

תפילת הדרך בכתב גדול וברור, תמיד בכיס

תפילת הדרך היא בקשה קצרה שאומרים כשיוצאים לדרך: שנגיע לשלום, שנינצל מכל רע בדרך, ושנחזור לשלום. כאן היא כתובה במלואה, בניקוד מלא ובאות גדולה, כך שאפשר לקרוא אותה בעשר שניות לפני שיוצאים מהעיר, גם בלי סידור ביד.

תפילת הדרך נוסח עדות המזרח, ספרד ואשכנז

בוחרים נוסח פעם אחת למעלה, והדף זוכר אותו. בנוסח עדות המזרח מופיעים גם הפסוקים שיש נוהגים להוסיף לשמירה. בנוסח ספרד מופיעים הפסוקים שאומרים כמה פעמים, תפילה מיוחדת לטיסה באווירון, ותפילות קצרות לכניסה לעיר וליציאה ממנה. בנוסח אשכנז הנוסח הקצר, עם הפסוקים ויעקב הלך לדרכו וברכת כהנים.

מתי אומרים, ומאיזה מרחק

בשאלות ותשובות שלמטה מובאים דברי קיצור שולחן ערוך (סימן סח) ושולחן ערוך (סימן קי) כפי שהם כתובים, עם מראה מקום: מתי אומרים אחרי שיוצאים מהעיר, מאיזה מרחק, מה עושה מי ששכח, ואם אומרים כל יום מחדש. אנחנו לא פוסקים הלכה: מביאים את לשון המקור, ובכל שאלה למעשה שואלים רב.

ברכת הגומל

מי שחזר בשלום מטיסה לחו"ל או מדרך מסוכנת, החלים ממחלה או ניצל מסכנה, מברך ברכת הגומל בפני עשרה, ובדרך כלל אחרי קריאת התורה. הנוסח ותשובת הקהל נמצאים בדף, מתחת לתפילת הדרך.

עובד גם בלי קליטה

אחרי הכניסה הראשונה הדף נשמר בטלפון, ונפתח גם בכביש בלי קליטה, בשטח או בטיסה. יש אות גדולה, מצב לילה ומסך שלא נכבה. לחיילים יש דף בגודל כיס להדפסה, ולמי שרוצה את כל ברכות הבוקר: ברכות השחר ותפילין. עוד בכלי קודש: ברכת המזון וספר תהילים המלא.

שאלות ותשובות

מתי אומרים תפילת הדרך?

בקיצור שולחן ערוך: לְאַחַר שֶׁיָּצָא מֵעִבּוּרָהּ שֶׁל הָעִיר, דְּהַיְנוּ שִׁבְעִים אַמָּה וּשְׁנֵי שְׁלִישֵׁי אַמָּה לְאַחַר שֶׁכָּלוּ כָּל הַבָּתִים, מִתְפַּלֵּל תְּפִלַּת הַדֶּרֶךְ (קיצור שולחן ערוך סח, א).

מאיזה מרחק אומרים תפילת הדרך?

אֵין לְהִתְפַּלֵּל אוֹתָהּ אֶלָּא אִם כֵּן יֵשׁ לוֹ לָלֶכֶת לְכָל הַפָּחוֹת פַּרְסָה (קיצור שולחן ערוך סח, ב). ובסידור עדות המזרח: לפחות מהלך שיעור פרסה (72 דקות) אומר תפלת הדרך. ומי ששכח: וְאִם שָׁכַח, יָכוֹל לְהִתְפַּלֵּל כָּל זְמַן שֶׁהוּא בַדֶּרֶךְ (שם).

אומרים תפילת הדרך כל פעם שיוצאים, או פעם אחת ביום?

אֵין אוֹמְרִים אוֹתָהּ אֶלָּא פַּעַם אַחַת בְּכָל יוֹם שֶׁהוּא הוֹלֵךְ (קיצור שולחן ערוך סח, ה). שם גם מפורט מתי אומרים אותה פעם שנייה.

אפשר לומר תפילת הדרך בנסיעה?

יֵשׁ לְאָמְרָהּ בַּעֲמִידָה, וְאִם הוּא רוֹכֵב אוֹ הוֹלֵךְ בַּעֲגָלָה, אִם אֶפְשָׁר לְהַעֲמִיד הַבְּהֵמָה, יַעֲמִידֶנָּהּ, מִשּׁוּם דְּרוֹכֵב כִּמְהַלֵּךְ דָּמִי. וְאִם לָאו, יֹאמְרָהּ כָּךְ (קיצור שולחן ערוך סח, ד). בכל שאלה למעשה, שואלים רב.

יש תפילה מיוחדת לטיסה?

בסידור ספרד ("תורת אמת") יש תפילה שבאוירון מוסיפים. היא מופיעה בדף בנוסח ספרד, מיד אחרי תפילת הדרך.

מי מברך ברכת הגומל?

ארבעה צריכים להודות יורדי הים כשעלו ממנה והולכי מדברות כשיגיעו ליישוב ומי שהיה חולה ונתרפא ומי שהיה חבוש בבית האסורים ויצא (שולחן ערוך, אורח חיים ריט, א). נוסח הברכה ותשובת הקהל נמצאים בדף.
