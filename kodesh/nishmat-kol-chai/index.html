נשמת כל חי

התפילה עצמה, מ"נשמת כל חי" ועד "וכל קרבי את שם קדשו"

כשאומר נשמת כל חי יכון לקבל תוספת נפש יתירה דבחינת היום

נִשְׁמַת כָּל־חַי תְּבָרֵךְ אֶת שִׁמְךָ ה' אֱלֹקֵינוּ, וְרוּחַ כָּל־בָּשָׂר תְּפָאֵר וּתְרוֹמֵם זִכְרְךָ מַלְכֵּנוּ תָּמִיד, מִן־הָעוֹלָם וְעַד־הָעוֹלָם אַתָּה אֵל. וּמִבַּלְעָדֶיךָ אֵין לָנוּ מֶלֶךְ גּוֹאֵל וּמוֹשִׁיעַ, פּוֹדֶה וּמַצִּיל, וְעוֹנֶה וּמְרַחֵם, בְּכָל־עֵת צָרָה וְצוּקָה, אֵין לָנוּ מֶלֶךְ עוֹזֵר וְסוֹמֵךְ אֶלָּא אָתָּה:

אֱלֹקֵי הָרִאשׁוֹנִים וְהָאַחֲרוֹנִים, אֱלוֹהַּ כָּל־בְּרִיּוֹת, אֲדוֹן כָּל־תּוֹלָדוֹת, הַמְּהֻלָּל בְּכָל־הַתִּשְׁבָּחוֹת, הַמְּנַהֵג עוֹלָמוֹ בְּחֶסֶד, וּבְרִיּוֹתָיו בְּרַחֲמִים. וַה' אֱלֹקִים אֱמֶת, לֹא יָנוּם וְלֹא יִישָׁן, הַמְעוֹרֵר יְשֵׁנִים וְהַמֵּקִיץ נִרְדָּמִים, מְחַיֶּה מֵתִים וְרוֹפֵא חוֹלִים, פּוֹקֵחַ עִוְרִים וְזוֹקֵף כְּפוּפִים, הַמֵּשִׂיחַ אִלְּמִים וְהַמְפַעֲנֵחַ נֶעֱלָמִים, וּלְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים:

וְאִלּוּ פִינוּ מָלֵא שִׁירָה כַיָּם, וּלְשׁוֹנֵנוּ רִנָּה כַּהֲמוֹן גַּלָּיו, וְשִׂפְתוֹתֵינוּ שֶׁבַח כְּמֶרְחֲבֵי רָקִיעַ, וְעֵינֵינוּ מְאִירוֹת כַּשֶּׁמֶשׁ וְכַיָּרֵחַ, וְיָדֵינוּ פְרוּשׂוֹת כְּנִשְׁרֵי שָׁמָיִם, וְרַגְלֵינוּ קַלּוֹת כָּאַיָּלוֹת, אֵין אֲנַחְנוּ מַסְפִּיקִין לְהוֹדוֹת לְךָ ה' אֱלֹקֵינוּ, וּלְבָרֵךְ אֶת־שִׁמְךָ מַלְכֵּנוּ, עַל־אַחַת מֵאֶלֶף אַלְפֵי אֲלָפִים, וְרוֹב רִבֵּי רְבָבוֹת פְּעָמִים, הַטּוֹבוֹת נִסִּים וְנִפְלָאוֹת שֶׁעָשִׂיתָ עִמָּנוּ וְעִם אֲבוֹתֵינוּ. מִלְּפָנִים מִמִּצְרַיִם גְּאַלְתָּנוּ ה' אֱלֹקֵינוּ, מִבֵּית עֲבָדִים פְּדִיתָנוּ, בְּרָעָב זַנְתָּנוּ וּבְשָׂבָע כִּלְכַּלְתָּנוּ, מֵחֶרֶב הִצַּלְתָּנוּ, מִדֶּבֶר מִלַּטְתָּנוּ, וּמֵחֳלָאִים רָעִים וְרַבִּים דִּלִּיתָנוּ. עַד הֵנָּה עֲזָרוּנוּ רַחֲמֶיךָ, וְלֹא עֲזָבוּנוּ חֲסָדֶיךָ. עַל כֵּן אֵבָרִים שֶׁפִּלַּגְתָּ בָּנוּ, וְרוּחַ וּנְשָׁמָה שֶׁנָּפַחְתָּ בְּאַפֵּינוּ, וְלָשׁוֹן אֲשֶׁר שַׂמְתָּ בְּפִינוּ, הֵן הֵם יוֹדוּ וִיבָרְכוּ וִישַׁבְּחוּ וִיפָאֲרוּ וִישׁוֹרְרוּ אֶת־שִׁמְךָ מַלְכֵּנוּ תָּמִיד. כִּי כָל־פֶּה לְךָ יוֹדֶה, וְכָל־לָשׁוֹן לְךָ תְשַׁבֵּחַ, וְכָל־עַיִן לְךָ תְצַפֶּה, וְכָל־בֶּרֶךְ לְךָ תִכְרַע, וְכָל־קוֹמָה לְפָנֶיךָ תִשְׁתַּחֲוֶה, וְהַלְּבָבוֹת יִירָאוּךָ, וְהַקֶּרֶב וְהַכְּלָיוֹת יְזַמְּרוּ לִשְׁמֶךָ, כַּדָּבָר שֶׁנֶּאֱמַר כָּל עַצְמוֹתַי תֹּאמַרְנָה ה' מִי כָמוֹךָ מַצִּיל עָנִי מֵחָזָק מִמֶּנּוּ וְעָנִי וְאֶבְיוֹן מִגֹּזְלוֹ:

שַׁוְעַת עֲנִיִּים אַתָּה תִשְׁמַע, צַעֲקַת הַדַּל תַּקְשִׁיב וְתוֹשִׁיעַ. וְכָתוּב רַנְּנוּ צַדִּיקִים בַּה' לַיְשָׁרִים נָאוָה תְהִלָּה:

תקנו לומר נשמת בשבת על שם נשמה יתירה הניתנת לאדם בשבת. יאמר כל השבח בשמחה עצומה. ומגודל התשוקה יקבל על עצמו מסירת נפש:

נִשְׁמַת כָּל חַי תְּבָרֵךְ אֶת שִׁמְךָ ה' אֱלֹקֵינוּ. וְרוּחַ כָּל בָּשָׂר תְּפָאֵר וּתְרוֹמֵם זִכְרְךָ מַלְכֵּנוּ תָּמִיד: מִן הָעוֹלָם וְעַד הָעוֹלָם אַתָּה אֵל. וּמִבַּלְעָדֶיךָ אֵין לָנוּ מֶלֶךְ גּוֹאֵל וּמוֹשִׁיעַ. פּוֹדֶה וּמַצִּיל וּמְפַרְנֵס וְעוֹנֶה וּמְרַחֵם בְּכָל עֵת צָרָה וְצוּקָה. אֵין לָנוּ מֶלֶךְ עוֹזֵר וְסוֹמֵךְ אֶלָּא אָתָּה: אֱלֹקֵי הָרִאשׁוֹנִים וְהָאַחֲרוֹנִים. אֱלוֹהַּ כָּל בְּרִיוֹת. אֲדוֹן כָּל תּוֹלָדוֹת. הַמְּהֻלָּל בְּכָל (בְּרוֹב) הַתִּשְׁבָּחוֹת. הַמְּנַהֵג עוֹלָמוֹ בְּחֶסֶד וּבְרִיּוֹתָיו בְּרַחֲמִים: וַה' עֵר. הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן. הַמְעוֹרֵר יְשֵׁנִים. וְהַמֵּקִיץ נִרְדָּמִים. (מְחַיֵה מֵתִים וְרוֹפֵא חוֹלִים פּוֹקֵחַ עִוְרִים וְזוֹקֵף כְּפוּפִים) וְהַמֵּשִׂיחַ אִלְמִים. וְהַמַּתִּיר אֲסוּרִים. וְהַסּוֹמֵךְ נוֹפְלִים. וְהַזּוֹקֵף כְּפוּפִים. וְהַמְּפַעֲנֵחַ נֶעֱלָמִים. וּלְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים: וְאִלּוּ פִינוּ מָלֵא שִׁירָה כַּיָּם. וּלְשׁוֹנֵנוּ רִנָּה כַּהֲמוֹן גַּלָּיו. וְשִׂפְתוֹתֵינוּ שֶׁבַח כְּמֶרְחֲבֵי רָקִיעַ. וְעֵינֵינוּ מְאִירוֹת כַּשֶּׁמֶשׁ וְכַיָּרֵחַ. וְיָדֵינוּ פְרוּשׂוֹת כְּנִשְׁרֵי שָׁמָיִם. וְרַגְלֵינוּ קַלּוֹת כָּאַיָּלוֹת. אֵין אֲנַחְנוּ מַסְפִּיקִים לְהוֹדוֹת לְךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ. וּלְבָרֵךְ אֶת שִׁמְךָ מַלְכֵּנוּ. עַל אַחַת מֵאָלֶף אֶלֶף אַלְפֵי אֲלָפִים וְרִבֵּי רְבָבוֹת פְּעָמִים. הַטּוֹבוֹת נִסִּים וְנִפְלָאוֹת שֶׁעָשִׂיתָ עִם אֲבוֹתֵינוּ וְעִמָּנוּ: מִלְּפָנִים מִמִּצְרַיִם גְּאַלְתָּנוּ ה' אֱלֹקֵינוּ. וּמִבֵּית עֲבָדִים פְּדִיתָנוּ. בְּרָעָב זַנְתָּנוּ. וּבְשָׂבָע כִּלְכַּלְתָּנוּ. מֵחֶרֶב הִצַּלְתָּנוּ. וּמִדֶּבֶר מִלַּטְתָּנוּ. וּמֵחֳלָיִם רָעִים וְרַבִּים וְנֶאֱמָנִים דִּלִּיתָנוּ: עַד הֵנָּה עֲזָרוּנוּ רַחֲמֶיךָ. וְלֹא עֲזָבוּנוּ חֲסָדֶיךָ ה' אֱלֹקֵינוּ. וְאַל תִּטְּשֵׁנוּ ה' אֱלֹקֵינוּ לָנֶצַח: עַל כֵּן אֵבָרִים שֶׁפִּלַּגְתָּ בָּנוּ. וְרוּחַ וּנְשָׁמָה שֶׁנָּפַחְתָּ בְּאַפֵּינוּ. וְלָשׁוֹן אֲשֶׁר שַׂמְתָּ בְּפִינוּ. הֵן הֵם. יוֹדוּ וִיבָרְכוּ וִישַׁבְּחוּ וִיפָאֲרוּ וִישׁוֹרְרוּ וִירוֹמְמוּ וְיַעֲרִיצוּ וְיַקְדִּישׁוּ וְיַמְלִיכוּ אֶת שִׁמְךָ מַלְכֵּנוּ תָּמִיד: כִּי כָל פֶּה לְךָ יוֹדֶה. וְכָל לָשׁוֹן לְךָ תִשָּׁבַע. וְכָל עַיִן לְךָ תְּצַפֶּה. וְכָל בֶּרֶךְ לְךָ תִכְרַע. וְכָל קוֹמָה לְפָנֶיךָ תִשְׁתַּחֲוֶה. וְכָל הַלְּבָבוֹת יִירָאוּךָ. וְכָל קֶרֶב וּכְלָיוֹת יְזַמְּרוּ לִשְׁמֶךָ. כַּדָּבָר שֶׁכָּתוּב. כָּל עַצְמוֹתַי תֹּאמַרְנָה ה' מִי כָמוֹךָ. מַצִּיל עָנִי מֵחָזָק מִמֶּנּוּ. וְעָנִי וְאֶבְיוֹן מִגּוֹזְלוֹ. שַׁוְעַת עֲנִיִּים אַתָּה תִשְׁמַע. צַעֲקַת הַדַּל תַּקְשִׁיב וְתוֹשִׁיעַ: מִי יִדְמֶה לָּךְ. וּמִי יִשְׁוֶה לָּךְ. וּמִי יַעֲרָךְ לָךְ. הָאֵל הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא אֵל עֶלְיוֹן. קוֹנֵה שָׁמַיִם וָאָרֶץ: נְהַלֶּלְךָ וּנְשַׁבֵּחֲךָ וּנְפָאֶרְךָ וּנְבָרֵךְ אֶת שֵׁם קָדְשֶׁךָ. כָּאָמוּר. לְדָוִד. בָּרְכִי נַפְשִׁי אֶת ה'. וְכָל קְרָבַי אֶת שֵׁם קָדְשׁוֹ:

נִשְׁמַת כָּל חַי תְּבָרֵךְ אֶת שִׁמְךָ ה' אֱלקֵינוּ. וְרוּחַ כָּל בָּשר תְּפָאֵר וּתְרומֵם זִכְרְךָ מַלְכֵּנוּ תָּמִיד: מִן הָעולָם וְעַד הָעולָם אַתָּה אֵל. וּמִבַּלְעָדֶיךָ אֵין לָנוּ מֶלֶךְ גּואֵל וּמושִׁיעַ. פּודֶה וּמַצִּיל וּמְפַרְנֵס וּמְרַחֵם בְּכָל עֵת צָרָה וְצוּקָה. אֵין לָנוּ מֶלֶךְ עוזֵר וְסומֵךְ אֶלָּא אָתָּה: אֱלקֵי הָרִאשׁונִים וְהָאַחֲרונִים. אֱלוהַּ כָּל בְּרִיות. אֲדון כָּל תּולָדות. הַמְּהֻלָּל בְּרוב הַתִּשְׁבָּחות. הַמְּנַהֵג עולָמו בְּחֶסֶד וּבְרִיּותָיו בְּרַחֲמִים: וה' לא יָנוּם וְלא יִישָׁן. הַמְעורֵר יְשֵׁנִים. וְהַמֵּקִיץ נִרְדָּמִים. וְהַמֵּשיחַ אִלְמִים. ישתבח

הברכה שחותמת את פסוקי דזמרה

שבח ברכת ישתבח הוא עצום ונורא, וצריך לאומרם בנעימה כמסדר שבחו של מלך, לכך לא יאמר אותם בנשימה אחת כמנהג קצת בני אדם, ומצאתי כתוב, שימנה אותם באצבעותיו, והוא דבר נכון ויציב, וכך אני נוהג (בא"ח ויגש הט"ו)

יִשְׁתַּבַּח שִׁמְךָ לָעַד מַלְכֵּנוּ, הָאֵל הַמֶּלֶךְ הַגָּדוֹל וְהַקָּדוֹשׁ, בַּשָּׁמַיִם וּבָאָרֶץ. כִּי לְךָ נָאֶה ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ לְעוֹלָם וָעֶד. א שִׁיר, ב וּשְׁבָחָה, ג הַלֵּל, ד וְזִמְרָה, ה עֹז, ו וּמֶמְשָׁלָה, ז נֶצַח, ח גְּדֻלָּה, ט גְּבוּרָה, י תְּהִלָּה, יא וְתִפְאֶרֶת, יב קְדֻשָּׁה, יג וּמַלְכוּת. בְּרָכוֹת וְהוֹדָאוֹת, לְשִׁמְךָ הַגָּדוֹל וְהַקָּדוֹשׁ, וּמֵעוֹלָם וְעַד עוֹלָם אַתָּה אֵל. בָּרוּךְ אַתָּה ה', מֶלֶךְ גָּדוֹל וּמְהֻלָּל בַּתִּשְׁבָּחוֹת, אֵל הַהוֹדָאוֹת, אֲדוֹן הַנִּפְלָאוֹת, בּוֹרֵא כָּל־הַנְּשָׁמוֹת, רִבּוֹן כָּל־הַמַּעֲשִׂים, הַבּוֹחֵר בְּשִׁירֵי זִמְרָה, מֶלֶךְ אֵל חַי הָעוֹלָמִים, אָמֵן:

וּבְכֵן יִשְׁתַּבַּח שִׁמְךָ לָעַד מַלְכֵּנוּ. הָאֵל הַמֶּלֶךְ הַגָּדוֹל וְהַקָּדוֹשׁ בַּשָּׁמַיִם וּבָאָרֶץ. כִּי לְךָ נָאֶה ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ: שִׁיר וּשְׁבָחָה. הַלֵּל וְזִמְרָה. עוֹז וּמֶמְשָׁלָה. נֶצַח גְּדֻלָּה וּגְבוּרָה. תְּהִלָּה וְתִפְאֶרֶת. קְדֻשָּׁה וּמַלְכוּת: בְּרָכוֹת וְהוֹדָאוֹת לְשִׁמְךָ הַגָּדוֹל וְהַקָּדוֹשׁ. וּמֵעוֹלָם וְעַד עוֹלָם אַתָּה אֵל: בָּרוּךְ אַתָּה ה'. אֵל מֶלֶךְ גָּדוֹל וּמְהֻלָּל בַּתִּשְׁבָּחוֹת. אֵל הַהוֹדָאוֹת. אֲדוֹן הַנִּפְלָאוֹת. בּוֹרֵא כָּל הַנְּשָׁמוֹת. רִבּוֹן כָּל הַמַּעֲשִׂים. הַבּוֹחֵר בְּשִׁירֵי זִמְרָה. מֶלֶךְ יָחִיד אֵל חֵי הָעוֹלָמִים:

יִשְׁתַּבַּח שִׁמְךָ לָעַד מַלְכֵּנוּ. הָאֵל הַמֶּלֶךְ הַגָּדול וְהַקָּדושׁ בַּשָּׁמַיִם וּבָאָרֶץ. כִּי לְךָ נָאֶה ה' אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ שִׁיר וּשְׁבָחָה. הַלֵּל וְזִמְרָה. עוז וּמֶמְשָׁלָה. נֶצַח גְּדֻלָּה וּגְבוּרָה. תְּהִלָּה וְתִפְאֶרֶת. קְדֻשָּׁה וּמַלְכוּת: בְּרָכות וְהודָאות מֵעַתָּה וְעַד עולָם: בָּרוּךְ אַתָּה ה'. אֵל מֶלֶךְ גָּדול בַּתִּשְׁבָּחות אֵל הַהודָאות. אֲדון הַנִּפְלָאות. הַבּוחֵר בְּשִׁירֵי זִמְרָה. מֶלֶךְ אֵל חֵי הָעולָמִים:

לשלוח לחבר בווטסאפ

נשמת כל חי, התפילה המלאה בניקוד

"נשמת כל חי" היא אחת מתפילות השבח וההודיה היפות שבסידור. היא פותחת במילים "נשמת כל חי תברך את שמך" וכולה הודאה על הטוב: "אילו פינו מלא שירה כים... אין אנחנו מספיקים להודות לך". כאן היא כתובה במלואה, בניקוד מלא ובאותיות גדולות וקריאות, בנוסח עדות המזרח, בנוסח ספרד ובנוסח אשכנז.

מתי אומרים נשמת כל חי

מקומה הקבוע הוא בתפילת שחרית של שבת ויום טוב, בסוף פסוקי דזמרה, וגם בסוף ההגדה בליל הסדר. מעבר לזה, בקהילות רבות נוהגים לומר אותה כהודיה אחרי ישועה: אחרי לידה, רפואה, שחרור מהצבא או כל בשורה טובה. יש שמתאספים עשרה ואומרים אותה יחד.

נשמת כל חי נוסח עדות המזרח, ספרד ואשכנז

בוחרים נוסח בכפתורים שבראש הדף, והדף זוכר את הבחירה לפעם הבאה. ההבדלים בין הנוסחים קטנים, כמה מילים וסדר השורות בסוף, אבל כל אחד רגיל לנוסח של בית אבא, ולכן הבאנו את שלושתם כמו שהם בסידורים.

איך קוראים בנוחות

בכפתורים א+ וא- מגדילים ומקטינים את האות, והירח מעביר לרקע כהה. המסך נשאר דלוק בזמן הקריאה (בדפדפנים שתומכים בזה), כך שלא צריך לגעת בטלפון באמצע התפילה. ההוראות כתובות באות קטנה ובצבע אחר. אין פרסומות, אין חלונות קופצים ואין הרשמה.

מה יש בדף

שלושה חלקים, כל אחד תחת כותרת: התפילה עצמה עד "וכל קרבי את שם קדשו"; ההמשך שאומרים בשבת, "שוכן עד" ו"ובמקהלות"; וברכת "ישתבח" שחותמת את פסוקי דזמרה. מי שאומר את התפילה כהודיה יכול לעצור בסוף החלק הראשון, או להמשיך, לפי מנהגו.

מאיפה הטקסט

המילים והניקוד נלקחו מספריא, מילה במילה. לא השלמנו שום מילה מהזיכרון. בתצוגה בלבד הורדנו את טעמי המקרא וכתבנו את שמות הקודש בכינוי. לתפילת השבת כולה ראו גם את תפילת שחרית של יום חול, ואת ברכת המזון לסעודת ההודיה.

כלים קרובים

תפילת שחרית, תפילת מנחה, תפילת ערבית, מוסף ראש חודש, מוסף חול המועד, נשמת כל חי, ברכת המזון, תפילת הדרך, קריאת שמע על המיטה, ספר תהילים המלא, זמני היום, לוח שנה עברי ומפת האתר.

שאלות ותשובות

מתי אומרים נשמת כל חי?

בתפילת שחרית של שבת ושל יום טוב, בסוף פסוקי דזמרה, אחרי שירת הים ולפני "ישתבח". אומרים אותה גם בליל הסדר, בסוף ההגדה. מלבד זה יש נוהגים לומר אותה כתפילת הודיה אחרי ישועה, בכל יום שרוצים.

מה זה "סגולת נשמת כל חי"?

מנהג שנפוץ בקהילות רבות: מי שנמצא בצרה מקבל על עצמו לומר "נשמת כל חי" בציבור, כשייוושע, ויש שאומרים אותה גם כבקשה. זה מנהג ולא חובה. התפילה עצמה כולה שבח והודיה, ולכן היא מתאימה לכל מי שרוצה לומר תודה.

מה ההבדל בין הנוסחים?

התפילה כמעט זהה בכל הנוסחים. ההבדלים הם בכמה מילים, ובעיקר בסוף: בנוסח עדות המזרח ובנוסח ספרד אומרים "בפי ישרים תתרומם", ובנוסח אשכנז "בפי ישרים תתהלל". בוחרים נוסח בכפתורים שלמעלה, והדף זוכר את הבחירה.

איפה נגמרת התפילה?

התפילה עצמה מסתיימת במילים "ברכי נפשי את ה' וכל קרבי את שם קדשו". בתפילת שבת ממשיכים מיד ל"האל בתעצומות עוזך", "שוכן עד", "ובמקהלות" ו"ישתבח", ולכן הבאנו גם אותם, כל אחד תחת כותרת משלו.

מי חיבר את נשמת כל חי?

התפילה עתיקה מאוד, והיא נזכרת כבר בתלמוד בשם "ברכת השיר" (פסחים קיח). שם המחבר אינו ידוע בוודאות.

מאיפה הטקסט שבדף?

מספריא, מילה במילה: עדות המזרח מסידור מהדורת שלייסבו, ספרד מסידור "תורת אמת", ואשכנז מסידור "דעת". לא השלמנו שום מילה מהזיכרון. בתצוגה בלבד הורדנו טעמים וכתבנו את שמות הקודש בכינוי. בנוסח אשכנז הניקוד במקור חסר בכמה אותיות (חולם), והשארנו אותו כמו שהוא.

אפשר להדפיס או לשלוח?

כן. בתפריט הדפדפן בוחרים "הדפסה" והדף יוצא נקי, בלי הכפתורים. ובכפתור הווטסאפ שולחים את הקישור למי שביקש שתאמרו איתו. הכול בחינם, בלי הרשמה ובלי פרסומות.

עוד כלים בחינם שאולי יעזרו

אולפן לרב בטלפון: לצלם שיעור בלי ציודאולפן לרב בטלפון: כלים חינמיים לצילום שיעור ודבר תורה: טלפרומפטר שעוקב אחרי הקול, מראה חכמה…פרקי אבות לספירת העומרפרקי אבות מחולקים ל-49 ימי ספירת העומר: בכל יום משנה או שתיים, מנוקדות, עם פירוש קליל ונקודה…פרשת השבוע בקצרה: תקציר, פסוק ורעיון לשולחן שבתפרשת השבוע בקצרה: מה הפרשה השבת, תקציר בכמה שורות, הפסוק המרכזי, רעיון קצר לשולחן שבת ושאלה…פוסט מעוצב: טקסט כתמונה לסטטוס ולרשתותפוסט מעוצב: כותבים משפט, ציטוט, הודעה לקהילה או מחשבה, ומקבלים תמונה מעוצבת לסטטוס, לפייסבוק…
