כִּי שֵׁם ה' אֶקְרָא הָבוּ גֹדֶל לֵאלֹקֵינוּ:

אֲ-דֹנָי, שְׂפָתַי תִּפְתָּח וּפִי יַגִּיד תְּהִלָּתֶךָ:

בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, אֱלֹקֵי אַבְרָהָם, אֱלֹקֵי יִצְחָק וֵאלֹקֵי יַעֲקֹב. הָאֵל הַגָּדוֹל, הַגִּבּוֹר וְהַנּוֹרָא, אֵל עֶלְיוֹן, גּוֹמֵל חֲסָדִים טוֹבִים, וְקוֹנֵה הַכֹּל, וְזוֹכֵר חַסְדֵי אָבוֹת, וּמֵבִיא גוֹאֵל לִבְנֵי בְנֵיהֶם לְמַעַן שְׁמוֹ, בְּאַהֲבָה. מֶלֶךְ עוֹזֵר וּמוֹשִׁיעַ וּמָגֵן, בָּרוּךְ אַתָּה ה', מָגֵן אַבְרָהָם:

גבורות

אַתָּה גִּבּוֹר לְעוֹלָם, אֲ-דֹנָי. מְחַיֵּה מֵתִים אַתָּה, רַב לְהוֹשִׁיעַ.

בקיץ: מוֹרִיד הַטָּל. בחורף: מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם.

מְכַלְכֵּל חַיִּים בְּחֶסֶד, מְחַיֵּה מֵתִים בְּרַחֲמִים רַבִּים, סוֹמֵךְ נוֹפְלִים, וְרוֹפֵא חוֹלִים, וּמַתִּיר אֲסוּרִים, וּמְקַיֵּם אֱמוּנָתוֹ לִישֵׁנֵי עָפָר. מִי כָמוֹךָ בַּעַל גְּבוּרוֹת, וּמִי דוֹמֶה לָּךְ, מֶלֶךְ מֵמִית וּמְחַיֶּה וּמַצְמִיחַ יְשׁוּעָה. וְנֶאֱמָן אַתָּה לְהַחֲיוֹת מֵתִים. בָּרוּךְ אַתָּה ה', מְחַיֵּה הַמֵּתִים:

למוסף של חג, הושענא רבה, או שבת חוה"מ:

כֶּתֶר יִתְּנוּ לְךָ, ה' אֱלֹקֵינוּ מַלְאָכִים הֲמוֹנֵי מַעְלָה. עִם עַמְּךָ יִשְׂרָאֵל קְבוּצֵי מַטָּה: יַחַד כֻּלָּם קְדֻשָּׁה לְךָ יְשַׁלֵּשׁוּ כַּדָּבָר הָאָמוּר עַל יַד נְבִיאֶךָ וְקָרָא זֶה אֶל זֶה וְאָמַר: קו"ח: קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ ה' צְבָאוֹת מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ: חזן: כְּבוֹדוֹ מָלֵא עוֹלָם. מְשָׁרְתָיו שׁוֹאֲלִים זֶה לָזֶה אַיֵּה מְקוֹם כְּבוֹדוֹ לְהַעֲרִיצוֹ. לְעֻמָּתָם מְשַׁבְּחִים וְאוֹמְרִים: קו"ח: בָּרוּךְ כְּבוֹד ה' מִמְּקוֹמוֹ: חזן: מִמְּקוֹמוֹ הוּא יִפֶן בְּרַחֲמָיו לְעַמּוֹ. וְיָחוֹן עַם הַמְיַחֲדִים שְׁמוֹ עֶרֶב וָבוֹקֶר בְּכָל יוֹם תָּמִיד. פַּעֲמַיִם בְּאַהֲבָה שְׁמַע אוֹמְרִים: קו"ח: שְׁמַע יִשְׂרָאֵל ה' אֱלֹקֵינוּ ה' אֶחָד: חזן: הוּא אֱלֹקֵינוּ הוּא אָבִינוּ. הוּא מַלְכֵּנוּ הוּא מוֹשִׁיעֵנוּ. וְהוּא יוֹשִׁיעֵנוּ וְיִגְאָלֵנוּ שֵׁנִית וְיַשְׁמִיעֵנוּ בְּרַחֲמָיו שֵׁנִית לְעֵינֵי כָּל חַי לֵאמֹר. הֵן גָּאַלְתִּי אֶתְכֶם אַחֲרִית כְּרֵאשִׁית לִהְיוֹת לָכֶם לֵאלֹקִים: קו"ח: אֲנִי ה' אֱלֹקֵיכֶם: חזן: וּבְדִבְרֵי קָדְשְׁךָ כָּתוּב לֵאמֹר: קו"ח: יִמְלֹךְ ה' לְעוֹלָם. אֱלֹקַיִךְ צִיּוֹן לְדֹר וָדֹר. הַלְלוּיָהּ:

למוסף של חוה"מ:

כֶּתֶר יִתְּנוּ לְךָ, ה' אֱלֹקֵינוּ מַלְאָכִים הֲמוֹנֵי מַעְלָה. עִם עַמְּךָ יִשְׂרָאֵל קְבוּצֵי מַטָּה: יַחַד כֻּלָּם קְדֻשָּׁה לְךָ יְשַׁלֵּשׁוּ כַּדָּבָר הָאָמוּר עַל יַד נְבִיאֶךָ וְקָרָא זֶה אֶל זֶה וְאָמַר: קו"ח: קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ ה' צְבָאוֹת מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ: חזן: לְעֻמָּתָם מְשַׁבְּחִים וְאוֹמְרִים: קו"ח: בָּרוּךְ כְּבוֹד ה' מִמְּקוֹמוֹ: חזן: וּבְדִבְרֵי קָדְשְׁךָ כָּתוּב לֵאמֹר: קו"ח: יִמְלֹךְ ה' לְעוֹלָם. אֱלֹקַיִךְ צִיּוֹן לְדֹר וָדֹר. הַלְלוּיָהּ:

חזן: לְדוֹר וָדוֹר נַגִּיד גָּדְלֶךָ. וּלְנֵצַח נְצָחִים קְדֻשָּׁתְךָ נַקְדִּישׁ. וְשִׁבְחֲךָ אֱלֹקֵינוּ מִפִּינוּ לֹא יָמוּשׁ לְעוֹלָם וָעֶד. כִּי אֵל מֶלֶךְ גָּדוֹל וְקָדוֹשׁ אָתָּה: בָּרוּךְ אַתָּה ה'. הָאֵל הַקָּדוֹשׁ:

אַתָּה קָדוֹשׁ, וְשִׁמְךָ קָדוֹשׁ, וּקְדוֹשִׁים בְּכָל יוֹם יְהַלְלוּךָ סֶּלָה. בָּרוּךְ אַתָּה ה', הָאֵל הַקָּדוֹשׁ:

קדושת היום

אַתָּה בְחַרְתָּנוּ מִכָּל הָעַמִּים, אָהַבְתָּ אוֹתָנוּ וְרָצִיתָ בָּנוּ, וְרוֹמַמְתָּנוּ מִכָּל הַלְּשׁוֹנוֹת, וְקִדַּשְׁתָּנוּ בְּמִצְוֹתֶיךָ, וְקֵרַבְתָּנוּ מַלְכֵּנוּ לַעֲבוֹדָתֶךָ, וְשִׁמְךָ הַגָּדוֹל וְהַקָּדוֹשׁ עָלֵינוּ קָרָאתָ:

וַתִּתֶּן לָנוּ ה' אֱלֹקֵינוּ בְּאַהֲבָה (לשבת שַׁבָּתוֹת לִמְנוּחָה וּ) מוֹעֲדִים לְשִׂמְחָה חַגִּים וּזְמַנִּים לְשָׂשׂוֹן אֶת יוֹם (לשבת הַשַׁבָּת הַזֶּה וְאֶת יוֹם) :

בשבועות: חַג הַשָׁבוּעוֹת הַזֶּה. זְמַן מַתַּן תּוֹרָתֵינוּ:

בסוכות: חַג הַסֻּכּוֹת הַזֶּה. זְמַן שִׂמְחָתֵנוּ:

וּמִפְּנֵי חֲטָאֵינוּ גָּלִינוּ מֵאַרְצֵנוּ. וְנִתְרַחַקְנוּ מֵעַל אַדְמָתֵנוּ. וְאֵין אֲנַחְנוּ יְכוֹלִים לַעֲלוֹת וְלֵרָאוֹת וּלְהִשְׁתַּחֲוֹת לְפָנֶיךָ. וְלַעֲשׂוֹת חוֹבוֹתֵינוּ בְּבֵית בְּחִירָתֶךָ. בַּבַּיִת הַגָּדוֹל וְהַקָּדוֹשׁ שֶׁנִּקְרָא שִׁמְךָ עָלָיו. מִפְּנֵי הַיָּד שֶׁנִּשְׁתַּלְּחָה בְּמִקְדָּשֶׁךָ: יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ. מֶלֶךְ רַחֲמָן. שֶׁתָּשׁוּב וּתְרַחֵם עָלֵינוּ וְעַל מִקְדָּשְׁךָ בְּרַחֲמֶיךָ הָרַבִּים. וְתִבְנֵהוּ מְהֵרָה וּתְגַדֵּל כְּבוֹדוֹ: אָבִינוּ מַלְכֵּנוּ. גַּלֵּה כְּבוֹד מַלְכוּתְךָ עָלֵינוּ מְהֵרָה. וְהוֹפַע וְהִנָּשֵׂא עָלֵינוּ לְעֵינֵי כָּל חָי. וְקָרֵב פְּזוּרֵינוּ מִבֵּין הַגּוֹיִם. וּנְפוּצוֹתֵינוּ כַּנֵּס מִיַּרְכְּתֵי אָרֶץ. וַהֲבִיאֵנוּ לְצִיּוֹן עִירְךָ בְּרִנָּה. וְלִירוּשָׁלַיִם בֵּית מִקְדָּשְׁךָ בְּשִׂמְחַת עוֹלָם. וְשָׁם נַעֲשֶׂה לְפָנֶיךָ אֶת קָרְבְּנוֹת חוֹבוֹתֵינוּ תְּמִידִים כְּסִדְרָם וּמוּסָפִים כְּהִלְכָתָם:

מוסף חול המועד מלא לקריאה בטלפון

מוסף של חול המועד הוא תפילת העמידה הנוספת שאומרים אחרי שחרית בימי חול המועד של פסח ושל סוכות. כאן הוא כתוב במלואו, בניקוד מלא ובאות גדולה, בדף אחד רצוף, בלי פרסומות באמצע התפילה ובלי חלונות קופצים.

הדף שלפניכם הוא בנוסח ספרד. בכפתורים שבראש הדף עוברים לנוסח עדות המזרח, לנוסח ספרד או לנוסח אשכנז. הטקסט הוא תפילת המוסף לשלוש רגלים, כמו שהיא כתובה בסידורים, ולכן יש בו הוראות שמבדילות בין יום טוב, שבת וחול המועד. את מה שאומרים בחול המועד מחפשים לפי ההוראות באותיות הקטנות.

איך משתמשים בדף

שורת הקיצורים שמתחת לכותרת מובילה ישר לחלק שצריך: קדושה, קדושת היום, ברכת כהנים. בכפתורים א+ וא- משנים את גודל האות, והירח מעביר לרקע כהה. ההוראות ("בחול המועד אומרים", "בשבת מוסיפים") כתובות באות קטנה ובצבע אחר, כדי שלא יתערבבו עם מילות התפילה.

סדר התפילה בקצרה

מוסף היא תפילת עמידה של שבע ברכות: שלוש ברכות שבח בהתחלה, ברכת קדושת היום באמצע, ושלוש ברכות הודאה בסוף. בקדושת היום של חול המועד מזכירים את קרבנות המוסף של החג ואת מה שייחודי לפסח או לסוכות.

אחרי שחרית, ובדרך כלל אחרי ההלל וקריאת התורה של חול המועד. זמן התפילה תלוי במקום ובתאריך. בשאלה למעשה שואלים רב.

המילים והניקוד נלקחו מספריא, מילה במילה, מתוך סידורים שמותר להפיץ. לא השלמנו שום מילה מהזיכרון. בתצוגה בלבד הורדנו את טעמי המקרא וכתבנו את שמות הקודש בכינוי. הדף אינו תחליף לפסיקת רב.

שאלות ותשובות

מה זה מוסף חול המועד?

אומרים מוסף חול המועד גם בשבת?

כן, בשבת של חול המועד אומרים מוסף עם התוספות לשבת, כמו שהן כתובות בהוראות בטקסט.

איזה נוסחים יש כאן?

שלושה: נוסח עדות המזרח, נוסח ספרד ונוסח אשכנז. לכל נוסח יש דף משלו, ועוברים ביניהם בכפתורים שבראש הדף. כל אחד מתפלל בנוסח של מנהג אבותיו.

מאיפה הטקסט, ואפשר לסמוך עליו?

המילים והניקוד מגיעים מספריא, מילה במילה, מתוך סידורים שמותר להפיץ: מהדורת שלייסבו לעדות המזרח, סידור "מצודה" ו"תורת אמת" לספרד, ו"מצודה" ו"דעת" לאשכנז. לא הוספנו ולא השלמנו שום מילה מהזיכרון. השינוי היחיד הוא בתצוגה: בלי טעמי המקרא, ושמות הקודש כתובים בכינוי, כדי שיהיה מותר לקרוא גם ממסך. מי שמצא טעות מוזמן לכתוב לנו.
