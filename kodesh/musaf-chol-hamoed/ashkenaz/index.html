בס"ד

נוסח: אשכנז

אבות

כִּי שֵׁם ה' אֶקְרָא הָבוּ גדֶל לֵאלקֵינוּ:

אֲדנָי שפָתַי תִּפְתָּח וּפִי יַגִּיד תְּהִלָּתֶךָ:

בָּרוּךְ אַתָּה ה' אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ. אֱלקֵי אַבְרָהָם. אֱלקֵי יִצְחָק. וֵאלקֵי יַעֲקב. הָאֵל הַגָּדול הַגִּבּור וְהַנּורָא אֵל עֶלְיון. גּומֵל חֲסָדִים טובִים. וְקונֵה הַכּל. וְזוכֵר חַסְדֵי אָבות. וּמֵבִיא גואֵל לִבְנֵי בְנֵיהֶם לְמַעַן שְׁמו בְּאַהֲבָה:

מֶלֶךְ עוזֵר וּמושִׁיעַ וּמָגֵן:

בָּרוּךְ אַתָּה ה'. מָגֵן אַבְרָהָם:

גבורות

אַתָּה גִבּור לְעולָם אֲדנָי מְחַיֶּה מֵתִים אַתָּה רַב לְהושִׁיעַ:

בחורף: מַשִׁיב הָרוּחַ וּמורִיד הַגֶּשֶׁם:

בקיץ: מורִיד הַטָּל:

מְכַלְכֵּל חַיִּים בְּחֶסֶד. מְחַיֶּה מֵתִים בְּרַחֲמִים רַבִּים. סומֵךְ נופְלִים וְרופֵא חולִים וּמַתִּיר אֲסוּרִים. וּמְקַיֵּם אֱמוּנָתו לִישֵׁנֵי עָפָר. מִי כָמוךָ בַּעַל גְּבוּרות וּמִי דומֶה לָךְ. מֶלֶךְ מֵמִית וּמְחַיֶּה וּמַצְמִיחַ יְשׁוּעָה:

וְנֶאֱמָן אַתָּה לְהַחֲיות מֵתִים:

בָּרוּךְ אַתָּה ה' מְחַיֶּה הַמֵּתִים:

קדושה

ליו"ט ולהושע"ר:

נַעֲרִיצְךָ וְנַקְדִישְׁךָ כְּסוד שיחַ שרְפֵי קדֶשׁ הַמַּקְדִּישִׁים שִׁמְךָ בַּקּדֶשׁ כַּכָּתוּב עַל יַד נְבִיאֶךָ. וְקָרָא זֶה אֶל זֶה וְאָמַר:

קו"ח - קָדושׁ קָדושׁ קָדושׁ ה' צְבָאות מְלא כָל הָאָרֶץ כְּבודו:

חזן - כְּבודו מָלֵא עולָם. מְשָׁרְתָיו שׁואֲלִים זֶה לָזֶה אַיֵּה מְקום כְּבודו. לְעֻמָּתָם בָּרוּךְ יאמֵרוּ:

קו"ח - בָּרוּךְ כְּבוד ה' מִמְּקומו:

חזן - מִמְּקומו הוּא יִפֶן בְּרַחֲמִים וְיָחון עַם הַמְיַחֲדִים שְׁמו עֶרֶב וָבוקֶר בְּכָל יום תָּמִיד. פַּעֲמַיִם בְּאַהֲבָה שְׁמַע אומְרִים:

קו"ח - שְׁמַע יִשרָאֵל ה' אֱלקֵינוּ ה' אֶחָד:

חזן - הוּא אֱלקֵינוּ הוּא אָבִינוּ. הוּא מַלְכֵּנוּ הוּא מושִׁיעֵנוּ. וְהוּא יַשְׁמִיעֵנוּ בְּרַחֲמָיו שֵׁנִית לְעֵינֵי כָּל חָי. לִהְיות לָכֶם לֵאלקִים:

קו"ח - אֲנִי ה' אֱלקֵיכֶם:

אֲנִי ה' אֱלקֵיכֶם:

לחוה"מ:

נְקַדֵּשׁ אֶת שִׁמְךָ בָּעולָם. כְּשֵׁם שֶׁמַּקְדִּישִׁים אותו בִּשְׁמֵי מָרום. כַּכָּתוּב עַל יַד נְבִיאֶךָ וְקָרָא זֶה אֶל זֶה וְאָמַר:

קו"ח - קָדושׁ קָדושׁ קָדושׁ ה' צְבָאות מְלא כָל הָאָרֶץ כְּבודו:

חזן - לְעֻמָּתָם בָּרוּךְ יאמֵרוּ:

קו"ח - בָּרוּךְ כְּבוד ה' מִמְּקומו:

חזן - וּבְדִבְרֵי קָדְשְׁךָ כָּתוּב לֵאמר:

קו"ח - יִמְלךְ ה' לְעולָם אֱלקַיִךְ צִיּון לְדור וָדור הַלְלוּיָהּ:

בשבת חול המועד מדלגים קטע זה:

קו"ח - אַדִּיר אַדִּירֵנוּ ה' אֲדנֵינוּ מָה אַדִּיר שִׁמְךָ בְּכָל הָאָרֶץ. וְהָיָה ה' לְמֶלֶךְ עַל כָּל הָאָרֶץ. בַּיּום הַהוּא יִהְיֶה ה' אֶחָד וּשְמו אֶחָד:

חזן - וּבְדִבְרֵי קָדְשְׁךָ כָּתוּב לֵאמר:

קו"ח - יִמְלךְ ה' לְעולָם אֱלקַיִךְ צִיּון לְדר וָדר הַלְלוָּיהּ:

חזן - לְדור וָדור נַגִּיד גָּדְלֶךָ. וּלְנֵצַח נְצָחִים קְדֻשָּׁתְךָ נַקְדִּישׁ. וְשִׁבְחֲךָ אֱלקֵינוּ מִפִּינוּ לא יָמוּשׁ לְעולָם וָעֶד. כִּי אֵל מֶלֶךְ גָּדול וְקָדושׁ אָתָּה:

בָּרוּךְ אַתָּה ה'. הָאֵל הַקָּדושׁ: [הש"ץ ממשיך: אתה בחרתנו]

קדושת השם

אַתָּה קָדושׁ וְשִׁמְךָ קָדושׁ. וּקְדושִׁים בְּכָל יום יְהַלְלוּךָ סֶּלָה:

כִּי אֵל מֶלֶךְ גָּדול וְקָדושׁ אָתָּה:

בָּרוּךְ אַתָּה ה' הָאֵל הַקָּדושׁ:

קדושת היום

אַתָּה בְּחַרְתָּנוּ מִכָּל הָעַמִּים אָהַבְתָּ אוֹתָנוּ וְרָצִיתָ בָּנוּ וְרוֹמַמְתָּנוּ מִכָּל הַלְּשׁוֹנוֹת וְקִדַּשְׁתָּנוּ בְּמִצְוֹתֶיךָ וְקֵרַבְתָּנוּ מַלְכֵּנוּ לַעֲבוֹדָתֶךָ וְשִׁמְךָ הַגָּדוֹל וְהַקָּדוֹשׁ עָלֵינוּ קָרָאתָ:

וַתִּתֶּן לָנוּ ה' אֱלקֵינוּ בְּאַהֲבָה לשבת: שַׁבָּתות לִמְנוּחָה וּ מועֲדִים לְשמְחָה חַגִּים וּזְמַנִּים לְששון אֶת יום:

לשבת - הַשַׁבָּת הַזֶּה וְאֶת יום:

לפסח - חַג הַמַּצּות הַזֶּה זְמַן חֵרוּתֵנוּ:

לשבועות - חַג הַשָׁבוּעות הַזֶּה זְמַן מַתַּן תּורָתֵינוּ:

סוכות - חַג הַסֻּכּות הַזֶּה זְמַן שמְחָתֵנוּ:

לשמ"ע ולש"ת - הַשְׁמִינִי חַג הָעֲצֶרֶת הַזֶּה זְמַן שמְחָתֵנוּ:

לשבת בְּאַהֲבָה מִקְרָא קדֶשׁ זֵכֶר לִיצִיאַת מִצְרָיִם:

וּמִפְּנֵי חֲטָאֵינוּ גָּלִינוּ מֵאַרְצֵנוּ. וְנִתְרַחַקְנוּ מֵעַל אַדְמָתֵנוּ. וְאֵין אֲנַחְנוּ יְכוֹלִים לַעֲלוֹת וְלֵרָאוֹת וּלְהִשְׁתַּחֲוֹת לְפָנֶיךָ. וְלַעֲשׂוֹת חוֹבוֹתֵינוּ בְּבֵית בְּחִירָתֶךָ. בַּבַּיִת הַגָּדוֹל וְהַקָּדוֹשׁ שֶׁנִּקְרָא שִׁמְךָ עָלָיו. מִפְּנֵי הַיָּד שֶׁנִּשְׁתַּלְּחָה בְּמִקְדָּשֶׁךָ:

יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ. מֶלֶךְ רַחֲמָן. שֶׁתָּשׁוּב וּתְרַחֵם עָלֵינוּ וְעַל מִקְדָּשְׁךָ בְּרַחֲמֶיךָ הָרַבִּים. וְתִבְנֵהוּ מְהֵרָה וּתְגַדֵּל כְּבוֹדוֹ:

אָבִינוּ מַלְכֵּנוּ. גַּלֵּה כְּבוֹד מַלְכוּתְךָ עָלֵינוּ מְהֵרָה. וְהוֹפַע וְהִנָּשֵׂא עָלֵינוּ לְעֵינֵי כָּל חַי. וְקָרֵב פְּזוּרֵינוּ מִבֵּין הַגּוֹיִם. וּנְפוּצוֹתֵינוּ כַּנֵּס מִיַּרְכְּתֵי אָרֶץ. וַהֲבִיאֵנוּ לְצִיּוֹן עִירְךָ בְּרִנָּה. וְלִירוּשָׁלַיִם בֵּית מִקְדָּשְׁךָ בְּשִׂמְחַת עוֹלָם. וְשָׁם נַעֲשֶׂה לְפָנֶיךָ אֶת קָרְבְּנוֹת חוֹבוֹתֵינוּ תְּמִידִים כְּסִדְרָם וּמוּסָפִים כְּהִלְכָתָם:

וְאֶת מוּסַף לשבת וְאֶת מוּסְפֵי יום הַשַּׁבָּת הַזֶּה וְ יום:

לפסח - חַג הַמַּצּות הַזֶּה:

לשבועות - חַג הַשָׁבוּעות הַזֶּה:

סוכות - חַג הַסֻּכּות הַזֶּה:

לשמ"ע ולש"ת - הַשְׁמִינִי חַג הָעֲצֶרֶת הַזֶּה:

נַעֲשֶׂה וְנַקְרִיב לְפָנֶיךָ בְּאַהֲבָה כְּמִצְוַת רְצוֹנֶךָ. כְּמוֹ שֶׁכָּתַבְתָּ עָלֵינוּ בְּתוֹרָתֶךָ עַל יְדֵי מֹשֶׁה עַבְדֶּךָ. מִפִּי כְּבוֹדֶךָ כָּאָמוּר:

לשבת וּבְיום הַשַּׁבָּת. שְׁנֵי כְבָשים בְּנֵי שָׁנָה תְּמִימִים. וּשְׁנֵי עֶשרנִים סלֶת מִנְחָה בְּלוּלָה בַשֶּׁמֶן וְנִסְכּו:

עלַת שַׁבַּת בְּשַׁבַּתּו. עַל עלַת הַתָּמִיד וְנִסְכָּהּ:

ליום א' וב' דפסח:

וּבַחדֶשׁ הָרִאשׁון בְּאַרְבָּעָה עָשר יום לַחדֶשׁ. פֶּסַח לה':

וּבַחֲמִשָּׁה עָשר יום לַחדֶשׁ הַזֶּה חָג. שִׁבְעַת יָמִים מַצּות יֵאָכֵל:

בַּיּום הָרִאשׁון מִקְרָא קדֶשׁ. כָּל מְלֶאכֶת עֲבדָה לא תַעֲשוּ:

וְהִקְרַבְתֶּם אִשֶּׁה עלָה לה'. פָּרִים בְּנֵי בָקָר שְׁנַיִם. וְאַיִל אֶחָד. וְשִׁבְעָה כְבָשים בְּנֵי שָׁנָה. תְּמִימִם יִהְיוּ לָכֶם:

ומנחתם:

לשבועות:

וּבְיום הַבִּכּוּרִים. בְּהַקְרִיבְכֶם מִנְחָה חֲדָשָׁה לה' בְּשָׁבֻעותֵיכֶם מִקְרָא קדֶשׁ יִהְיֶה לָכֶם. כָּל מְלֶאכֶת עֲבדָה לא תַעֲשוּ:

וְהִקְרַבְתֶּם עלָה לְרֵיחַ נִיחחַ לה'. פָּרִים בְּנֵי בָקָר שְׁנַיִם. אַיִל אֶחָד. שִׁבְעָה כְבָשים בְּנֵי שָׁנָה:

ומנחתם:

ליום א' וב' דסכות:

וּבַחֲמִשָּׁה עָשר יום לַחדֶשׁ הַשְּׁבִיעִי. מִקְרָא קדֶשׁ יִהְיֶה לָכֶם כָּל מְלֶאכֶת עֲבדָה לא תַעֲשוּ. וְחַגּתֶם חַג לה' שִׁבְעַת יָמִים:

וְהִקְרַבְתֶּם עלָה אִשֵּׁה רֵיחַ נִיחחַ לה'. פָּרִים בְּנֵי בָקָר שְׁלשָׁה עָשר. אֵילִם שְׁנָיִם. כְּבָשים בְּנֵי שָׁנָה אַרְבָּעָה עָשר. תְּמִימִם יִהְיוּ:

ומנחתם:

לשבת: יִשמְחוּ בְּמַלְכוּתְךָ שׁומְרֵי שַׁבָּת וְקורְאֵי עונֶג עַם מְקַדְּשֵׁי שְׁבִיעִי. כֻּלָּם יִשבְּעוּ וְיִתְעַנְּגוּ מִטּוּבֶךָ. וּבַשְּׁבִיעִי רָצִיתָ בּו וְקִדַּשְׁתּו חֶמְדַת יָמִים אותו קָרָאתָ. זֵכֶר לְמַעֲשה בְרֵאשִׁית:

אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ. לשבת רְצֵה בִמְנוּחָתֵנוּ מֶלֶךְ רַחֲמָן רַחֵם עָלֵינוּ. טוב וּמֵטִיב הִדָּרֶשׁ לָנוּ. שׁוּבָה אֵלֵינוּ בַּהֲמון רַחֲמֶיךָ. בִּגְלַל אָבות שֶׁעָשוּ רְצונֶךָ. בְּנֵה בֵיתְךָ כְּבַתְּחִלָּה וְכונֵן מִקְדָּשְׁךָ עַל מְכונו. וְהַרְאֵנוּ בְּבִנְיָנו וְשמְּחֵנוּ בְּתִקּוּנו. וְהָשֵׁב כּהֲנִים לַעֲבודָתָם וּלְוִיִּם לְשִׁירָם וּלְזִמְרָם. וְהָשֵׁב יִשרָאֵל לִנְוֵיהֶם. וְשָׁם נַעֲלֶה וְנֵרָאֶה וְנִשְׁתַּחֲוֶה לְפָנֶיךָ בְּשָׁלשׁ פַּעֲמֵי רְגָלֵינוּ כַּכָּתוּב בְּתורָתֶךָ. שָׁלשׁ פְּעָמִים בַּשָּׁנָה יֵרָאֶה כָל זְכוּרְךָ אֶת פְּנֵי ה' אֱלקֶיךָ בַּמָּקום אֲשֶׁר יִבְחָר. בְּחַג הַמַּצּות. וּבְחַג הַשָּׁבֻעות. וּבְחַג הַסֻּכּות. וְלא יֵרָאֶה אֶת פְּנֵי ה' רֵיקָם. אִישׁ כְּמַתְּנַת יָדו כְּבִרְכַּת ה' אֱלקֶיךָ אֲשֶׁר נָתַן לָךְ:

וְהַשּיאֵנוּ ה' אֱלקֵינוּ אֶת בִּרְכַּת מועֲדֶיךָ. לְחַיִּים וּלְשָׁלום לְשמְחָה וּלְששון. כַּאֲשֶׁר רָצִיתָ וְאָמַרְתָּ לְבָרְכֵנוּ. לשבת אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ רְצֵה נָא בִמְנוּחָתֵנוּ קַדְּשֵׁנוּ בְּמִצְותֶיךָ. וְתֵן חֶלְקֵנוּ בְּתורָתֶךָ. שבְּעֵנוּ מִטּוּבֶךָ וְשמְּחֵנוּ בִּישׁוּעָתֶךָ. וְטַהֵר לִבֵּנוּ לְעָבְדְּךָ בֶּאֱמֶת. וְהַנְחִילֵנוּ ה' אֱלקֵינוּ לשבת בְּאַהֲבָה וּבְרָצון בְּשמְחָה וּבְששון שַׁבָּת וּ מועֲדֵי קָדְשֶׁךָ. וְיִשמְחוּ בְךָ יִשרָאֵל מְקַדְּשֵׁי שְׁמֶךָ:

מודִים אֲנַחְנוּ לָךְ. שָׁאַתָּה הוּא ה' אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ לְעולָם וָעֶד. צוּר חַיֵּינוּ. מָגֵן יִשְׁעֵנוּ אַתָּה הוּא לְדור וָדור נודֶה לְּךָ וּנְסַפֵּר תְּהִלָּתֶךָ עַל חַיֵּינוּ הַמְּסוּרִים בְּיָדֶךָ. וְעַל נִשְׁמותֵינוּ הַפְּקוּדות לָךְ. וְעַל נִסֶּיךָ שֶׁבְּכָל יום עִמָּנוּ. וְעַל נִפְלְאותֶיךָ וְטובותֶיךָ שֶׁבְּכָל עֵת. עֶרֶב וָבקֶר וְצָהֳרָיִם:

מודים דרבנן:

מודִים אֲנַחְנוּ לָךְ. שָׁאַתָּה הוּא ה' אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ. אֱלקֵי כָל בָּשר. יוצְרֵנוּ יוצֵר בְּרֵאשִׁית. בְּרָכות וְהודָאות לְשִׁמְךָ הַגָּדול וְהַקָּדושׁ. עַל שֶׁהֶחֱיִיתָנוּ וְקִיַּמְתָּנוּ. כֵּן תְּחַיֵּנוּ וּתְקַיְּמֵנוּ. וְתֶאֱסוף גָּלֻיּותֵינוּ לְחַצְרות קָדְשֶׁךָ. לִשְׁמור חֻקֶּיךָ. וְלַעֲשות רְצונֶךָ. וּלְעָבְדְךָ בְּלֵבָב שָׁלֵם. עַל שֶׁאָנוּ מודִים לָךְ. בָּרוּךְ אֵל הַהודָאות:

וְעַל כֻּלָּם יִתְבָּרֵךְ וְיִתְרומַם וְיִתְנַשּא שִׁמְךָ מַלְכֵּנוּ תָּמִיד לְעולָם וָעֶד:

וְכל הַחַיִּים יודוּךָ סֶּלָה. וִיהַלְלוּ אֶת שִׁמְךָ בֶּאֱמֶת, לְעולָם כִּי טוב. הָאֵל יְשׁוּעָתֵנוּ וְעֶזְרָתֵנוּ סֶלָה:

בָּרוּךְ אַתָּה ה'. הַטּוב שִׁמְךָ וּלְךָ נָאֶה לְהודות:

ברכת כוהנים

יְהִי רָצון מִלְּפָנֶיךָ ה' אֱלקֵינוּ וֵאלקֵי אֲבותֵינוּ. שֶׁתְּהֵא הַבְּרָכָה הַזּאת לכהנים: שֶׁצִּוִּיתָנו

לקהל: שֶׁצִּוִּיתָ לְבָרֵךָ אֶת עַמְּךָ יִשרָאֵל. בְּרָכָה שְׁלֵמָה. וְלא יִהְיֶה בָּהּ שׁוּם מִכְשׁול וְעָון. מֵעַתָּה וְעַד עולָם:

וְתֵעָרֵב עָלֶיךָ [נ"א: לפניך] עֲתִירָתֵנוּ כְּעוֹלָה וּכְקָרְבָּן. אָנָּא רַחוּם, בְּרַחֲמֶיךָ הָרַבִּים הָשֵׁב שְׁכִינָתְךָ לְצִיּוֹן (עִירֶךָ) וְסֵדֶר הַעֲבוֹדָה לִירוּשָׁלָיִם.

ברוב קהילות ארה״ב: וְתֶחֱזֶינָה עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמוֹנִיּוֹת. בָּרוּך אַתָּה ה׳ שֶׁאוֹתְךָ לְבַדְּךָ בְּיִרְאָה נַעֲבֹד.

נוסח הגר״א ומנהג ארץ ישראל: וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמוֹנִיּוֹת. וְתֶחֱזֶינָה עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. בָּרוּךְ אַתָּה ה׳ הַמַּחֲזִיר שְׁכִינָתוֹ לְצִיּוֹן.

מודים וכו׳ עד להודות. ויש נוהגין שהכהנים אומרים בלחש: יְהִי רָצוֹן מִלְּפָנֶיךָ ה׳ אֱלקֵינוּ וֵאלקֵי אֲבוֹתֵינוּ. שֶׁתְּהֵא הַבְּרָכָה הַזֹּאת שֶׁצִּוִּיתָנוּ (נ״א שֶׁצִּוִּיתָ) לְבָרֵךָ אֶת עַמְּךָ יִשְׂרָאֵל. בְּרָכָה שְׁלֵמָה. וְלא יִהְיֶה בָּהּ שׁוּם מִכְשׁוֹל וְעָוֹן. מֵעַתָּה וְעַד עוֹלָם:

החזן בלחש: אֱלֹקֵינוּ וֵאלֹקֵי אֲבותֵינוּ. בָּרְכֵנוּ בַבְּרָכָה הַמְּשֻׁלֶּשֶׁת בַּתּוֹרָה הַכְּתוּבָה עַל יְדֵי משֶׁה עַבְדֶּךָ. הָאֲמוּרָה מִפִּי אַהֲרֹן וּבָנָיו—

החזן בקול רם: כֹּהֲנִים—

ועוד שהכהנים אומרים: רִבּוֹנוֹ שֶׁל עוֹלָם. עָשִׂינוּ מַה שֶׁגָּזַרְתָּ עָלֵינוּ. אַף אַתָּה עֲשֵׂה עִמָּנוּ כְּמוֹ שֶׁהִבְטַחְתָּנוּ. הַשְׁקִיפָה מִמְּעוֹן קָדְשְׁךָ מִן הַשָּׁמַיִם. וּבָרֵךָ אֶת עַמְּךָ אֶת יִשְׂרָאֵל. וְאֵת הָאֲדָמָה אֲשֶׁר נָתַתָּה לָנוּ. כַּאֲשֶׁר נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ. אֶרֶץ זָבַת חָלָב וּדְבָשׁ.

בָּרוּךְ אַתָּה ה'. הַמְבָרֵךְ אֶת עַמּו יִשרָאֵל בַּשָּׁלום:

סיום עמידה

אֱלקַי. נְצר לְשׁונִי מֵרָע וּשפָתַי מִדַּבֵּר מִרְמָה. וְלִמְקַלְלַי נַפְשִׁי תִדּם. וְנַפְשִׁי כֶּעָפָר לַכּל תִּהְיֶה. פְּתַח לִבִּי בְּתורָתֶךָ. וּבְמִצְותֶיךָ תִּרְדּף נַפְשִׁי. וְכָל הַחושְׁבִים עָלַי רָעָה. מְהֵרָה הָפֵר עֲצָתָם וְקַלְקֵל מַחֲשַׁבְתָּם:

וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עולָם וּכְשָׁנִים קַדְמונִיות:

וְעָרְבָה לה' מִנְחַת יְהוּדָה וִירוּשָׁלָיִם. כִּימֵי עולָם וּכְשָׁנִים קַדְמונִיות:

המילים והניקוד מתוך ספריא, מילה במילה: סידור אשכנז, מהדורת "דעת" (נחלת הכלל). בתצוגה בלבד: בלי טעמים, ושמות הקודש בכינוי.

אחרי התפילה: תהילים לפי ימי השבוע, זמני היום ותפילת הדרך.

בתפריט הדפדפן: "הוספה למסך הבית", והדף נפתח כמו אפליקציה.

לשלוח לחבר בווטסאפ

מוסף חול המועד מלא לקריאה בטלפון

מוסף של חול המועד הוא תפילת העמידה הנוספת שאומרים אחרי שחרית בימי חול המועד של פסח ושל סוכות. כאן הוא כתוב במלואו, בניקוד מלא ובאות גדולה, בדף אחד רצוף, בלי פרסומות באמצע התפילה ובלי חלונות קופצים.

הדף שלפניכם הוא בנוסח אשכנז. בכפתורים שבראש הדף עוברים לנוסח עדות המזרח, לנוסח ספרד או לנוסח אשכנז. הטקסט הוא תפילת המוסף לשלוש רגלים, כמו שהיא כתובה בסידורים, ולכן יש בו הוראות שמבדילות בין יום טוב, שבת וחול המועד. את מה שאומרים בחול המועד מחפשים לפי ההוראות באותיות הקטנות.

איך משתמשים בדף

שורת הקיצורים שמתחת לכותרת מובילה ישר לחלק שצריך: קדושה, קדושת היום, ברכת כהנים. בכפתורים א+ וא- משנים את גודל האות, והירח מעביר לרקע כהה. ההוראות ("בחול המועד אומרים", "בשבת מוסיפים") כתובות באות קטנה ובצבע אחר, כדי שלא יתערבבו עם מילות התפילה.

סדר התפילה בקצרה

מוסף היא תפילת עמידה של שבע ברכות: שלוש ברכות שבח בהתחלה, ברכת קדושת היום באמצע, ושלוש ברכות הודאה בסוף. בקדושת היום של חול המועד מזכירים את קרבנות המוסף של החג ואת מה שייחודי לפסח או לסוכות.

מתי אומרים מוסף חול המועד

אחרי שחרית, ובדרך כלל אחרי ההלל וקריאת התורה של חול המועד. זמן התפילה תלוי במקום ובתאריך. בשאלה למעשה שואלים רב.

פסח וסוכות

בחול המועד פסח ובחול המועד סוכות יש שוני בקרבנות היום, וההוראות בטקסט מפרטות אותו. בסוכות יש שוני בין ימי חול המועד לפי היום. מי שרוצה לדעת מה בדיוק אומרים ביום מסוים בוחר לפי ההוראה, ובספק שואלים רב.

מאיפה הטקסט

המילים והניקוד נלקחו מספריא, מילה במילה, מתוך סידורים שמותר להפיץ. לא השלמנו שום מילה מהזיכרון. בתצוגה בלבד הורדנו את טעמי המקרא וכתבנו את שמות הקודש בכינוי. הדף אינו תחליף לפסיקת רב.

למה בנינו את זה

הכלים של גילעד רחמים הם מיזם חסד: הכול בחינם, בלי הגבלה, בלי הרשמה ובלי פרסומות. רצינו שמי שבטיול של חול המועד, בדרך או בבית, יפתח את הטלפון ויתפלל בנחת.

כלים קרובים

תפילת שחרית, תפילת מנחה, תפילת ערבית, מוסף ראש חודש, מוסף חול המועד, נשמת כל חי, ברכת המזון, תפילת הדרך, קריאת שמע על המיטה, ספר תהילים המלא, זמני היום, לוח שנה עברי ומפת האתר.

שאלות ותשובות

מה זה מוסף חול המועד?

תפילת עמידה נוספת, אחרי שחרית, שאומרים בכל יום מימי חול המועד של פסח ושל סוכות. היא כוללת את קדושת היום של החג עם הזכרת קרבנות המוסף.

האם זה אותו מוסף כמו ביום טוב?

הנוסח דומה, והסידורים מביאים אותו יחד, כמו שמופיע כאן, עם הוראות שמבדילות בין יום טוב לחול המועד. מה שאומרים בחול המועד מופיע לפי ההוראות באותיות הקטנות.

אומרים מוסף חול המועד גם בשבת?

כן, בשבת של חול המועד אומרים מוסף עם התוספות לשבת, כמו שהן כתובות בהוראות בטקסט.

איזה נוסחים יש כאן?

שלושה: נוסח עדות המזרח, נוסח ספרד ונוסח אשכנז. לכל נוסח יש דף משלו, ועוברים ביניהם בכפתורים שבראש הדף. כל אחד מתפלל בנוסח של מנהג אבותיו.

מאיפה הטקסט, ואפשר לסמוך עליו?

המילים והניקוד מגיעים מספריא, מילה במילה, מתוך סידורים שמותר להפיץ: מהדורת שלייסבו לעדות המזרח, סידור "מצודה" ו"תורת אמת" לספרד, ו"מצודה" ו"דעת" לאשכנז. לא הוספנו ולא השלמנו שום מילה מהזיכרון. השינוי היחיד הוא בתצוגה: בלי טעמי המקרא, ושמות הקודש כתובים בכינוי, כדי שיהיה מותר לקרוא גם ממסך. מי שמצא טעות מוזמן לכתוב לנו.

איך קוראים בנוחות מהטלפון?

בכפתורים א+ וא- מגדילים ומקטינים את האות, בירח עוברים לרקע כהה, ובשורת הקיצורים קופצים ישר לחלק שצריך. המסך נשאר דלוק בזמן הקריאה (בדפדפנים שתומכים בזה). אין פרסומות, אין הרשמה ואין הגבלה.
