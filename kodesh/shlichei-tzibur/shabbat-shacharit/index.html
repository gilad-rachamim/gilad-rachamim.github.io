תפילת שחרית של שבת נוסח מרוקו

הדף נועד לשליח ציבור בשבת משפחתית במלון או בצימר שאין בהם סידורים בנוסח מרוקו. הנוסח מוקלד מילה במילה מסידור כמנהג מרוקו.

[המשך מי שברך לחיילי צה"ל מעמ' 457] יַדְבֵּר שׂוֹנְאֵינוּ תַּחְתֵּיהֶם וִיעַטְּרֵם בְּכֶתֶר יְשׁוּעָה וּבַעֲטֶרֶת נִצָּחוֹן. וִיקֻיַּם בָּהֶם הַכָּתוּב: כִּי יְיָ אֱלֹהֵיכֶם הַהֹלֵךְ עִמָּכֶם לְהִלָּחֵם לָכֶם עִם־אֹיְבֵיכֶם לְהוֹשִׁיעַ אֶתְכֶם. וְנֹאמַר אָמֵן:

תְּפִלָּה לִשְׁלוֹם הַמְּדִינָה

[הוראה בסידור] אמרו חכמינו ז"ל (אבות ג, ב): "רבי חנינא סגן הכהנים אומר, הוי מתפלל בשלומה של מלכות, שאלמלא מוראה, איש את רעהו חיים בלעו". לפיכך נהגו ישראל במרוצת הדורות לברך את המלך ולהתפלל שהקב"ה יתן בלבו לעשות טובה עם ישראל (ר"ד אבודרהם). וכן במרוקו היו אומרים בפתיחת ההיכל תפילת "הַנּוֹתֵן תְּשׁוּעָה לַמְּלָכִים" (בהמשך העמוד). בקום המדינה תיקנו הרבנים הראשיים דאז הגר"י הרצוג והגרב"צ עוזיאל זצ"ל בקשה זו:

אָבִינוּ שֶׁבַּשָּׁמַיִם, צוּר יִשְׂרָאֵל וְגוֹאֲלוֹ, בָּרֵךְ אֶת מְדִינַת יִשְׂרָאֵל, רֵאשִׁית צְמִיחַת גְּאֻלָּתֵנוּ. הָגֵן עָלֶיהָ בְּאֶבְרַת חַסְדֶּךָ, וּפְרֹשׂ עָלֶיהָ סֻכַּת שְׁלוֹמֶךָ, וּשְׁלַח אוֹרְךָ וַאֲמִתְּךָ לְרָאשֶׁיהָ, שָׂרֶיהָ וְיוֹעֲצֶיהָ, וְתַקְּנֵם בְּעֵצָה טוֹבָה מִלְּפָנֶיךָ. חַזֵּק אֶת יְדֵי מְגִנֵּי אֶרֶץ קָדְשֵׁנוּ, וְהַנְחִילֵם אֱלֹהֵינוּ יְשׁוּעָה וַעֲטֶרֶת נִצָּחוֹן תְּעַטְּרֵם, וְנָתַתָּ שָׁלוֹם בָּאָרֶץ וְשִׂמְחַת עוֹלָם לְיוֹשְׁבֶיהָ. וְאֶת אַחֵינוּ כָּל־בֵּית יִשְׂרָאֵל, פְּקָד־נָא בְּכָל־אַרְצוֹת פְּזוּרֵיהֶם, וְתוֹלִיכֵם מְהֵרָה קוֹמְמִיּוּת לְצִיּוֹן עִירֶךָ וְלִירוּשָׁלַיִם מִשְׁכַּן שְׁמֶךָ, כַּכָּתוּב בְּתוֹרַת מֹשֶׁה עַבְדֶּךָ: אִם־יִהְיֶה נִדַּחֲךָ בִּקְצֵה הַשָּׁמַיִם, מִשָּׁם יְקַבֶּצְךָ יְיָ אֱלֹהֶיךָ וּמִשָּׁם יִקָּחֶךָ: וֶהֱבִיאֲךָ יְיָ אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר יָרְשׁוּ־אֲבֹתֶיךָ וִירִשְׁתָּהּ, וְהֵיטִבְךָ וְהִרְבְּךָ מֵאֲבֹתֶיךָ: וְיַחֵד לְבָבֵנוּ לְאַהֲבָה וּלְיִרְאָה אֶת שְׁמֶךָ, וְלִשְׁמֹר אֶת כָּל־דִּבְרֵי תוֹרָתֶךָ, וּשְׁלַח לָנוּ מְהֵרָה בֶּן דָּוִד מְשִׁיחַ צִדְקֶךָ, לִפְדּוֹת מְחַכֵּי קֵץ יְשׁוּעָתֶךָ. הוֹפַע בַּהֲדַר גְּאוֹן עֻזֶּךָ עַל כָּל־יוֹשְׁבֵי תֵבֵל אַרְצֶךָ, וְיֹאמַר כֹּל אֲשֶׁר נְשָׁמָה בְּאַפּוֹ: יְיָ אֱלֹהֵי יִשְׂרָאֵל מֶלֶךְ, וּמַלְכוּתוֹ בַּכֹּל מָשָׁלָה, אָמֵן סֶלָה.

[הוראה בסידור] בערי המערב נהגו מימים קדמונים בשבתות ובימים טובים, בעת פתיחת ההיכל לברך את המלך או הנשיא המקומי, אשר אפשרו ליהודים לנהל חייהם כהלכה:

הַנּוֹתֵן תְּשׁוּעָה לַמְּלָכִים וּמֶמְשָׁלָה לַנְּסִיכִים, וּמַלְכוּתוֹ מַלְכוּת כָּל־עוֹלָמִים, הַפּוֹצֶה אֶת דָּוִד עַבְדּוֹ מֵחֶרֶב רָעָה. הַנּוֹתֵן בַּיָּם [המשך בעמוד 459, בצילום 16 עמוד שמאל]

[המשך מעמ' 458] הַנּוֹתֵן בַּיָּם דֶּרֶךְ, וּבְמַיִם עַזִּים נְתִיבָה. הוּא יְבָרֵךְ וְיִשְׁמֹר וְיִנְצֹר וְיַעֲזֹר וִירוֹמֵם וִיגַדֵּל, וִינַשֵּׂא לְמַעְלָה לְמַעְלָה, לַאֲדוֹנֵנוּ הַמֶּלֶךְ וּלְשָׂרֵי מֶמְשַׁלְתּוֹ. מֶלֶךְ מַלְכֵי הַמְּלָכִים בְּרַחֲמָיו יִשְׁמְרֵם וִיחַיֵּם, וּמִכָּל־צָרָה וָנֶזֶק יַצִּילֵם. מֶלֶךְ מַלְכֵי הַמְּלָכִים בְּרַחֲמָיו יָרִים וִיגַבִּיהַּ כּוֹכַב מַעֲרַכְתָּם, וְיַאֲרִיכוּ יָמִים עַל מַמְלַכְתָּם. מֶלֶךְ מַלְכֵי הַמְּלָכִים בְּרַחֲמָיו יִתֵּן בְּלִבָּם וּבְלֵב כָּל־יוֹעֲצֵיהֶם וְשָׂרֵיהֶם רַחֲמָנוּת, לַעֲשׂוֹת טוֹבָה עִמָּנוּ וְעִם כָּל־יִשְׂרָאֵל אַחֵינוּ. בִּימֵיהֶם וּבִימֵינוּ תִּוָּשַׁע יְהוּדָה, וְיִשְׂרָאֵל יִשְׁכֹּן לָבֶטַח, וּבָא לְצִיּוֹן גּוֹאֵל, וְכֵן יְהִי רָצוֹן וְנֹאמַר אָמֵן:

[הוראה בסידור] יש אומרים פסוק זה:

צְאֶנָה וּרְאֶינָה בְּנוֹת צִיּוֹן בַּמֶּלֶךְ שְׁלֹמֹה, בָּעֲטָרָה שֶׁעִטְּרָה־לּוֹ אִמּוֹ בְּיוֹם חֲתֻנָּתוֹ, וּבְיוֹם שִׂמְחַת לִבּוֹ:

[הוראה בסידור] אומר החזן פסוק זה בנעימה, והציבור שבים ואומרים אחריו:

אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ, אַשְׁרֵי הָעָם שֶׁיְיָ אֱלֹהָיו:

[הוראה בסידור] מוציאים את ספר התורה, ואומר החזן לבדו:

גַּדְּלוּ לַייָ אִתִּי, וּנְרוֹמְמָה שְׁמוֹ יַחְדָּו:

[הוראה בסידור] וממשיכים החזן עם הציבור:

רוֹמְמוּ יְיָ אֱלֹהֵינוּ, וְהִשְׁתַּחֲווּ לַהֲדֹם רַגְלָיו, קָדוֹשׁ הוּא: רוֹמְמוּ יְיָ אֱלֹהֵינוּ וְהִשְׁתַּחֲווּ לְהַר קָדְשׁוֹ, כִּי־קָדוֹשׁ יְיָ אֱלֹהֵינוּ: תּוֹרָה צִוָּה־לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: עֵץ־חַיִּים הִיא לַמַּחֲזִיקִים בָּהּ, וְתֹמְכֶיהָ מְאֻשָּׁר: דְּרָכֶיהָ דַרְכֵי־נֹעַם, וְכָל־נְתִיבוֹתֶיהָ [המשך בעמ' 460]

[המשך עמ' 459] שָׁלוֹם: שָׁלוֹם רָב לְאֹהֲבֵי תוֹרָתֶךָ, וְאֵין־לָמוֹ מִכְשׁוֹל: יְיָ עֹז לְעַמּוֹ יִתֵּן, יְיָ יְבָרֵךְ אֶת־עַמּוֹ בַשָּׁלוֹם:

[הוראה בסידור] כשמגיע ספר התורה אל התיבה, נוהגים לומר תפילה זו (מסכת סופרים יד, יב):

יִשְׂמְחוּ הַשָּׁמַיִם וְתָגֵל הָאָרֶץ, וְיֹאמְרוּ בַגּוֹיִם יְיָ מָלָךְ:

יִשְׂמַח הַר־צִיּוֹן תָּגֵלְנָה בְּנוֹת יְהוּדָה, לְמַעַן מִשְׁפָּטֶיךָ:

עַל הַכֹּל יִתְגַּדַּל וְיִתְקַדַּשׁ, יִתְבָּרַךְ וְיִתְרוֹמַם וְיִתְנַשֵּׂא שְׁמוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ־הוּא: בְּעוֹלָמוֹת שֶׁבָּרָא, הָעוֹלָם הַזֶּה וְהָעוֹלָם הַבָּא כִּרְצוֹנוֹ וְכִרְצוֹן יְרֵאָיו וְכִרְצוֹן כָּל־עַמּוֹ בֵּית יִשְׂרָאֵל, תִּגָּלֶה [לאימות] וְתֵרָאֶה מַלְכוּתוֹ עָלֵינוּ בִּמְהֵרָה, וְיָחוֹן וִירַחֵם עַל פְּלֵטָתֵנוּ: וְעַל פְּלֵיטַת עַמּוֹ בֵּית יִשְׂרָאֵל, בְּחֵן בְּחֶסֶד בְּרַחֲמִים וּבְרָצוֹן, וְאִמְרוּ כָל־הָעָם אָמֵן:

הָרַחֲמָן הוּא יְחַזִּיר [לאימות] שְׁבִיתֵנוּ וּשְׁבִית עַמּוֹ בֵּית יִשְׂרָאֵל בִּמְהֵרָה בְּיָמֵינוּ, וְיוֹצִיאֵנוּ מִכָּל־צָרוֹתֵינוּ לִרְוָחָה, וּמֵאֲפֵלָה לְאוֹרָה בְּקָרוֹב, וְאִמְרוּ כָל־הָעָם אָמֵן:

כִּי שֵׁם יְיָ אֶקְרָא, הָבוּ גֹדֶל לֵאלֹהֵינוּ: יְהִי שֵׁם יְיָ מְבֹרָךְ, מֵעַתָּה וְעַד־עוֹלָם: הַכֹּל תְּנוּ עֹז לֵאלֹהִים, וּתְנוּ כָבוֹד לַתּוֹרָה.

[הוראה בסידור] מגביה את ספר התורה ומראה את הכתב לציבור, ואומרים:

וְזֹאת הַתּוֹרָה אֲשֶׁר־שָׂם מֹשֶׁה לִפְנֵי בְּנֵי יִשְׂרָאֵל: תּוֹרָה צִוָּה־לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: אֵל שַׁדַּי אֱמֶת, וּמֹשֶׁה אֱמֶת וְתוֹרָתוֹ אֱמֶת.

[הוראה בסידור] והחזן מסיים ואומר:

הָאֵל תָּמִים דַּרְכּוֹ, אִמְרַת־יְיָ צְרוּפָה, מָגֵן הוּא לְכֹל הַחֹסִים בּוֹ:

[הוראה בסידור] קוראים בספר התורה שבעה עולים בפרשת השבוע, אין פוחתין מהם אך רשאים להוסיף עליהם עולים נוספים (שו"ע רפב, א; קיצור שו"ע [טולידאנו] רסא, א; וראה הלכה 339).

ברכות התורה

העולה אומר: הַשֵּׁם עִמָּכֶם. ועונים: יְבָרֶכְךָ הַשֵּׁם:

העולה: בָּרְכוּ אֶת יְיָ הַמְבֹרָךְ:

הקהל: בָּרוּךְ יְיָ הַמְבֹרָךְ לְעוֹלָם וָעֶד:

וחוזר העולה: בָּרוּךְ יְיָ הַמְבֹרָךְ לְעוֹלָם וָעֶד:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר בָּחַר־בָּנוּ מִכָּל־הָעַמִּים, וְנָתַן לָנוּ אֶת־תּוֹרָתוֹ. בָּרוּךְ אַתָּה יְיָ, נוֹתֵן הַתּוֹרָה:

אחר קריאת התורה אומר העולה: אֱמֶת תּוֹרָתֵנוּ הַקְּדוֹשָׁה:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר נָתַן־לָנוּ [אֶת־תּוֹרָתוֹ] תּוֹרַת אֱמֶת, וְחַיֵּי עוֹלָם נָטַע בְּתוֹכֵנוּ. בָּרוּךְ אַתָּה יְיָ, נוֹתֵן הַתּוֹרָה:

[הוראה בסידור] בסיום חומש מחמשת החומשים, אחר חצי קדיש שלאחר עליית שביעי, אומרים הקהל:

חִזְקוּ וְיַאֲמֵץ לְבַבְכֶם, כָּל־הַמְיַחֲלִים לַיְיָ:

[הוראה בסידור] בסיום הקללות בחוקותי וכי תבוא, אחר ברכה אחרונה אומר החזן בנעימת משלי:

מוּסַר יְיָ בְּנִי אַל־תִּמְאָס, וְאַל־תָּקֹץ בְּתוֹכַחְתּוֹ: כִּי אֶת אֲשֶׁר יֶאֱהַב יְיָ יוֹכִיחַ, וּכְאָב אֶת־בֵּן יִרְצֶה: וְלַמּוֹכִיחִים יִנְעָם, וַעֲלֵיהֶם תָּבוֹא בִּרְכַּת טוֹב.

בִּרְכַּת הַגּוֹמֵל

[הוראה בסידור] כאשר היה בסכנה וניצל (ראה הלכה 223), יברך 'הגומל':

הַלְלוּיָהּ | אוֹדֶה יְיָ בְּכָל־לֵבָב, בְּסוֹד יְשָׁרִים וְעֵדָה: הַדּוּ לַיְיָ כִּי־טוֹב, כִּי לְעוֹלָם חַסְדּוֹ: יוֹדוּ לַיְיָ חַסְדּוֹ, וְנִפְלְאוֹתָיו לִבְנֵי אָדָם:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, הַגּוֹמֵל לַחַיָּבִים טוֹבוֹת שֶׁגְּמָלַנִי כָּל־טוֹב:

[הוראה בסידור] והקהל עונים 'אמן', ואומרים:

הָאֵל אֲשֶׁר גְּמָלְךָ כָּל־טוֹב וְחֵן וְחֶסֶד וְחֶמְלָה, יִתְבָּרַךְ וְיִתְרוֹמַם עַל כָּל־בְּרָכָה וּתְהִלָּה, הוּא בְּרַחֲמָיו יִשְׁמְרְךָ וְיִגְמְלְךָ לָעַד כָּל־טוֹב, סֶלָה.

[הוראה בסידור] בסיום קריאת התורה אומרים חצי קדיש. והמנהג הנפוץ שהעולה השביעי אומר קדיש זה בסיום ברכתו האחרונה.

יִתְגַּדַּל וְיִתְקַדַּשׁ שְׁמֵיהּ רַבָּא (עונים: אמן). בְּעָלְמָא דִּי בְרָא כִּרְעוּתֵהּ, וְיַמְלִיךְ מַלְכוּתֵהּ, וְיַצְמַח פֻּרְקָנֵהּ וִיקָרֵב מְשִׁיחֵהּ (עונים: אמן). בְּחַיֵּיכוֹן וּבְיוֹמֵיכוֹן וּבְחַיֵּי דְכָל־בֵּית יִשְׂרָאֵל, בַּעֲגָלָא וּבִזְמַן קָרִיב, וְאִמְרוּ אָמֵן: (עונים: אמן – 'יְהֵא שְׁמֵיהּ רַבָּא' עד 'דַּאֲמִירָן בְּעָלְמָא')

יְהֵא שְׁמֵיהּ רַבָּא מְבָרַךְ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא יִתְבָּרַךְ, וְיִשְׁתַּבַּח, וְיִתְפָּאַר, וְיִתְרוֹמַם, וְיִתְנַשֵּׂא, וְיִתְהַדָּר, וְיִתְעַלֶּה וְיִתְהַלָּל שְׁמֵיהּ דְּקֻדְשָׁא בְּרִיךְ־הוּא (עונים: אמן) לְעֵלָּא מִן כָּל־בִּרְכָתָא, שִׁירָתָא, תֻּשְׁבְּחָתָא וְנֶחֱמָתָא, דַּאֲמִירָן בְּעָלְמָא, וְאִמְרוּ אָמֵן (עונים: אמן).

[הוראה בסידור] נוסח 'מי שברך' לעולה לתורה, לחתן, לחולים, וכן סדר ההשכבות – להלן בסוף הסידור, בקונטרס 'שער הגבאי' (עמ' 1250 והלאה).

[הוראה בסידור] בשבת חתן, לאחר שעלה החתן לתורה ובירך ברכה אחרונה, נוהגים לקרוא לו פרשת 'וְאַבְרָהָם זָקֵן', כל פסוק מקרא ותרגומו (בעמ' 1253).

[הוראה בסידור] קודם עליית המפטיר, נוהגים כל הציבור לשורר פיוט זה בנעימה:

קְדוֹשַׁת שַׁבָּת כְּתַקָּנַת עֶזְרָא, יַעֲלוּ שִׁבְעָה לְסֵפֶר תּוֹרָה.

וּמַפְטִיר יַשְׁלִים הַהַפְטָרָה, וְיִתְפַּלְלוּ מוּסָף כְּהִלְכָתוֹ.

כִּי בוֹ, כִּי בוֹ שָׁבַת מִכָּל־מְלַאכְתּוֹ, וִיבָרֵךְ עַל סֵפֶר תּוֹרָתוֹ:

בִּרְכוֹת הַהַפְטָרָה

[הוראה בסידור] לפני קריאת ההפטרה מברך המפטיר ברכה זו:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר בָּחַר בִּנְבִיאִים טוֹבִים, וְרָצָה בְדִבְרֵיהֶם הַנֶּאֱמָרִים בֶּאֱמֶת: בָּרוּךְ אַתָּה יְיָ, הַבּוֹחֵר בַּתּוֹרָה וּבְמֹשֶׁה עַבְדּוֹ וּבְיִשְׂרָאֵל עַמּוֹ וּבִנְבִיאֵי הָאֱמֶת וָצֶדֶק:

[הוראה בסידור] לפני שיתחיל בקריאת ההפטרה, יש נוהגים להוסיף פסוק זה:

וָאָשִׂם דְּבָרַי בְּפִיךָ, וּבְצֵל יָדִי כִּסִּיתִיךָ, לִנְטֹעַ שָׁמַיִם וְלִיסֹד אָרֶץ, וְלֵאמֹר לְצִיּוֹן עַמִּי־אָתָּה:

[הוראה בסידור] וקורא המפטיר את ההפטרה לבדו. ואת הפסוק האחרון נוהגים שהציבור קוראו בקול רם, והמפטיר שב וקוראו.

[הוראה בסידור] בסיום קריאת ההפטרה, יאמר המפטיר פסוק זה, לפני הברכה:

גֹּאֲלֵנוּ יְיָ צְבָאוֹת שְׁמוֹ, קְדוֹשׁ יִשְׂרָאֵל:

בָּרוּךְ אַתָּה יְיָ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, צוּר כָּל־הָעוֹלָמִים, צַדִּיק בְּכָל־הַדּוֹרוֹת, הָאֵל הַנֶּאֱמָן, הָאוֹמֵר וְעֹשֶׂה, הַמְדַבֵּר וּמְקַיֵּם, כִּי־כָל־דְּבָרָיו אֱמֶת וָצֶדֶק. נֶאֱמָן אַתָּה הוּא יְיָ אֱלֹהֵינוּ וְנֶאֱמָנִים דְּבָרֶיךָ, וְדָבָר אֶחָד מִדְּבָרֶיךָ אָחוֹר לֹא־יָשׁוּב רֵיקָם, כִּי אֵל מֶלֶךְ נֶאֱמָן וְרַחֲמָן אָתָּה. בָּרוּךְ אַתָּה יְיָ, הָאֵל הַנֶּאֱמָן בְּכָל־דְּבָרָיו:

רַחֵם עַל צִיּוֹן כִּי הִיא בֵּית־חַיֵּינוּ, וְלַעֲלוּבַת נֶפֶשׁ תּוֹשִׁיעַ [וּתְשַׂמַּח] בִּמְהֵרָה בְיָמֵינוּ. בָּרוּךְ אַתָּה יְיָ, מְשַׂמֵּחַ צִיּוֹן בְּבָנֶיהָ:

שַׂמְּחֵנוּ יְיָ אֱלֹהֵינוּ בְּאֵלִיָּהוּ הַנָּבִיא עַבְדֶּךָ, וּבְמַלְכוּת בֵּית־דָּוִד מְשִׁיחֶךָ, בִּמְהֵרָה יָבֹא וְיָגֵל לִבֵּנוּ. עַל־כִּסְאוֹ לֹא־יֵשֵׁב זָר, וְלֹא יִנְחֲלוּ עוֹד אֲחֵרִים אֶת־כְּבוֹדוֹ, כִּי בְשֵׁם קָדְשְׁךָ נִשְׁבַּעְתָּ לּוֹ שֶׁלֹּא יִכְבֶּה נֵרוֹ לְעוֹלָם וָעֶד. בָּרוּךְ אַתָּה יְיָ, מָגֵן דָּוִד:

עַל־הַתּוֹרָה וְעַל־הָעֲבוֹדָה וְעַל־הַנְּבִיאִים וְעַל־יוֹם הַשַּׁבָּת הַזֶּה, שֶׁנָּתַתָּ לָּנוּ יְיָ אֱלֹהֵינוּ, לִקְדֻשָּׁה וְלִמְנוּחָה, לְכָבוֹד וּלְתִפְאָרֶת. עַל־הַכֹּל יְיָ אֱלֹהֵינוּ אָנוּ מוֹדִים לָךְ וּמְבָרְכִים אוֹתָךְ, יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל־חַי תָּמִיד לְעוֹלָם וָעֶד. בָּרוּךְ אַתָּה יְיָ, מְקַדֵּשׁ הַשַּׁבָּת:

[הוראה בסידור] ואין המברך עונה 'אמן' אחר ברכת עצמו (ראה הלכה 347).

תְּפִלַּת מוּסָף לְשַׁבָּת

[יְהִי־חַסְדְּךָ יְיָ עָלֵינוּ, כַּאֲשֶׁר יִחַלְנוּ לָךְ:]

אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ, עוֹד יְהַלְלוּךָ סֶּלָה: אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ, אַשְׁרֵי הָעָם שֶׁיְיָ אֱלֹהָיו:

תְּהִלָּה לְדָוִד, אֲרוֹמִמְךָ אֱלוֹהַי הַמֶּלֶךְ, וַאֲבָרְכָה שִׁמְךָ לְעוֹלָם וָעֶד: בְּכָל־יוֹם אֲבָרְכֶךָּ, וַאֲהַלְלָה שִׁמְךָ לְעוֹלָם וָעֶד: גָּדוֹל יְיָ וּמְהֻלָּל מְאֹד, וְלִגְדֻלָּתוֹ אֵין חֵקֶר: דּוֹר לְדוֹר יְשַׁבַּח מַעֲשֶׂיךָ, וּגְבוּרֹתֶיךָ יַגִּידוּ: הֲדַר כְּבוֹד הוֹדֶךָ, וְדִבְרֵי נִפְלְאֹתֶיךָ אָשִׂיחָה: וֶעֱזוּז נוֹרְאֹתֶיךָ יֹאמֵרוּ, וּגְדֻלָּתְךָ אֲסַפְּרֶנָּה: זֵכֶר רַב־טוּבְךָ יַבִּיעוּ, וְצִדְקָתְךָ יְרַנֵּנוּ: חַנּוּן וְרַחוּם יְיָ, אֶרֶךְ אַפַּיִם וּגְדָל־חָסֶד: טוֹב־יְיָ לַכֹּל, וְרַחֲמָיו עַל־כָּל־מַעֲשָׂיו: יוֹדוּךָ יְיָ כָּל־מַעֲשֶׂיךָ, וַחֲסִידֶיךָ יְבָרְכוּכָה: כְּבוֹד מַלְכוּתְךָ יֹאמֵרוּ, [המשך עמ' 470: וּגְבוּרָתְךָ יְדַבֵּרוּ…]

אשרי (המשך מעמ' 469)

וּגְבוּרָתְךָ יְדַבֵּרוּ: לְהוֹדִיעַ | לִבְנֵי הָאָדָם גְּבוּרֹתָיו, וּכְבוֹד הֲדַר מַלְכוּתוֹ: מַלְכוּתְךָ מַלְכוּת כָּל־עֹלָמִים, וּמֶמְשַׁלְתְּךָ בְּכָל־דּוֹר וָדֹר: סוֹמֵךְ יְיָ לְכָל־הַנֹּפְלִים, וְזוֹקֵף לְכָל־הַכְּפוּפִים: עֵינֵי־כֹל אֵלֶיךָ יְשַׂבֵּרוּ, וְאַתָּה נוֹתֵן־לָהֶם אֶת־אָכְלָם בְּעִתּוֹ:

[הוראה בסידור] פורשים ידיים כלפי מעלה, ואומרים בכוונה:

[הוראה בסידור] מלווים את ספר התורה להיכל באמירת מזמור זה:

מִזְמוֹר לְדָוִד, הָבוּ לַיְיָ בְּנֵי אֵלִים, הָבוּ לַיְיָ כָּבוֹד וָעֹז: הָבוּ לַיְיָ כְּבוֹד שְׁמוֹ, הִשְׁתַּחֲווּ לַיְיָ בְּהַדְרַת־קֹדֶשׁ: קוֹל יְיָ עַל־הַמָּיִם, אֵל־הַכָּבוֹד הִרְעִים, יְיָ עַל־מַיִם רַבִּים: קוֹל־יְיָ בַּכֹּחַ, קוֹל יְיָ בֶּהָדָר: קוֹל יְיָ שֹׁבֵר אֲרָזִים, וַיְשַׁבֵּר יְיָ אֶת־אַרְזֵי הַלְּבָנוֹן: וַיַּרְקִידֵם כְּמוֹ־עֵגֶל, לְבָנוֹן וְשִׂרְיֹן כְּמוֹ בֶן־רְאֵמִים: קוֹל־יְיָ חֹצֵב לַהֲבוֹת אֵשׁ: קוֹל יְיָ יָחִיל מִדְבָּר, יָחִיל יְיָ מִדְבַּר קָדֵשׁ: קוֹל יְיָ יְחוֹלֵל אַיָּלוֹת וַיֶּחֱשֹׂף יְעָרוֹת, וּבְהֵיכָלוֹ כֻּלּוֹ אֹמֵר כָּבוֹד: יְיָ לַמַּבּוּל יָשָׁב, וַיֵּשֶׁב יְיָ מֶלֶךְ לְעוֹלָם: יְיָ עֹז לְעַמּוֹ יִתֵּן, יְיָ | יְבָרֵךְ אֶת־עַמּוֹ בַשָּׁלוֹם:

מהו נוסח מרוקו בתפילת שבת?

איפה פתיחת ההיכל?

בדף נפרד: פתיחת ההיכל בשבת, מ'בריך שמיה'.

מי שברך לקהל וזכרון נשמות?

בדף מי שברך נוסח מרוקאי, עם קישור לכלי שממלא שמות.

אפשר להדפיס את הדף?

כן. יש גרסת הדפסה נקיה, בלי תפריטים ובאותיות גדולות, כדי להחזיק ביד ליד התיבה.

הטקסט מנוקד?

כן, כל התפילות מנוקדות כפי שבסידור. מקום שלא ברור בצילום מסומן 'לאימות' ולא מנוחש.

האם צריך להירשם?

לא. הדף בחינם, בלי הרשמה ובלי פרסומות, כמיזם חסד.
