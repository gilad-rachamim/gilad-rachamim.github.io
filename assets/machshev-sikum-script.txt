function runDigest() {
  var props = PropertiesService.getUserProperties();
  var now = new Date();
  var last = Number(props.getProperty('lastRun')) || 0;
  var defDays = CONFIG.freq === 'weekly' ? 7 : CONFIG.freq === 'every' ? (Number(CONFIG.everyDays) || 3) : 1;
  var since = last ? new Date(last) : new Date(now.getTime() - defDays * 864e5);
  var maxBack = new Date(now.getTime() - 14 * 864e5);
  if (since < maxBack) since = maxBack;
  var after = Math.floor(since.getTime() / 1000);

  processUnderstood_();
  var me = meAddresses_();
  var data = classify_(after, me, now);
  var to = (CONFIG.to || '').trim() || Session.getActiveUser().getEmail() || Session.getEffectiveUser().getEmail();

  var total = data.total;
  if (!(CONFIG.skipEmpty && countShown_(data) === 0)) {
    var html = renderHtml_(data, since, now);
    var plain = renderPlain_(data, since, now);
    GmailApp.sendEmail(to, subject_(data, now), plain, { htmlBody: html, name: 'סיכום המיילים שלי' });
  }
  if (CONFIG.labels) applyLabels_(data);
  if (CONFIG.tasks) addTasks_(data);
  if (CONFIG.archivePromo) archivePromo_(data);
  props.setProperty('lastRun', String(now.getTime()));
  return { total: total, to: to };
}

function meAddresses_() {
  var list = [Session.getEffectiveUser().getEmail()];
  try { list = list.concat(GmailApp.getAliases()); } catch (e) {}
  var set = {};
  list.forEach(function (a) { if (a) set[String(a).toLowerCase()] = 1; });
  return set;
}

function addrOf_(s) {
  var m = String(s || '').match(/<([^>]+)>/);
  return (m ? m[1] : String(s || '')).trim().toLowerCase();
}
function nameOf_(s) {
  s = String(s || '');
  var m = s.match(/^\s*"?([^"<]+?)"?\s*</);
  return (m ? m[1] : s.replace(/<.*>/, '')).trim() || addrOf_(s);
}
function hasWord_(text, words) {
  text = String(text || '').toLowerCase();
  for (var i = 0; i < (words || []).length; i++) {
    var w = String(words[i] || '').trim().toLowerCase();
    if (w && text.indexOf(w) >= 0) return words[i];
  }
  return '';
}
function idSet_(q) {
  var set = {};
  try { GmailApp.search(q, 0, MAX_THREADS).forEach(function (t) { set[t.getId()] = 1; }); } catch (e) {}
  return set;
}
function isVip_(addr) {
  var a = String(addr || '').toLowerCase();
  return (CONFIG.vip || []).some(function (v) {
    v = String(v || '').trim().toLowerCase();
    if (!v) return false;
    if (v.charAt(0) === '@') return a.slice(-v.length) === v;
    return v.indexOf('@') > 0 ? a === v : a.indexOf(v) >= 0;
  });
}
// סימנים של דיוור בגוף המייל עצמו (בלי הציטוט של מה שאתם כתבתם קודם)
var AD_MARKS = /(להסרה|להסרת|הסרה מרשימ|הסר אותי|\(פרסומת\)|פרסומת|לצפי[יה]+ בדפדפן|unsubscribe|view (this email )?in (a |your )?browser|manage (your )?preferences|email preferences)/i;
function isBulk_(msg, addr, top) {
  if (/(no-?reply|do-?not-?reply|newsletter|mailer-daemon|notification|news@|marketing@|promo|recommendation@|digest@|updates?@)/i.test(addr)) return true;
  try { if (msg.getHeader('List-Unsubscribe') || msg.getHeader('List-Id')) return true; } catch (e) {}
  try { if (/bulk|list/i.test(msg.getHeader('Precedence') || '')) return true; } catch (e) {}
  if (top && AD_MARKS.test(top)) return true;
  return false;
}
/** גוף המייל בלי ציטוטים של מיילים קודמים (טקסט מלא, עד 4000 תווים). */
function bodyTop_(msg) {
  var t = '';
  try { t = msg.getPlainBody() || ''; } catch (e) {}
  t = t.split(/\n(?:>|On .+wrote:|ב-.+כתב|ב.+ כתב\/ה:|בתאריך .+ מאת|-----Original|________________)/)[0];
  return t.slice(0, 4000);
}
function snippet_(msg, top) {
  var t = top !== undefined ? top : bodyTop_(msg);
  t = t.replace(/https?:\/\/\S+/g, '').replace(/\s+/g, ' ').trim();
  return t.length > 160 ? t.slice(0, 157) + '…' : t;
}
/** הסכום בחשבונית/קבלה: קודם ליד "סה"כ / Amount paid / Total", אחרת הסכום הראשון. */
function amountOf_(text) {
  text = String(text || '').replace(/ /g, ' ');
  var num = '(\\d{1,3}(?:,\\d{3})+(?:\\.\\d{1,2})?|\\d+(?:\\.\\d{1,2})?)';
  var cur = '(\\$|₪|€|£|USD|ILS|EUR|ש"ח|ש״ח|שח)';
  var re1 = new RegExp(cur + '\\s?' + num), re2 = new RegExp(num + '\\s?' + cur);
  function pick(s) {
    var m = s.match(re1), c, v;
    var m2 = s.match(re2);
    if (m && (!m2 || m.index <= m2.index)) { c = m[1]; v = m[2]; } else if (m2) { c = m2[2]; v = m2[1]; } else return null;
    c = { 'USD': '$', 'ILS': '₪', 'EUR': '€', 'ש"ח': '₪', 'ש״ח': '₪', 'שח': '₪' }[c] || c;
    var val = parseFloat(v.replace(/,/g, ''));
    return isNaN(val) || val <= 0 ? null : { cur: c, val: val };
  }
  var key = /(amount paid|amount due|total|סה["״]?כ|סכום לתשלום|סכום|לתשלום|שולם)/ig, k;
  while ((k = key.exec(text))) { var a = pick(text.slice(k.index, k.index + 60)); if (a) return a; }
  return pick(text);
}
/** נושא בלי Re:/Fwd: – כדי לאחד את אותו מייל שהגיע בכמה שיחות. */
function subjKey_(s) {
  return String(s || '').replace(/^\s*((re|fw|fwd|השב|הועבר)\s*:\s*)+/i, '').replace(/\s+/g, ' ').trim().toLowerCase();
}
function ago_(d, now) {
  var h = Math.floor((now - d) / 36e5);
  if (h < 1) return 'עכשיו';
  if (h < 24) return 'לפני ' + h + ' שע׳';
  var dd = Math.floor(h / 24);
  return dd === 1 ? 'אתמול' : 'לפני ' + dd + ' ימים';
}

function classify_(after, me, now) {
  var out = { total: 0, promoSenders: {}, promoCount: 0 };
  Object.keys(CATS).forEach(function (k) { out[k] = []; });
  var q = 'in:inbox after:' + after + ' -label:' + labelQuery_(understoodName_());
  var threads = GmailApp.search(q, 0, MAX_THREADS);
  out.total = threads.length;
  var promoIds = idSet_(q + ' (category:promotions OR category:social)');
  var updatesIds = idSet_(q + ' category:updates');
  var importantIds = idSet_(q + ' is:important');
  // קריאה אחת לכל ההודעות של כל השיחות (מהיר בהרבה מקריאה לכל שיחה)
  var allMsgs = GmailApp.getMessagesForThreads(threads);

  threads.forEach(function (t, ti) {
    var msgs = allMsgs[ti] || t.getMessages();
    var lastMsg = msgs[msgs.length - 1];
    // שיחה שכולה ממני (התראות ששלחתי לעצמי, והסיכום עצמו) – לא נכנסת לסיכום
    if (msgs.every(function (x) { return me[addrOf_(x.getFrom())]; })) { out.total--; return; }
    var lastFrom = addrOf_(lastMsg.getFrom());
    var fromMe = !!me[lastFrom];
    // ההודעה האחרונה שלא ממני
    var m = lastMsg;
    for (var i = msgs.length - 1; i >= 0; i--) { if (!me[addrOf_(msgs[i].getFrom())]) { m = msgs[i]; break; } }
    var from = m.getFrom(), addr = addrOf_(from);
    var subj = t.getFirstMessageSubject() || '(בלי נושא)';
    var top = bodyTop_(m);
    var body = snippet_(m, top);
    var text = subj + ' ' + body;
    var item = {
      id: t.getId(), subj: subj, who: nameOf_(from), addr: addr, snip: body,
      date: m.getDate(), unread: t.isUnread(), thread: t, why: ''
    };
    var id = t.getId();
    var promoTab = !!promoIds[id];
    var bulk = promoTab || isBulk_(m, addr, top);
    var toMe = false;
    try { toMe = String(m.getTo() || '').toLowerCase().split(',').some(function (x) { return me[addrOf_(x)]; }); } catch (e) {}

    // כללים אישיים: "מילה או כתובת -> תווית"
    (CONFIG.rules || []).forEach(function (r) {
      if (r && r.match && r.label && (addr.indexOf(String(r.match).toLowerCase()) >= 0 || hasWord_(text, [r.match]))) {
        (item.rules = item.rules || []).push(r.label);
      }
    });

    var w;
    // אזהרה של שירות (חשבון יושעה, תשלום נכשל, API נסגר) – חשוב גם כשזה noreply.
    // לא ממיילים בלשונית קידומי מכירות ולא ממייל שכתוב בו "פרסומת".
    if (CONFIG.on.important && !fromMe && !promoTab && !/פרסומת/.test(subj) &&
        (w = hasWord_(subj + ' ' + top.slice(0, 1500), CONFIG.alertWords))) {
      item.why = 'אזהרה: ' + w; item.alert = true; out.important.push(item); return;
    }
    // ענית אחרון בשיחה – אין מה לענות (אם לא ענו לך, זה יופיע ב"כתבת ולא ענו לך")
    if (fromMe) { out.other = out.other || []; out.other.push(item); return; }

    if (bulk && !isVip_(addr)) {
      if (CONFIG.on.money && hasWord_(text, CONFIG.moneyWords)) { item.why = hasWord_(text, CONFIG.moneyWords); item.amount = amountOf_(subj + ' ' + top); out.money.push(item); return; }
      if (CONFIG.on.ship && hasWord_(text, CONFIG.shipWords)) { item.why = hasWord_(text, CONFIG.shipWords); out.ship.push(item); return; }
      out.promoCount++;
      var key = item.who || addr;
      out.promoSenders[key] = (out.promoSenders[key] || 0) + 1;
      out.promo.push(item);
      return;
    }
    var sys = !!updatesIds[id];   // לשונית "עדכונים": מערכות ושירותים, לא אדם שמחכה לתשובה
    if (CONFIG.on.sales && (w = hasWord_(text, CONFIG.salesWords))) { item.why = w; out.sales.push(item); return; }
    if (CONFIG.on.reply && !sys && (toMe || msgs.length > 1) && looksLikeAsk_(text)) {
      item.why = 'שאלו אותך משהו';
      item.wait = now - lastMsg.getDate();
      out.reply.push(item); return;
    }
    if (CONFIG.on.money && (w = hasWord_(text, CONFIG.moneyWords))) { item.why = w; item.amount = amountOf_(subj + ' ' + top); out.money.push(item); return; }
    if (CONFIG.on.important && (isVip_(addr) || (importantIds[id] && !sys) || (w = hasWord_(text, CONFIG.importantWords)))) {
      item.why = isVip_(addr) ? 'ברשימה שלך' : (w || 'גוגל סימן כחשוב'); out.important.push(item); return;
    }
    if (CONFIG.on.ship && (w = hasWord_(text, CONFIG.shipWords))) { item.why = w; out.ship.push(item); return; }
    var thanks = body.length < 80 && /^(תודה|תודה רבה|תודה!|קיבלתי|מעולה|סגור|thanks|thank you|got it)/i.test(body);
    if (CONFIG.on.reply && toMe && !sys && !thanks) { item.why = 'נשלח ישירות אליך'; item.wait = now - lastMsg.getDate(); out.reply.push(item); return; }
    out.other = out.other || []; out.other.push(item);
  });

  // מיילים מסיכומים קודמים שעדיין לא ענית עליהם – חוזרים כל עוד לא ענית (עד שבועיים)
  if (CONFIG.on.reply && CONFIG.labels) {
    var seen = {};
    out.reply.forEach(function (x) { seen[x.id] = 1; });
    var oldQ = 'in:inbox label:' + labelQuery_(String(CONFIG.labelRoot || 'סיכום') + '/' + CATS.reply.name) +
      ' before:' + after + ' newer_than:14d -label:' + labelQuery_(understoodName_());
    var old = [], oldMsgs = [];
    try { old = GmailApp.search(oldQ, 0, 40); oldMsgs = GmailApp.getMessagesForThreads(old); } catch (e) {}
    old.forEach(function (t, ti) {
      if (seen[t.getId()]) return;
      var msgs = oldMsgs[ti] || t.getMessages(), last = msgs[msgs.length - 1];
      if (me[addrOf_(last.getFrom())]) return;               // כבר ענית
      var lastTop = bodyTop_(last);
      if (isBulk_(last, addrOf_(last.getFrom()), lastTop)) return;   // דיוור שסומן בטעות בגרסה קודמת
      out.reply.push({ id: t.getId(), subj: t.getFirstMessageSubject() || '(בלי נושא)', who: nameOf_(last.getFrom()),
        addr: addrOf_(last.getFrom()), snip: snippet_(last, lastTop), date: last.getDate(), thread: t,
        why: 'עדיין לא ענית', wait: now - last.getDate(), old: true });
    });
  }

  if (CONFIG.on.follow) out.follow = followUps_(me, now);
  ['sales', 'reply', 'important', 'money', 'ship'].forEach(function (k) { out[k] = dedupe_(out[k]); });
  out.reply.sort(function (a, b) { return (b.wait || 0) - (a.wait || 0); });
  out.vendors = vendors_(out.money);
  return out;
}

/** אותו נושא מאותו שולח בכמה שיחות (למשל אותה תשובה שהגיעה פעמיים) – שורה אחת עם "×2". */
function dedupe_(list) {
  var seen = {}, res = [];
  (list || []).forEach(function (it) {
    var k = (it.addr || '') + '|' + subjKey_(it.subj);
    var g = seen[k];
    if (!g) { seen[k] = it; res.push(it); return; }
    g.n = (g.n || 1) + 1;
    (g.more = g.more || []).push(it.thread);
    if (it.date > g.date) g.date = it.date;
  });
  res.forEach(function (it) { if (it.n > 1) it.why = (it.why ? it.why + ' · ' : '') + '×' + it.n; });
  return res;
}

/** חשבוניות לפי ספק: כמה, וכמה כסף בסך הכול (לכל מטבע בנפרד). */
function vendors_(money) {
  var by = {}, order = [];
  (money || []).forEach(function (it) {
    var k = it.who || it.addr;
    var v = by[k];
    if (!v) { v = by[k] = { who: k, n: 0, sums: {} }; order.push(v); }
    v.n += it.n || 1;
    if (it.amount) v.sums[it.amount.cur] = (v.sums[it.amount.cur] || 0) + it.amount.val * (it.n || 1);
  });
  order.sort(function (a, b) { return b.n - a.n; });
  return order;
}
function vendorText_(v) {
  var s = Object.keys(v.sums).map(function (c) {
    var x = Math.round(v.sums[c] * 100) / 100;
    return (c === '$' || c === '€' || c === '£') ? c + x : x + ' ' + c;
  }).join(' + ');
  return v.who + ': ' + v.n + (v.n === 1 ? ' מסמך' : ' מסמכים') + (s ? ', ' + s : '');
}

function looksLikeAsk_(text) {
  return /\?|אפשר|תוכל|תוכלי|תוכלו|בבקשה|נא ל|מחכה ל|אשמח ל|תעדכן|תעדכני|תחזור|תחזרי|תשלח|תשלחי|תאשר|תאשרי|מה דעתך|מתי|האם|please|could you|can you|let me know/i.test(text);
}

/** מיילים ששלחתי לפני X ימים ואין עליהם תשובה. */
function followUps_(me, now) {
  var days = Math.max(1, Number(CONFIG.followDays) || 3);
  var list = [];
  var threads = [];
  var all = [];
  try {
    // -to:me: בלי התראות ששלחת לעצמך (הן ממלאות את הרשימה ומסתירות את המיילים האמיתיים)
    threads = GmailApp.search('in:sent -to:me older_than:' + days + 'd newer_than:' + (days + 14) + 'd', 0, 100);
    all = GmailApp.getMessagesForThreads(threads);
  } catch (e) {}
  var bySubj = {};
  threads.forEach(function (t, ti) {
    var msgs = all[ti] || t.getMessages();
    var last = msgs[msgs.length - 1];
    if (!me[addrOf_(last.getFrom())]) return;            // ענו לי
    var toList = String(last.getTo() || '');
    if (!toList || /no-?reply/i.test(toList)) return;
    if (me[addrOf_(toList.split(',')[0])]) return;         // שלחתי לעצמי
    var subj = t.getFirstMessageSubject() || '(בלי נושא)';
    if (/^\s*(בדיקה|ניסיון|טסט|test(ing)?)\b/i.test(subj)) return;   // מייל בדיקה
    // נשלח לכתובת אחרת שלך שמעבירה חזרה לתיבה הזאת (למשל כתובת של האתר) – זה לא מישהו שלא ענה
    try { if (t.isInInbox() && msgs.every(function (x) { return me[addrOf_(x.getFrom())]; })) return; } catch (e) {}
    var who = nameOf_(toList.split(',')[0]);
    // אותו מייל לכמה נמענים (למשל פנייה לכמה עיתונים) – שורה אחת
    var g = bySubj[subj];
    if (g) { g.n++; if (g.names.length < 3) g.names.push(who); return; }
    g = bySubj[subj] = { id: t.getId(), subj: subj, who: who, names: [who], n: 1,
      addr: addrOf_(toList.split(',')[0]), last: last, date: last.getDate(), thread: t, wait: now - last.getDate() };
    list.push(g);
  });
  // הכי חדשים קודם: על מה ששלחת לפני כמה ימים עוד כדאי להזכיר; מה שמלפני שבועיים כבר פחות
  list.sort(function (a, b) { return a.wait - b.wait; });
  list = list.slice(0, MAX_SHOW);
  list.forEach(function (g) {
    if (g.n > 1) g.who = g.names.join(', ') + (g.n > g.names.length ? ' ועוד ' + (g.n - g.names.length) : '');
    g.why = 'שלחת ' + ago_(g.date, now) + (g.n > 1 ? ' ל-' + g.n + ' נמענים' : '');
    g.snip = snippet_(g.last);
    delete g.last; delete g.names;
  });
  return list;
}

function countShown_(d) {
  var n = 0;
  CONFIG.order.forEach(function (k) { if (k !== 'promo' && CONFIG.on[k]) n += (d[k] || []).length; });
  return n;
}

function esc_(s) { return String(s || '').replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
function link_(id) { return 'https://mail.google.com/mail/u/0/#all/' + id; }
function fmtDate_(d) { return Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd.MM HH:mm'); }
function freqText_() {
  if (CONFIG.freq === 'weekly') return 'פעם בשבוע';
  if (CONFIG.freq === 'every') return 'כל ' + CONFIG.everyDays + ' ימים';
  return 'כל בוקר';
}
function subject_(d, now) {
  var parts = [];
  if (d.sales.length) parts.push(d.sales.length + ' מכירות');
  if (d.reply.length) parts.push(d.reply.length + ' לענות');
  if (d.important.length) parts.push(d.important.length + ' חשובים');
  return '📬 סיכום המיילים · ' + Utilities.formatDate(now, Session.getScriptTimeZone(), 'dd.MM') +
    (parts.length ? ' · ' + parts.join(', ') : ' · אין משהו דחוף');
}

function renderPlain_(d, since, now) {
  var s = 'סיכום המיילים שלך – ' + d.total + ' שיחות חדשות מאז ' + fmtDate_(since) + '\n\n';
  CONFIG.order.forEach(function (k) {
    if (!CONFIG.on[k] || !CATS[k]) return;
    if (k === 'promo') { if (d.promoCount) s += CATS.promo.name + ': ' + d.promoCount + '\n\n'; return; }
    var list = d[k] || [];
    if (!list.length) return;
    s += CATS[k].name + ' (' + list.length + ')\n';
    if (k === 'money' && (d.vendors || []).length) s += 'לפי ספק: ' + d.vendors.map(vendorText_).join(' · ') + '\n';
    list.slice(0, MAX_SHOW).forEach(function (it) { s += '- ' + it.subj + ' | ' + it.who + (it.why ? ' | ' + it.why : '') + ' | ' + link_(it.id) + '\n'; });
    s += '\n';
  });
  return s + SITE;
}

function applyLabels_(d) {
  var root = String(CONFIG.labelRoot || 'סיכום');
  Object.keys(CATS).forEach(function (k) {
    if (k === 'follow' || !CONFIG.on[k]) return;
    var list = d[k] || [];
    if (!list.length) return;
    var lb = label_(root + '/' + CATS[k].name);
    var ths = [];
    list.forEach(function (x) { ths.push(x.thread); (x.more || []).forEach(function (t) { ths.push(t); }); });
    for (var i = 0; i < ths.length; i += 100) lb.addToThreads(ths.slice(i, i + 100));
  });
  // כללים אישיים
  var byLabel = {};
  Object.keys(CATS).concat(['other']).forEach(function (k) {
    (d[k] || []).forEach(function (it) { (it.rules || []).forEach(function (l) { (byLabel[l] = byLabel[l] || []).push(it.thread); }); });
  });
  Object.keys(byLabel).forEach(function (l) { label_(root + '/' + l).addToThreads(byLabel[l].slice(0, 100)); });
}

/** משימות בגוגל משימות (Google Tasks). צריך להוסיף את השירות Tasks בעורך – ראו בדף הכלי. */
function addTasks_(d) {
  if (typeof Tasks === 'undefined') { Logger.log('כדי להוסיף משימות: בעורך, שירותים (+) ← Google Tasks API ← הוספה.'); return; }
  var listName = 'מיילים לטיפול', listId = null;
  var lists = Tasks.Tasklists.list({ maxResults: 100 }).items || [];
  lists.forEach(function (l) { if (l.title === listName) listId = l.id; });
  if (!listId) listId = Tasks.Tasklists.insert({ title: listName }).id;
  var props = PropertiesService.getUserProperties();
  var done = JSON.parse(props.getProperty('taskIds') || '{}');
  var due = new Date(); due.setHours(0, 0, 0, 0);
  ['sales', 'reply', 'important', 'money'].forEach(function (k) {
    if (!CONFIG.on[k]) return;
    (d[k] || []).slice(0, 10).forEach(function (it) {
      if (done[it.id]) return;
      var verb = k === 'reply' ? 'לענות ל-' : k === 'sales' ? 'לחזור ל-' : k === 'money' ? 'לבדוק תשלום: ' : 'לקרוא: ';
      Tasks.Tasks.insert({ title: verb + it.who + ' – ' + it.subj, notes: (it.snip || '') + '\n' + link_(it.id), due: due.toISOString() }, listId);
      done[it.id] = Date.now();
    });
  });
  // שומרים רק 30 יום אחורה
  var cut = Date.now() - 30 * 864e5;
  Object.keys(done).forEach(function (k) { if (done[k] < cut) delete done[k]; });
  props.setProperty('taskIds', JSON.stringify(done));
}

/** שם התווית ל"הבנתי": שיחה עם התווית הזאת לא חוזרת בסיכום. אפשר להוסיף אותה גם ידנית, בג'ימייל בטלפון: ⋮ ← שינוי תוויות. */
function understoodName_() { return String(CONFIG.labelRoot || 'סיכום') + '/הבנתי'; }

/** כתובת המייל שלך (אליה נשלח מייל ה"הבנתי"). */
function myAddr_() {
  var a = '';
  try { a = Session.getEffectiveUser().getEmail() || ''; } catch (e) {}
  if (!a) { try { a = Session.getActiveUser().getEmail() || ''; } catch (e) {} }
  return a;
}

/** קישור "✓ הבנתי": פותח מייל מוכן אליך. שולחים אותו, ובסיכום הבא (או בהרצת runNow) השורה נעלמת. */
function understoodLink_(id) {
  var a = myAddr_();
  if (!a || !id) return '';
  return 'mailto:' + encodeURIComponent(a) + '?subject=' + encodeURIComponent('הבנתי ' + id) +
    '&body=' + encodeURIComponent('רק לשלוח. הסיכום יסמן את השיחה כ"הבנתי" ולא יחזור עליה.');
}

/** מעבד מיילי "הבנתי": מסמן את השיחה בתווית, ומעביר את מייל ההוראה לארכיון כנקרא. */
function processUnderstood_() {
  var cmds = [];
  try { cmds = GmailApp.search('subject:הבנתי newer_than:30d from:me to:me', 0, 100); } catch (e) { return; }
  if (!cmds.length) return;
  var lb = label_(understoodName_()), done = [];
  cmds.forEach(function (c) {
    var m = /הבנתי\s+([0-9a-f]{10,20})/i.exec(c.getFirstMessageSubject() || '');
    if (!m) return;
    try {
      var t = GmailApp.getThreadById(m[1]);
      if (t) { lb.addToThread(t); var rl = GmailApp.getUserLabelByName(String(CONFIG.labelRoot || 'סיכום') + '/' + CATS.reply.name); if (rl) rl.removeFromThread(t); }
    } catch (e) {}
    done.push(c);
  });
  for (var i = 0; i < done.length; i += 100) { GmailApp.markThreadsRead(done.slice(i, i + 100)); GmailApp.moveThreadsToArchive(done.slice(i, i + 100)); }
}

/** תווית בחיפוש של ג'ימייל: רווחים ו-/ הופכים למקף. */

}
